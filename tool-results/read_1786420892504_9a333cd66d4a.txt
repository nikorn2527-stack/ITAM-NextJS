    77→    77→/**
    78→    78→ * Animated count-up hook — tweens from the previous value to the next
    79→    79→ * over 500ms using requestAnimationFrame.
    80→    80→ */
    81→    81→function useCountUp(target: number, duration = 500) {
    82→    82→  const [display, setDisplay] = React.useState(target)
    83→    83→  const fromRef = React.useRef(target)
    84→    84→  const rafRef = React.useRef<number | null>(null)
    85→    85→  const startRef = React.useRef<number | null>(null)
    86→    86→
    87→    87→  React.useEffect(() => {
    88→    88→    if (typeof window === 'undefined') return
    89→    89→    const from = fromRef.current
    90→    90→    if (from === target) {
    91→    91→      setDisplay(target)
    92→    92→      return
    93→    93→    }
    94→    94→    startRef.current = null
    95→    95→    const step = (ts: number) => {
    96→    96→      if (startRef.current === null) startRef.current = ts
    97→    97→      const elapsed = ts - startRef.current
    98→    98→      const t = Math.min(1, elapsed / duration)
    99→    99→      // easeOutCubic
   100→   100→      const eased = 1 - Math.pow(1 - t, 3)
   101→   101→      const next = Math.round(from + (target - from) * eased)
   102→   102→      setDisplay(next)
   103→   103→      if (t < 1) {
   104→   104→        rafRef.current = window.requestAnimationFrame(step)
   105→   105→      } else {
   106→   106→        fromRef.current = target
   107→   107→        setDisplay(target)
   108→   108→      }
   109→   109→    }
   110→   110→    rafRef.current = window.requestAnimationFrame(step)
   111→   111→    return () => {
   112→   112→      if (rafRef.current !== null) {
   113→   113→        window.cancelAnimationFrame(rafRef.current)
   114→   114→      }
   115→   115→      // remember where we landed so the next tween starts smoothly
   116→   116→      fromRef.current = display
   117→   117→    }
   118→   118→  }, [target, duration])
   119→   119→
   120→   120→  return display
   121→   121→}
