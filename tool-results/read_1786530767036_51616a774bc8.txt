   337→// 6b) Meter_Report_Snapshots — Immutable billing snapshots
   338→//     Aligned with Apps Script MeterSnapshotService.gs (commit 67f8e54)
   339→//     Created automatically when a cycle is CLOSED. Contains frozen copies
   340→//     of all meter readings for that cycle month, with SHA-256 hash for
   341→//     tamper detection. Prevents retroactive edits from changing historical bills.
   342→// ============================================================
   343→model MeterReportSnapshot {
   344→  id              String   @id @default(cuid())
   345→  snapshotId      String   @unique @map("Snapshot_ID") // e.g. MRS-202608-R1-20260812150000
   346→  cycleMonth      String   @map("Cycle_Month") // e.g. 2026-08
   347→  revision        Int      @default(1) @map("Revision") // increments on amendments
   348→  status          String   @default("ACTIVE") @map("Status") // ACTIVE | SUPERSEDED
   349→  ruleVersion     String?  @map("Rule_Version") // calc logic version, e.g. "v2-initial-baseline"
   350→  rowCount        Int      @default(0) @map("Row_Count")
   351→  totalPagesBw    Int      @default(0) @map("Total_Pages_BW")
   352→  totalPagesColor Int      @default(0) @map("Total_Pages_Color")
   353→  totalCost       Float    @default(0) @map("Total_Cost")
   354→  contentHash     String   @map("Content_Hash") // SHA-256 hex (64 chars)
   355→  createdBy       String?  @map("Created_By") // user email
   356→  createdAt       DateTime @default(now()) @map("created_at")
   357→
   358→  // Relation — one snapshot has many frozen rows
   359→  rows            MeterReportSnapshotRow[]
   360→  amendments      MeterReportAmendment[]
   361→
   362→  @@index([cycleMonth])
   363→  @@index([status])
   364→  @@index([cycleMonth, revision])
   365→  @@map("meter_report_snapshots")
   366→}
