    94→// ============ Search debounce hook (200ms) ============
    95→function useDebounced<T>(value: T, delayMs = 200): [T, boolean] {
    96→  const [debounced, setDebounced] = React.useState<T>(value)
    97→  const [isPending, setIsPending] = React.useState(false)
    98→  React.useEffect(() => {
    99→    setIsPending(true)
   100→    const t = setTimeout(() => {
   101→      setDebounced(value)
   102→      setIsPending(false)
   103→    }, delayMs)
   104→    return () => clearTimeout(t)
   105→  }, [value, delayMs])
   106→  return [debounced, isPending]
   107→}

   109→// ============ Highlight matched text ============
   110→function Highlight({ text, query }: { text: string; query: string }) {
   111→  if (!query.trim()) return <>{text || '—'}</>
   112→  const q = query.trim()
   113→  // Escape regex special chars
   114→  const esc = q.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
   115→  const parts = (text || '').split(new RegExp(`(${esc})`, 'ig'))
   116→  return (
   117→    <>
   118→      {parts.map((p, i) =>
   119→        p.toLowerCase() === q.toLowerCase()
   120→          ? <mark key={i} className="rounded bg-orange-200 px-0.5 text-slate-900 dark:bg-orange-500/60 dark:text-white">{p}</mark>
   121→          : <React.Fragment key={i}>{p}</React.Fragment>,
   122→      )}
   123→    </>
   124→  )
   125→}

   127→// ============ Recent searches (localStorage) ============
   128→const RECENT_KEY = 'itam-recent-searches'
   129→function loadRecent(): string[] {
   130→  if (typeof window === 'undefined') return []
   131→  try {
   132→    const raw = window.localStorage.getItem(RECENT_KEY)
   133→    if (!raw) return []
   134→    const arr = JSON.parse(raw)
   135→    return Array.isArray(arr) ? arr.filter((x) => typeof x === 'string').slice(0, 6) : []
   136→  } catch {
   137→    return []
   138→  }
   139→}
   140→function saveRecent(q: string) {
   141→  if (typeof window === 'undefined' || !q.trim()) return
   142→  try {
   143→    const cur = loadRecent()
   144→    const next = [q.trim(), ...cur.filter((x) => x !== q.trim())].slice(0, 6)
   145→    window.localStorage.setItem(RECENT_KEY, JSON.stringify(next))
   146→  } catch { /* ignore */ }
   147→}
