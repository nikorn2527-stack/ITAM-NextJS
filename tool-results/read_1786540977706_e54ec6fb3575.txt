   198→   198→function formatDateTime(iso: string | null): string {
   199→   199→  if (!iso) return '—'
   200→   200→  try {
   201→   201→    return new Date(iso).toLocaleString('th-TH', {
   202→   202→      day: '2-digit',
   203→   203→      month: '2-digit',
   204→   204→      year: 'numeric',
   205→   205→      hour: '2-digit',
   206→   206→      minute: '2-digit',
   207→   207→    })
   208→   208→  } catch {
   209→   209→    return iso
   210→   210→  }
