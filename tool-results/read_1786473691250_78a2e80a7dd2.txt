    92→// ============ Search debounce hook (200ms) ============
    93→function useDebounced<T>(value: T, delayMs = 200): [T, boolean] {
    94→  const [debounced, setDebounced] = React.useState<T>(value)
    95→  const [isPending, setIsPending] = React.useState(false)
    96→  React.useEffect(() => {
    97→    setIsPending(true)
    98→    const t = setTimeout(() => {
    99→      setDebounced(value)
   100→      setIsPending(false)
   101→    }, delayMs)
   102→    return () => clearTimeout(t)
   103→  }, [value, delayMs])
   104→  return [debounced, isPending]
   105→}

   107→// ============ Highlight matched text ============
   108→function Highlight({ text, query }: { text: string; query: string }) {
   109→  if (!query.trim()) return <>{text || '—'}</>
   110→  const q = query.trim()
   111→  // Escape regex special chars
   112→  const esc = q.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
   113→  const parts = (text || '').split(new RegExp(`(${esc})`, 'ig'))
   114→  return (
   115→    <>
   116→      {parts.map((p, i) =>
   117→        p.toLowerCase() === q.toLowerCase()
   118→          ? <mark key={i} className="rounded bg-orange-200 px-0.5 text-slate-900 dark:bg-orange-500/60 dark:text-white">{p}</mark>
   119→          : <React.Fragment key={i}>{p}</React.Fragment>,
   120→      )}
   121→    </>
   122→  )
   123→}

   125→// ============ Recent searches (localStorage) ============
   126→const RECENT_KEY = 'itam-recent-searches'
   127→function loadRecent(): string[] {
   128→  if (typeof window === 'undefined') return []
   129→  try {
   130→    const raw = window.localStorage.getItem(RECENT_KEY)
   131→    if (!raw) return []
   132→    const arr = JSON.parse(raw)
   133→    return Array.isArray(arr) ? arr.filter((x) => typeof x === 'string').slice(0, 6) : []
   134→  } catch {
   135→    return []
   136→  }
   137→}
   138→function saveRecent(q: string) {
   139→  if (typeof window === 'undefined' || !q.trim()) return
   140→  try {
   141→    const cur = loadRecent()
   142→    const next = [q.trim(), ...cur.filter((x) => x !== q.trim())].slice(0, 6)
   143→    window.localStorage.setItem(RECENT_KEY, JSON.stringify(next))
   144→  } catch { /* ignore */ }
   145→}
