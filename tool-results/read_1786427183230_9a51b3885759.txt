    92→function formatThaiDateTime(iso: string): string {
    93→  try {
    94→    return new Date(iso).toLocaleString('th-TH', {
    95→      day: '2-digit',
    96→      month: '2-digit',
    97→      year: 'numeric',
    98→      hour: '2-digit',
    99→      minute: '2-digit',
   100→    })
   101→  } catch {
   102→    return iso
   103→  }
   104→}
   105→
   106→function todayISO(): string {
   107→  return new Date().toISOString().slice(0, 10)
