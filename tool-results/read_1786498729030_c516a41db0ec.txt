    44→function genTemplateId(): string {

    48→function makeElement(type: StickerElementType): StickerElement {
    49→  const base = {
    50→    id: genElementId(),
    51→    x: 5, y: 5, width: 20, height: 6,
    52→    opacity: 1, zIndex: 1, rotation: 0,
    53→  }
    54→  if (type === 'text') {
    55→    return { ...base, type, width: 40, height: 5, content: 'ข้อความใหม่', fontSize: 7, fontWeight: 500, color: '#1e293b', align: 'left' }
    56→  }
    57→  if (type === 'image') {
    58→    return { ...base, type, width: 20, height: 20, content: '', source: '' }
    59→  }
    60→  if (type === 'qr') {
    61→    return { ...base, type, width: 15, height: 15, content: '{{AssetNo}}' }
    62→  }
    63→  // rect
    64→  return { ...base, type, width: 30, height: 5, background: '#f97316', border: 'none', borderRadius: 0 }
    65→}

    71→function WorkspaceElement({
    72→  el,
    73→  selected,
    74→  exceedsBounds,
    75→  onMouseDown,
    76→  onResizeMouseDown,
    77→}: {
    78→  el: StickerElement
    79→  selected: boolean
    80→  exceedsBounds: boolean
    81→  onMouseDown: (e: React.MouseEvent, id: string) => void
    82→  onResizeMouseDown: (e: React.MouseEvent, id: string) => void
    83→}) {
    84→  const style: React.CSSProperties = {
    85→    position: 'absolute',
    86→    left: `${el.x}mm`,
    87→    top: `${el.y}mm`,
    88→    width: `${el.width}mm`,
    89→    height: `${el.height}mm`,
    90→    opacity: el.opacity ?? 1,
    91→    zIndex: el.zIndex ?? 0,
    92→    transform: el.rotation ? `rotate(${el.rotation}deg)` : undefined,
    93→    cursor: 'move',
    94→    userSelect: 'none',
    95→  }
    96→
    97→  let content: React.ReactNode = null
    98→  if (el.type === 'text') {
    99→    Object.assign(style, {
   100→      fontSize: `${el.fontSize}pt`,
   101→      fontWeight: el.fontWeight,
   102→      color: el.color,
   103→      textAlign: el.align,
   104→      lineHeight: 1.15,
   105→      whiteSpace: 'pre-wrap',
   106→      wordBreak: 'break-word',
   107→      overflow: 'hidden',
   108→      fontFamily: '"Sukhumvit Set","Noto Sans Thai","Tahoma","Segoe UI",sans-serif',
   109→    })
   110→    content = el.content || '​'
   111→  } else if (el.type === 'rect') {
   112→    Object.assign(style, {
   113→      background: el.background || 'transparent',
   114→      border: el.border || 'none',
   115→      borderRadius: el.borderRadius ? `${el.borderRadius}mm` : undefined,
   116→    })
   117→  } else if (el.type === 'image') {
   118→    Object.assign(style, { objectFit: 'contain' as const })
   119→    const src = el.source || el.content
   120→    if (src) {
   121→      return (
   122→        <img
   123→          src={src}
   124→          alt=""
   125→          style={style}
   126→          draggable={false}
   127→          onMouseDown={(e) => onMouseDown(e, el.id)}
   128→        />
   129→      )
   130→    }
   131→    Object.assign(style, {
   132→      background: '#f1f5f9', border: '1px dashed #cbd5e1',
   133→      display: 'flex', alignItems: 'center', justifyContent: 'center',
   134→      color: '#94a3b8', fontSize: '6pt',
   135→    })
   136→    content = '(image)'
   137→  } else if (el.type === 'qr') {
   138→    Object.assign(style, {
   139→      background: '#f8fafc', border: '1px solid #e2e8f0',
   140→      display: 'flex', alignItems: 'center', justifyContent: 'center',
   141→      color: '#94a3b8', fontSize: '5pt',
   142→    })
   143→    content = 'QR'
   144→  }
   145→
   146→  return (
   147→    <div
   148→      style={style}
   149→      onMouseDown={(e) => onMouseDown(e, el.id)}
   150→      className={selected ? 'ring-2 ring-[#f97316] ring-offset-1' : exceedsBounds ? 'ring-1 ring-orange-400' : ''}
   151→    >
   152→      {content}
   153→      {selected && (
   154→        <div
   155→          onMouseDown={(e) => onResizeMouseDown(e, el.id)}
   156→          style={{
   157→            position: 'absolute',
   158→            right: -4, bottom: -4,
   159→            width: 10, height: 10,
   160→            background: '#f97316',
   161→            border: '2px solid #ffffff',
   162→            borderRadius: 2,
   163→            cursor: 'nwse-resize',
   164→          }}
   165→        />
   166→      )}
   167→    </div>
   168→  )
   169→}

   336→  const dragStateRef = React.useRef<{
   337→    mode: 'move' | 'resize'

   348→  function calcPxPerMm(): number {
   349→    const ws = workspaceRef.current
   350→    if (!ws) return 3.78
   351→    const rect = ws.getBoundingClientRect()
   352→    // Workspace has 30mm padding all around, so usable width is total - 60mm
   353→    const widthMm = (draft?.canvas.width ?? 75.2) + 60
   354→    return rect.width / widthMm
   355→  }

   357→  function onElementMouseDown(e: React.MouseEvent, id: string) {
   358→    if (!draft) return
   359→    e.stopPropagation()
   360→    setSelectedElId(id)
   361→    const el = draft.elements.find((x) => x.id === id)
   362→    if (!el) return
   363→    dragStateRef.current = {
   364→      mode: 'move',
   365→      elId: id,
   366→      startClientX: e.clientX,
   367→      startClientY: e.clientY,
   368→      startElX: el.x,
   369→      startElY: el.y,
   370→      startElW: el.width,
   371→      startElH: el.height,
   372→      pxPerMm: calcPxPerMm(),
   373→    }
   374→  }

   376→  function onResizeMouseDown(e: React.MouseEvent, id: string) {
   377→    if (!draft) return
   378→    e.stopPropagation()
   379→    e.preventDefault()
   380→    const el = draft.elements.find((x) => x.id === id)
   381→    if (!el) return
   382→    dragStateRef.current = {
   383→      mode: 'resize',
   384→      elId: id,
   385→      startClientX: e.clientX,
   386→      startClientY: e.clientY,
   387→      startElX: el.x,
   388→      startElY: el.y,
   389→      startElW: el.width,
   390→      startElH: el.height,
   391→      pxPerMm: calcPxPerMm(),
   392→    }
   393→  }

   395→  React.useEffect(() => {
   396→    function onMouseMove(e: MouseEvent) {
   397→      const ds = dragStateRef.current
   398→      if (!ds || !draft) return
   399→      const dxPx = e.clientX - ds.startClientX
   400→      const dyPx = e.clientY - ds.startClientY
   401→      const dxMm = dxPx / ds.pxPerMm
   402→      const dyMm = dyPx / ds.pxPerMm
   403→      setDraft((cur) => {
   404→        if (!cur) return cur
   405→        return {
   406→          ...cur,
   407→          elements: cur.elements.map((el) => {
   408→            if (el.id !== ds.elId) return el
   409→            if (ds.mode === 'move') {
   410→              let newX = ds.startElX + dxMm
   411→              let newY = ds.startElY + dyMm
   412→              // Snap to 0.5mm grid for cleaner UX
   413→              newX = Math.round(newX * 2) / 2
   414→              newY = Math.round(newY * 2) / 2
   415→              return { ...el, x: newX, y: newY }
   416→            }
   417→            // resize
   418→            let newW = ds.startElW + dxMm
   419→            let newH = ds.startElH + dyMm
   420→            newW = Math.max(2, Math.round(newW * 2) / 2)
   421→            newH = Math.max(2, Math.round(newH * 2) / 2)
   422→            return { ...el, width: newW, height: newH }
   423→          }),
   424→        }
   425→      })
   426→    }
   427→    function onMouseUp() {
   428→      dragStateRef.current = null
   429→    }
   430→    window.addEventListener('mousemove', onMouseMove)
   431→    window.addEventListener('mouseup', onMouseUp)
   432→    return () => {
   433→      window.removeEventListener('mousemove', onMouseMove)
   434→      window.removeEventListener('mouseup', onMouseUp)
   435→    }

   450→        elements: draft.elements.filter((el) => el.id !== selectedElId),
   451→      })
   452→      setSelectedElId(null)
   453→    }
   454→    window.addEventListener('keydown', onKey)
   455→    return () => window.removeEventListener('keydown', onKey)
   456→  }, [selectedElId, draft])

   492→  function duplicateTemplate(t: StickerTemplate) {
   493→    const dup: StickerTemplate = {
   494→      ...structuredCloneSafe(t),
   495→      id: genTemplateId(),
   496→      name: `${t.name} (สำเนา)`,
   497→      isDefault: false,
   498→    }
   499→    // Reassign element ids to avoid collisions
   500→    dup.elements = dup.elements.map((el) => ({ ...el, id: genElementId() }))
   501→    createMutation.mutate(dup.name, {
   502→      onSuccess: async (data) => {
   503→        // Now overwrite with the duplicated elements via PUT
   504→        await updateMutation.mutateAsync({ id: data.template.id, payload: dup })
   505→        setEditingId(data.template.id)
   506→        setDraft(structuredCloneSafe(dup))
   507→        setSelectedElId(null)
   508→      },
   509→    })
   510→  }

   517→  async function previewTemplate() {
   518→    if (!draft) return
   519→    try {
   520→      // Find a real device assetNo to render with (preview with sample data)
   521→      let assetNo: string | null = null
   522→      try {
   523→        const devRes = await fetch('/api/itam/devices?limit=1')
   524→        if (devRes.ok) {
   525→          const devJson = (await devRes.json()) as { devices: { assetNo: string }[] }
   526→          assetNo = devJson.devices[0]?.assetNo ?? null
   527→        }
   528→      } catch {
   529→        /* ignore — fall through to sample */
   530→      }
   531→      if (!assetNo) assetNo = SAMPLE_DEVICE.assetNo
   532→
   533→      const res = await fetch('/api/itam/sticker/render', {
   534→        method: 'POST',
   535→        headers: { 'Content-Type': 'application/json' },
   536→        body: JSON.stringify({ assetNo, templateId: draft.id }),
   537→      })
   538→      if (!res.ok) {
   539→        // Fallback: try with bulk-render
   540→        const bulk = await fetch('/api/itam/sticker/bulk-render', {
   541→          method: 'POST',
   542→          headers: { 'Content-Type': 'application/json' },
   543→          body: JSON.stringify({ assetNos: [assetNo], templateId: draft.id }),
   544→        })
   545→        if (!bulk.ok) throw new Error('Preview failed')
   546→        const j = await bulk.json() as { stickers: { html: string }[] }
   547→        setPreviewHtml(j.stickers[0]?.html ?? '')
   548→      } else {
   549→        const j = await res.json() as { html: string }
   550→        setPreviewHtml(j.html)
   551→      }

  1243→function NumInput({
  1244→  label, value, onChange, step = 1, min, max,
  1245→}: {
  1246→  label: string
  1247→  value: number
  1248→  onChange: (v: number) => void
  1249→  step?: number
  1250→  min?: number
  1251→  max?: number
  1252→}) {
  1253→  return (
  1254→    <div className="space-y-1.5">
  1255→      <Label className="text-xs">{label}</Label>
  1256→      <Input
  1257→        type="number"
  1258→        value={Number.isFinite(value) ? value : ''}
  1259→        onChange={(e) => {
  1260→          const n = Number(e.target.value)
  1261→          if (!Number.isFinite(n)) return
  1262→          onChange(n)
  1263→        }}
  1264→        step={step}
  1265→        min={min}
  1266→        max={max}
  1267→        className="text-xs dark:bg-slate-800 dark:border-slate-700"
  1268→      />
  1269→    </div>
  1270→  )
  1271→}

  1273→// Safe structured clone (with fallback for older browsers)
  1274→function structuredCloneSafe<T>(obj: T): T {
  1275→  if (typeof structuredClone === 'function') {
  1276→    try {
  1277→      return structuredClone(obj)
  1278→    } catch {
  1279→      /* fall through */
  1280→    }
  1281→  }
  1282→  return JSON.parse(JSON.stringify(obj)) as T
  1283→}
