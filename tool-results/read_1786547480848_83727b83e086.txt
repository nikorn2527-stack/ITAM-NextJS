    44→// 1) Asset Number Pattern — ข้อ 4: เลขทะเบียนกำหนดเองได้
    45→//    รูปแบบ: {prefix}-{seq:5} หรือ {dept:4}-{type:3}-{seq:3}-{year:2}

    50→model AssetNumberPattern {
    51→  id            String   @id @default(cuid())

    54→  description   String?
    55→  isActive      Boolean  @default(false) // ใช้ได้ทีละ 1 pattern
    56→  // ── ค่า default สำหรับแต่ละ segment ──
    57→  defaultPrefix String?  // ถ้า pattern มี {prefix}
    58→  seqPadding    Int      @default(5) // จำนวนหลักของเลขลำดับ
    59→  seqStart      Int      @default(1) // เลขเริ่มต้น
    60→  // ── Audit ──
    61→  createdAt     DateTime @default(now())
    62→  updatedAt     DateTime @updatedAt
    63→}

   397→  workOrderId     String?
   398→  workOrderNo     String?  // maps from WorkOrderNo — for CSV import
   399→  deviceId        String?
   400→  cost            Float?   // maps from TotalValue

   405→  txnDate         String   // maps from Date
   406→  performedBy     String?  // maps from PerformedBy
   407→  remark          String?  // maps from Remark
   408→  sourceKey       String?  // maps from source_key

   420→
   421→  stockItem       StockItem @relation(fields: [stockItemId], references: [id], onDelete: Cascade)
