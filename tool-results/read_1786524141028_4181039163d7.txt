    44→    44→function genTemplateId(): string {

    48→    48→function makeElement(type: StickerElementType): StickerElement {
    49→    49→  const base = {
    50→    50→    id: genElementId(),
    51→    51→    x: 5, y: 5, width: 20, height: 6,
    52→    52→    opacity: 1, zIndex: 1, rotation: 0,
    53→    53→  }
    54→    54→  if (type === 'text') {
    55→    55→    return { ...base, type, width: 40, height: 5, content: 'ข้อความใหม่', fontSize: 7, fontWeight: 500, color: '#1e293b', align: 'left' }
    56→    56→  }
    57→    57→  if (type === 'image') {
    58→    58→    return { ...base, type, width: 20, height: 20, content: '', source: '' }
    59→    59→  }
    60→    60→  if (type === 'qr') {
    61→    61→    return { ...base, type, width: 15, height: 15, content: '{{AssetNo}}' }
    62→    62→  }
    63→    63→  // rect
    64→    64→  return { ...base, type, width: 30, height: 5, background: '#f97316', border: 'none', borderRadius: 0 }
    65→    65→}

    71→    71→function WorkspaceElement({
    72→    72→  el,
    73→    73→  selected,
    74→    74→  exceedsBounds,
    75→    75→  onMouseDown,
    76→    76→  onResizeMouseDown,
    77→    77→}: {
    78→    78→  el: StickerElement
    79→    79→  selected: boolean
    80→    80→  exceedsBounds: boolean
    81→    81→  onMouseDown: (e: React.MouseEvent, id: string) => void
    82→    82→  onResizeMouseDown: (e: React.MouseEvent, id: string) => void
    83→    83→}) {
    84→    84→  const style: React.CSSProperties = {
    85→    85→    position: 'absolute',
    86→    86→    left: `${el.x}mm`,
    87→    87→    top: `${el.y}mm`,
    88→    88→    width: `${el.width}mm`,
    89→    89→    height: `${el.height}mm`,
    90→    90→    opacity: el.opacity ?? 1,
    91→    91→    zIndex: el.zIndex ?? 0,
    92→    92→    transform: el.rotation ? `rotate(${el.rotation}deg)` : undefined,
    93→    93→    cursor: 'move',
    94→    94→    userSelect: 'none',
    95→    95→  }
    96→    96→
    97→    97→  let content: React.ReactNode = null
    98→    98→  if (el.type === 'text') {
    99→    99→    Object.assign(style, {
   100→   100→      fontSize: `${el.fontSize}pt`,
   101→   101→      fontWeight: el.fontWeight,
   102→   102→      color: el.color,
   103→   103→      textAlign: el.align,
   104→   104→      lineHeight: 1.15,
   105→   105→      whiteSpace: 'pre-wrap',
   106→   106→      wordBreak: 'break-word',
   107→   107→      overflow: 'hidden',
   108→   108→      fontFamily: '"Sukhumvit Set","Noto Sans Thai","Tahoma","Segoe UI",sans-serif',
   109→   109→    })
   110→   110→    content = el.content || '​'
   111→   111→  } else if (el.type === 'rect') {
   112→   112→    Object.assign(style, {
   113→   113→      background: el.background || 'transparent',
   114→   114→      border: el.border || 'none',
   115→   115→      borderRadius: el.borderRadius ? `${el.borderRadius}mm` : undefined,
   116→   116→    })
   117→   117→  } else if (el.type === 'image') {
   118→   118→    Object.assign(style, { objectFit: 'contain' as const })
   119→   119→    const src = el.source || el.content
   120→   120→    if (src) {
   121→   121→      return (
   122→   122→        <img
   123→   123→          src={src}
   124→   124→          alt=""
   125→   125→          style={style}
   126→   126→          draggable={false}
   127→   127→          onMouseDown={(e) => onMouseDown(e, el.id)}
   128→   128→        />
   129→   129→      )
   130→   130→    }
   131→   131→    Object.assign(style, {
   132→   132→      background: '#f1f5f9', border: '1px dashed #cbd5e1',
   133→   133→      display: 'flex', alignItems: 'center', justifyContent: 'center',
   134→   134→      color: '#94a3b8', fontSize: '6pt',
   135→   135→    })
   136→   136→    content = '(image)'
   137→   137→  } else if (el.type === 'qr') {
   138→   138→    Object.assign(style, {
   139→   139→      background: '#f8fafc', border: '1px solid #e2e8f0',
   140→   140→      display: 'flex', alignItems: 'center', justifyContent: 'center',
   141→   141→      color: '#94a3b8', fontSize: '5pt',
   142→   142→    })
   143→   143→    content = 'QR'
   144→   144→  }
   145→   145→
   146→   146→  return (
   147→   147→    <div
   148→   148→      style={style}
   149→   149→      onMouseDown={(e) => onMouseDown(e, el.id)}
   150→   150→      className={selected ? 'ring-2 ring-[#f97316] ring-offset-1' : exceedsBounds ? 'ring-1 ring-orange-400' : ''}
   151→   151→    >
   152→   152→      {content}
   153→   153→      {selected && (
   154→   154→        <div
   155→   155→          onMouseDown={(e) => onResizeMouseDown(e, el.id)}
   156→   156→          style={{
   157→   157→            position: 'absolute',
   158→   158→            right: -4, bottom: -4,
   159→   159→            width: 10, height: 10,
   160→   160→            background: '#f97316',
   161→   161→            border: '2px solid #ffffff',
   162→   162→            borderRadius: 2,
   163→   163→            cursor: 'nwse-resize',
   164→   164→          }}
   165→   165→        />
   166→   166→      )}
   167→   167→    </div>
   168→   168→  )
   169→   169→}

   343→   343→  // ── Drag logic ─────────────────────────────────────────────────────────
   344→   344→  const dragStateRef = React.useRef<{

   356→   356→  function calcPxPerMm(): number {
   357→   357→    const ws = workspaceRef.current
   358→   358→    if (!ws) return 3.78
   359→   359→    const rect = ws.getBoundingClientRect()
   360→   360→    // Workspace has 30mm padding all around, so usable width is total - 60mm
   361→   361→    const widthMm = (draft?.canvas.width ?? 75.2) + 60
   362→   362→    return rect.width / widthMm
   363→   363→  }

   372→   372→      mode: 'move',
   373→   373→      elId: id,
   374→   374→      startClientX: e.clientX,
   375→   375→      startClientY: e.clientY,
   376→   376→      startElX: el.x,
   377→   377→      startElY: el.y,
   378→   378→      startElW: el.width,
   379→   379→      startElH: el.height,
   380→   380→      pxPerMm: calcPxPerMm(),
   381→   381→    }
   382→   382→  }
   383→   383→
   384→   384→  function onResizeMouseDown(e: React.MouseEvent, id: string) {
   385→   385→    if (!draft) return
   386→   386→    e.stopPropagation()
   387→   387→    e.preventDefault()
   388→   388→    const el = draft.elements.find((x) => x.id === id)
   389→   389→    if (!el) return
   390→   390→    dragStateRef.current = {
   391→   391→      mode: 'resize',
   392→   392→      elId: id,
   393→   393→      startClientX: e.clientX,
   394→   394→      startClientY: e.clientY,
   395→   395→      startElX: el.x,
   396→   396→      startElY: el.y,
   397→   397→      startElW: el.width,
   398→   398→      startElH: el.height,
   399→   399→      pxPerMm: calcPxPerMm(),
   400→   400→    }
   401→   401→  }
   402→   402→
   403→   403→  React.useEffect(() => {
   404→   404→    function onMouseMove(e: MouseEvent) {
   405→   405→      const ds = dragStateRef.current
   406→   406→      if (!ds || !draft) return
   407→   407→      const dxPx = e.clientX - ds.startClientX
   408→   408→      const dyPx = e.clientY - ds.startClientY
   409→   409→      const dxMm = dxPx / ds.pxPerMm
   410→   410→      const dyMm = dyPx / ds.pxPerMm
   411→   411→      setDraft((cur) => {
   412→   412→        if (!cur) return cur
   413→   413→        return {
   414→   414→          ...cur,
   415→   415→          elements: cur.elements.map((el) => {
   416→   416→            if (el.id !== ds.elId) return el
   417→   417→            if (ds.mode === 'move') {
   418→   418→              let newX = ds.startElX + dxMm
   419→   419→              let newY = ds.startElY + dyMm
   420→   420→              // Snap to 0.5mm grid for cleaner UX
   421→   421→              newX = Math.round(newX * 2) / 2
   422→   422→              newY = Math.round(newY * 2) / 2
   423→   423→              return { ...el, x: newX, y: newY }
   424→   424→            }
   425→   425→            // resize
   426→   426→            let newW = ds.startElW + dxMm
   427→   427→            let newH = ds.startElH + dyMm
   428→   428→            newW = Math.max(2, Math.round(newW * 2) / 2)
   429→   429→            newH = Math.max(2, Math.round(newH * 2) / 2)
   430→   430→            return { ...el, width: newW, height: newH }
   431→   431→          }),
   432→   432→        }
   433→   433→      })
   434→   434→    }
   435→   435→    function onMouseUp() {
   436→   436→      dragStateRef.current = null
   437→   437→    }
   438→   438→    window.addEventListener('mousemove', onMouseMove)
   439→   439→    window.addEventListener('mouseup', onMouseUp)
   440→   440→    return () => {
   441→   441→      window.removeEventListener('mousemove', onMouseMove)
   442→   442→      window.removeEventListener('mouseup', onMouseUp)
   443→   443→    }
   444→   444→  }, [draft])
   445→   445→
   446→   446→  // Delete selected element with Delete/Backspace key (but not when typing)
   447→   447→  React.useEffect(() => {
   448→   448→    function onKey(e: KeyboardEvent) {
   449→   449→      if (e.key !== 'Delete' && e.key !== 'Backspace') return
