    45→//    รูปแบบ: {prefix}-{seq:5} หรือ {dept:4}-{type:3}-{seq:3}-{year:2}

    56→  // ── ค่า default สำหรับแต่ละ segment ──
    57→  defaultPrefix String?  // ถ้า pattern มี {prefix}

    63→}

   385→model StockTransaction {
   386→  id              String   @id @default(cuid())
   387→  txnNumber       String?  // STX-YYYYMMDD-NNN — maps from ReceiptNo (IN) or IssueNo (OUT)
   388→  stockItemId     String
   389→  productCode     String?  // maps from ProductCode — denormalized for CSV import
   390→  productName     String?  // maps from ProductName — denormalized
   391→  type            String   // IN | OUT | ADJUST
   392→  quantity        Int      // maps from Quantity
   393→  unit            String?  // maps from Unit
   394→  balanceAfter    Int      @default(0)
   395→  reason          String?
   396→  requester       String?  // maps from Requester (StockOut)
   397→  department      String?  // maps from Department (StockOut)
   398→  purpose         String?  // maps from Purpose (StockOut)
   399→  approver        String?  // maps from Approver (StockOut)
   400→  approvedAt      String?  // maps from ApprovedAt (StockOut)
   401→  workOrderId     String?
   402→  workOrderNo     String?  // maps from WorkOrderNo — for CSV import
   403→  deviceId        String?
   404→  cost            Float?   // maps from TotalValue
   405→  unitCost        Float?   // maps from UnitPrice
   406→  vendor          String?  // maps from Supplier
   407→  receiver        String?  // maps from Receiver (StockIn)
   408→  purchaseOrderNo String?  // maps from PurchaseOrderNo (StockIn)
   409→  txnDate         String   // maps from Date
   410→  performedBy     String?  // maps from PerformedBy
   411→  remark          String?  // maps from Remark
   412→  sourceKey       String?  // maps from source_key
   413→  processedFlag   String?  // maps from processed_flag
   414→  // ── Approval workflow (PART 1: Stock Pending Approval System) ──
   415→  // null = immediate (legacy/IN-OUT-ADJUST processed at once)
   416→  // PENDING = waiting for approval (stock NOT yet reduced)
   417→  // APPROVED = approved (stock reduced on approval)
   418→  // REJECTED = rejected (no stock change)
   419→  approvalStatus  String?  // null (immediate) | PENDING | APPROVED | REJECTED
   420→  approvalMode    String?  // manual | auto
   421→  autoApproveAt   String?  // ISO datetime for auto-approval
   422→  rejectReason    String?  // reason when rejected
   423→  createdAt       DateTime @default(now())
   424→
   425→  stockItem       StockItem @relation(fields: [stockItemId], references: [id], onDelete: Cascade)
   426→  device          Device?   @relation(fields: [deviceId], references: [id], onDelete: SetNull)
   427→}
