   161→/* ---------- App tab ---------- */
   162→function AppTab() {
   163→  const qc = useQueryClient()
   164→  const { data: settings, isLoading } = useQuery<Record<string, string>>({
   165→    queryKey: ['settings'],
   166→    queryFn: async () => {
   167→      const res = await fetch('/api/settings')
   168→      if (!res.ok) throw new Error('Failed to load settings')
   169→      const json = await res.json()
   170→      return json.settings as Record<string, string>
   171→    },
   172→  })
   173→  const { data: sites } = useQuery<Site[]>({
   174→    queryKey: ['sites'],
   175→    queryFn: async () => {
   176→      const res = await fetch('/api/sites')
   177→      if (!res.ok) throw new Error('Failed to load sites')
   178→      const json = await res.json()
   179→      return json.sites as Site[]
   180→    },
   181→  })
   182→
   183→  const [form, setForm] = React.useState<Record<string, string>>({})
   184→  const [saving, setSaving] = React.useState(false)
   185→
   186→  React.useEffect(() => {
   187→    if (settings) setForm({ ...settings })
   188→  }, [settings])
   189→
   190→  async function save() {
   191→    try {
   192→      setSaving(true)
   193→      const res = await fetch('/api/settings', {
   194→        method: 'PUT',
   195→        headers: { 'Content-Type': 'application/json' },
   196→        body: JSON.stringify(form),
   197→      })
   198→      if (!res.ok) throw new Error('Save failed')
   199→      toast.success('บันทึกการตั้งค่าแล้ว')
   200→      await qc.invalidateQueries({ queryKey: ['settings'] })
   201→    } catch (e) {
   202→      toast.error(e instanceof Error ? e.message : 'Save failed')
   203→    } finally {
   204→      setSaving(false)
   205→    }
   206→  }
   207→
   208→  if (isLoading) {
   209→    return (
   210→      <Card className="dark:border-slate-800 dark:bg-slate-900">
   211→        <CardContent className="p-6">
   212→          <Skeleton className="h-64 w-full dark:bg-slate-800" />
   213→        </CardContent>
   214→      </Card>
   215→    )
   216→  }
   217→
   218→  return (
   219→    <Card className="dark:border-slate-800 dark:bg-slate-900">
   220→      <CardHeader>
   221→        <CardTitle className="flex items-center gap-2 text-base text-slate-800 dark:text-slate-100">
   222→          <SettingsIcon className="h-4 w-4 text-[#f97316]" />
   223→          ตั้งค่าทั่วไป
   224→        </CardTitle>
   225→      </CardHeader>
   226→      <CardContent className="space-y-4">
   227→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   228→          <div className="space-y-1.5">
   229→            <Label className="text-xs font-medium text-slate-600 dark:text-slate-300">
   230→              ชื่อองค์กร
   231→            </Label>
   232→            <Input
   233→              value={form.orgName ?? ''}
   234→              onChange={(e) => setForm({ ...form, orgName: e.target.value })}
   235→              placeholder="PNG TEAM"
   236→              className="dark:bg-slate-800 dark:border-slate-700"
   237→            />
   238→          </div>
   239→          <div className="space-y-1.5">
   240→            <Label className="text-xs font-medium text-slate-600 dark:text-slate-300">
   241→              สาขาเริ่มต้น
   242→            </Label>
   243→            <Select
   244→              value={form.defaultSite ?? ''}
   245→              onValueChange={(v) => setForm({ ...form, defaultSite: v })}
   246→            >
   247→              <SelectTrigger className="dark:bg-slate-800 dark:border-slate-700">
   248→                <SelectValue placeholder="เลือกสาขา" />
   249→              </SelectTrigger>
   250→              <SelectContent>
   251→                {(sites ?? []).map((s) => (
   252→                  <SelectItem key={s.id} value={s.code}>
   253→                    {s.code} — {s.name}
   254→                  </SelectItem>
   255→                ))}
   256→              </SelectContent>
   257→            </Select>
   258→          </div>
   259→        </div>
   260→
   261→        <div className="flex items-center justify-between rounded-md border border-slate-200 bg-slate-50 px-4 py-3 dark:border-slate-800 dark:bg-slate-800/40">
   262→          <div>
   263→            <div className="text-sm font-medium text-slate-700 dark:text-slate-200">
   264→              ล็อกอินด้วยรหัสผ่าน
   265→            </div>
   266→            <div className="text-xs text-slate-500 dark:text-slate-400">
   267→              อนุญาตนอกเหนือจาก SSO (Google)
   268→            </div>
   269→          </div>
   270→          <Switch
   271→            checked={form.enablePasswordLogin === 'true'}
   272→            onCheckedChange={(c) =>
   273→              setForm({ ...form, enablePasswordLogin: c ? 'true' : 'false' })
   274→            }
   275→          />
   276→        </div>
   277→
   278→        <div className="flex justify-end">
   279→          <Button
   280→            onClick={save}
   281→            disabled={saving}
   282→            className="bg-[#f97316] text-white hover:bg-[#ea580c] focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   283→          >
   284→            <Save className="h-4 w-4" />
   285→            {saving ? 'กำลังบันทึก...' : 'บันทึก'}
   286→          </Button>
   287→        </div>
   288→      </CardContent>
   289→    </Card>
   290→  )
   291→}

   293→/* ---------- Master tab ---------- */
   294→function MasterTab() {
   295→  const qc = useQueryClient()
   296→  const [categoryFilter, setCategoryFilter] = React.useState<string>('all')
   297→  const [modalOpen, setModalOpen] = React.useState(false)
   298→  const [editTarget, setEditTarget] = React.useState<MasterItem | null>(null)
   299→  const [deleteTarget, setDeleteTarget] = React.useState<MasterItem | null>(null)
   300→  const [deleting, setDeleting] = React.useState(false)
   301→  const [syncing, setSyncing] = React.useState<string | null>(null)
   302→
   303→  const { data: items, isLoading } = useQuery<MasterItem[]>({
   304→    queryKey: ['master', categoryFilter],
   305→    queryFn: async () => {
   306→      const params = new URLSearchParams()
   307→      if (categoryFilter !== 'all') params.set('category', categoryFilter)
   308→      const res = await fetch(`/api/master?${params.toString()}`)
   309→      if (!res.ok) throw new Error('Failed to load master')
   310→      const json = await res.json()
   311→      return json.items as MasterItem[]
   312→    },
   313→  })
   314→
   315→  function openAdd() {
   316→    setEditTarget(null)
   317→    setModalOpen(true)
   318→  }
   319→  function openEdit(item: MasterItem) {
   320→    setEditTarget(item)
   321→    setModalOpen(true)
   322→  }
   323→
   324→  async function runSync(type: 'model' | 'dept' | 'labels') {
   325→    try {
   326→      setSyncing(type)
   327→      const res = await fetch(`/api/master/sync?type=${type}`, {
   328→        method: 'POST',
   329→      })
   330→      if (!res.ok) {
   331→        const j = await res.json().catch(() => ({}))
   332→        throw new Error(j.error ?? 'Sync failed')
   333→      }
   334→      const json = await res.json()
   335→      toast.success(
   336→        `ซิงค์ข้อมูลสำเร็จ (${type}) — อัปเดต ${json.updated ?? 0} รายการ`,
   337→      )
   338→      await qc.invalidateQueries({ queryKey: ['master'] })
   339→      await qc.invalidateQueries({ queryKey: ['devices'] })
   340→    } catch (e) {
   341→      toast.error(e instanceof Error ? e.message : 'Sync failed')
   342→    } finally {
   343→      setSyncing(null)
   344→    }
   345→  }
   346→
   347→  async function confirmDelete() {
   348→    if (!deleteTarget) return
   349→    try {
   350→      setDeleting(true)
   351→      const res = await fetch(`/api/master/${deleteTarget.id}`, {
   352→        method: 'DELETE',
   353→      })
   354→      if (!res.ok) {
   355→        const j = await res.json().catch(() => ({}))
   356→        throw new Error(j.error ?? 'Delete failed')
   357→      }
   358→      toast.success('ลบรายการแล้ว')
   359→      setDeleteTarget(null)
   360→      await qc.invalidateQueries({ queryKey: ['master'] })
   361→    } catch (e) {
   362→      toast.error(e instanceof Error ? e.message : 'Delete failed')
   363→    } finally {
   364→      setDeleting(false)
   365→    }
   366→  }
   367→
   368→  return (
   369→    <Card className="dark:border-slate-800 dark:bg-slate-900">
   370→      <CardHeader>
   371→        <CardTitle className="flex items-center gap-2 text-base text-slate-800 dark:text-slate-100">
   372→          <Database className="h-4 w-4 text-[#f97316]" />
   373→          ข้อมูลมาตรฐาน (Master Data)
   374→        </CardTitle>
   375→      </CardHeader>
   376→      <CardContent className="space-y-4">
   377→        {/* Toolbar */}
   378→        <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   379→          <div className="flex flex-1 flex-col gap-2 sm:flex-row sm:items-center">
   380→            <Select
   381→              value={categoryFilter}
   382→              onValueChange={setCategoryFilter}
   383→            >
   384→              <SelectTrigger className="w-full sm:w-44">
   385→                <SelectValue placeholder="หมวดหมู่" />
   386→              </SelectTrigger>
   387→              <SelectContent>
   388→                <SelectItem value="all">หมวดหมู่ทั้งหมด</SelectItem>
   389→                {MASTER_CATEGORIES.map((c) => (
   390→                  <SelectItem key={c} value={c}>
   391→                    {c}
   392→                  </SelectItem>
   393→                ))}
   394→              </SelectContent>
   395→            </Select>
   396→          </div>
   397→          <div className="flex flex-wrap items-center gap-2">
   398→            <Button
   399→              variant="outline"
   400→              size="sm"
   401→              onClick={() => runSync('model')}
   402→              disabled={syncing !== null}
