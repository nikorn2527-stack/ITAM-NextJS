   381→/* ---------- Master tab ---------- */
   382→function MasterTab() {
   383→  const qc = useQueryClient()
   384→  const [categoryFilter, setCategoryFilter] = React.useState<string>('all')
   385→  const [modalOpen, setModalOpen] = React.useState(false)
   386→  const [editTarget, setEditTarget] = React.useState<MasterItem | null>(null)
   387→  const [deleteTarget, setDeleteTarget] = React.useState<MasterItem | null>(null)
   388→  const [deleting, setDeleting] = React.useState(false)
   389→  const [syncing, setSyncing] = React.useState<string | null>(null)
   390→
   391→  const { data: items, isLoading } = useQuery<MasterItem[]>({
   392→    queryKey: ['master', categoryFilter],
   393→    queryFn: async () => {
   394→      const params = new URLSearchParams()
   395→      if (categoryFilter !== 'all') params.set('category', categoryFilter)
   396→      const res = await fetch(`/api/master?${params.toString()}`)
   397→      if (!res.ok) throw new Error('Failed to load master')
   398→      const json = await res.json()
   399→      return json.items as MasterItem[]
   400→    },
   401→  })
   402→
   403→  function openAdd() {
   404→    setEditTarget(null)
   405→    setModalOpen(true)
   406→  }
   407→  function openEdit(item: MasterItem) {
   408→    setEditTarget(item)
   409→    setModalOpen(true)
   410→  }
   411→
   412→  async function runSync(type: 'model' | 'dept' | 'labels') {
   413→    try {
   414→      setSyncing(type)
   415→      const res = await fetch(`/api/master/sync?type=${type}`, {
   416→        method: 'POST',
   417→      })
   418→      if (!res.ok) {
   419→        const j = await res.json().catch(() => ({}))
   420→        throw new Error(j.error ?? 'Sync failed')
   421→      }
   422→      const json = await res.json()
   423→      toast.success(
   424→        `ซิงค์ข้อมูลสำเร็จ (${type}) — อัปเดต ${json.updated ?? 0} รายการ`,
   425→      )
   426→      await qc.invalidateQueries({ queryKey: ['master'] })
   427→      await qc.invalidateQueries({ queryKey: ['devices'] })
   428→    } catch (e) {
   429→      toast.error(e instanceof Error ? e.message : 'Sync failed')
   430→    } finally {
   431→      setSyncing(null)
   432→    }
   433→  }
   434→
   435→  async function confirmDelete() {
   436→    if (!deleteTarget) return
   437→    try {
   438→      setDeleting(true)
   439→      const res = await fetch(`/api/master/${deleteTarget.id}`, {
   440→        method: 'DELETE',
   441→      })
   442→      if (!res.ok) {
   443→        const j = await res.json().catch(() => ({}))
   444→        throw new Error(j.error ?? 'Delete failed')
   445→      }
   446→      toast.success('ลบรายการแล้ว')
   447→      setDeleteTarget(null)
   448→      await qc.invalidateQueries({ queryKey: ['master'] })
   449→    } catch (e) {
   450→      toast.error(e instanceof Error ? e.message : 'Delete failed')
   451→    } finally {
   452→      setDeleting(false)
   453→    }
   454→  }
   455→
   456→  return (
   457→    <Card className="dark:border-slate-800 dark:bg-slate-900">
   458→      <CardHeader>
   459→        <CardTitle className="flex items-center gap-2 text-base text-slate-800 dark:text-slate-100">
   460→          <Database className="h-4 w-4 text-[#f97316]" />
   461→          ข้อมูลมาตรฐาน (Master Data)
   462→        </CardTitle>
   463→      </CardHeader>
   464→      <CardContent className="space-y-4">
   465→        {/* Toolbar */}
   466→        <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   467→          <div className="flex flex-1 flex-col gap-2 sm:flex-row sm:items-center">
   468→            <Select
   469→              value={categoryFilter}
   470→              onValueChange={setCategoryFilter}
   471→            >
   472→              <SelectTrigger className="w-full sm:w-44">
   473→                <SelectValue placeholder="หมวดหมู่" />
   474→              </SelectTrigger>
   475→              <SelectContent>
   476→                <SelectItem value="all">หมวดหมู่ทั้งหมด</SelectItem>
   477→                {MASTER_CATEGORIES.map((c) => (
   478→                  <SelectItem key={c} value={c}>
   479→                    {c}
   480→                  </SelectItem>
   481→                ))}
   482→              </SelectContent>
   483→            </Select>
   484→          </div>
   485→          <div className="flex flex-wrap items-center gap-2">
   486→            <Button
   487→              variant="outline"
   488→              size="sm"
   489→              onClick={() => runSync('model')}
   490→              disabled={syncing !== null}
   491→            >
   492→              <Link2 className="h-3.5 w-3.5" />
   493→              {syncing === 'model' ? '...' : 'Model'}
   494→            </Button>
   495→            <Button
   496→              variant="outline"
   497→              size="sm"
   498→              onClick={() => runSync('dept')}
   499→              disabled={syncing !== null}
   500→            >
   501→              <Link2 className="h-3.5 w-3.5" />
   502→              {syncing === 'dept' ? '...' : 'Dept'}
   503→            </Button>
   504→            <Button
   505→              variant="outline"
   506→              size="sm"
   507→              onClick={() => runSync('labels')}
   508→              disabled={syncing !== null}
   509→            >
   510→              <Link2 className="h-3.5 w-3.5" />
   511→              {syncing === 'labels' ? '...' : 'Labels'}
   512→            </Button>
   513→            <Button
   514→              onClick={openAdd}
