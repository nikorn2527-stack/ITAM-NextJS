    85→    85→/**
    86→    86→ * Animated count-up hook — tweens from the previous value to the next
    87→    87→ * over 500ms using requestAnimationFrame.
    88→    88→ */
    89→    89→function useCountUp(target: number, duration = 500) {
    90→    90→  const [display, setDisplay] = React.useState(target)
    91→    91→  const fromRef = React.useRef(target)
    92→    92→  const rafRef = React.useRef<number | null>(null)
    93→    93→  const startRef = React.useRef<number | null>(null)
    94→    94→
    95→    95→  React.useEffect(() => {
    96→    96→    if (typeof window === 'undefined') return
    97→    97→    const from = fromRef.current
    98→    98→    if (from === target) {
    99→    99→      setDisplay(target)
   100→   100→      return
   101→   101→    }
   102→   102→    startRef.current = null
   103→   103→    const step = (ts: number) => {
   104→   104→      if (startRef.current === null) startRef.current = ts
   105→   105→      const elapsed = ts - startRef.current
   106→   106→      const t = Math.min(1, elapsed / duration)
   107→   107→      // easeOutCubic
   108→   108→      const eased = 1 - Math.pow(1 - t, 3)
   109→   109→      const next = Math.round(from + (target - from) * eased)
   110→   110→      setDisplay(next)
   111→   111→      if (t < 1) {
   112→   112→        rafRef.current = window.requestAnimationFrame(step)
   113→   113→      } else {
   114→   114→        fromRef.current = target
   115→   115→        setDisplay(target)
   116→   116→      }
   117→   117→    }
   118→   118→    rafRef.current = window.requestAnimationFrame(step)
   119→   119→    return () => {
   120→   120→      if (rafRef.current !== null) {
   121→   121→        window.cancelAnimationFrame(rafRef.current)
   122→   122→      }
   123→   123→      // remember where we landed so the next tween starts smoothly
   124→   124→      fromRef.current = display
   125→   125→    }
   126→   126→  }, [target, duration])
   127→   127→
   128→   128→  return display
   129→   129→}
