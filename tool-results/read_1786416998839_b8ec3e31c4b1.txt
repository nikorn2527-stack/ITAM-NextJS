   111→export function DevicesPage() {
   112→  const qc = useQueryClient()
   113→  const [search, setSearch] = React.useState('')
   114→  const [statusFilter, setStatusFilter] = React.useState('all')
   115→  const [siteFilter, setSiteFilter] = React.useState('all')
   116→  const [dialogOpen, setDialogOpen] = React.useState(false)
   117→  const [form, setForm] = React.useState<FormState>(EMPTY_FORM)
   118→  const [saving, setSaving] = React.useState(false)
   119→  const [deleteTarget, setDeleteTarget] = React.useState<Device | null>(null)
   120→  const [deleting, setDeleting] = React.useState(false)
   121→  const [detailDeviceId, setDetailDeviceId] = React.useState<string | null>(
   122→    null,
   123→  )
   124→  const [exporting, setExporting] = React.useState(false)
   125→  const [importOpen, setImportOpen] = React.useState(false)
   126→  const [stickerOpen, setStickerOpen] = React.useState(false)
   127→
   128→  // Settings query for org name (used in sticker header)
   129→  const { data: settings } = useQuery<Record<string, string>>({
   130→    queryKey: ['settings'],
   131→    queryFn: async () => {
   132→      const res = await fetch('/api/settings')
   133→      if (!res.ok) return {}
   134→      const json = await res.json()
   135→      return (json.settings ?? {}) as Record<string, string>
   136→    },
   137→  })
   138→
   139→  const { data: devices, isLoading } = useQuery<Device[]>({
   140→    queryKey: ['devices', search, statusFilter, siteFilter],
   141→    queryFn: async () => {
   142→      const params = new URLSearchParams()
   143→      if (search) params.set('search', search)
   144→      if (statusFilter !== 'all') params.set('status', statusFilter)
   145→      if (siteFilter !== 'all') params.set('site', siteFilter)
   146→      const res = await fetch(`/api/devices?${params.toString()}`)
   147→      if (!res.ok) throw new Error('Failed to load devices')
   148→      const json = await res.json()
   149→      return json.devices as Device[]
   150→    },
   151→  })
   152→
   153→  const { data: sites } = useQuery<Site[]>({
   154→    queryKey: ['sites'],
   155→    queryFn: async () => {
   156→      const res = await fetch('/api/sites')
   157→      if (!res.ok) return []
   158→      const json = await res.json()
   159→      return json.sites as Site[]
   160→    },
   161→  })
   162→
   163→  const { data: masterItems } = useQuery<{
   164→    category: string
   165→    code: string
   166→    label: string
   167→  }[]>({
   168→    queryKey: ['master-all'],
   169→    queryFn: async () => {
   170→      const res = await fetch('/api/master')
   171→      if (!res.ok) return []
   172→      const json = await res.json()
   173→      return (json.items ?? []) as {
   174→        category: string
   175→        code: string
   176→        label: string
   177→      }[]
   178→    },
   179→  })
   180→
   181→  const brands = (masterItems ?? []).filter((m) => m.category === 'Brand')
   182→  const types = (masterItems ?? []).filter((m) => m.category === 'Type')
   183→  const departments = (masterItems ?? []).filter(
   184→    (m) => m.category === 'Department',
   185→  )
   186→
   187→  function openAdd() {
   188→    setForm({ ...EMPTY_FORM })
   189→    setDialogOpen(true)
   190→  }
   191→
   192→  function openEdit(d: Device) {
   193→    setForm({
   194→      id: d.id,
   195→      assetCode: d.assetCode,
   196→      name: d.name,
   197→      brand: d.brand,
   198→      model: d.model,
   199→      type: d.type,
   200→      serialNumber: d.serialNumber ?? '',
   201→      status: d.status,
   202→      site: d.site,
   203→      department: d.department ?? '',
   204→      departmentCode: d.departmentCode ?? '',
   205→      parentRef: d.parentRef ?? '',
   206→      displayLabel: d.displayLabel ?? '',
   207→      location: d.location ?? '',
   208→      purchaseDate: d.purchaseDate ?? '',
   209→    })
   210→    setDialogOpen(true)
   211→  }
   212→
   213→  async function save() {
   214→    if (!form.assetCode || !form.name || !form.brand || !form.model || !form.type) {
   215→      toast.error('กรุณากรอกข้อมูลที่จำเป็น (รหัส, ชื่อ, แบรนด์, รุ่น, ประเภท)')
   216→      return
   217→    }
   218→    try {
   219→      setSaving(true)
   220→      const payload = {
   221→        ...form,
   222→        serialNumber: form.serialNumber || null,
   223→        department: form.department || null,
   224→        departmentCode: form.departmentCode || null,
   225→        parentRef: form.parentRef || null,
   226→        displayLabel: form.displayLabel || null,
   227→        location: form.location || null,
   228→        purchaseDate: form.purchaseDate || null,
   229→      }
   230→      const isEdit = Boolean(form.id)
   231→      const url = isEdit ? `/api/devices/${form.id}` : '/api/devices'
   232→      const method = isEdit ? 'PUT' : 'POST'
   233→      const res = await fetch(url, {
   234→        method,
   235→        headers: { 'Content-Type': 'application/json' },
   236→        body: JSON.stringify(payload),
   237→      })
   238→      if (!res.ok) {
   239→        const j = await res.json().catch(() => ({}))
   240→        throw new Error(j.error ?? 'Save failed')
   241→      }
   242→      toast.success(isEdit ? 'แก้ไขอุปกรณ์แล้ว' : 'เพิ่มอุปกรณ์ใหม่แล้ว')
   243→      setDialogOpen(false)
   244→      await qc.invalidateQueries({ queryKey: ['devices'] })
   245→      await qc.invalidateQueries({ queryKey: ['dashboard'] })
   246→    } catch (e) {
   247→      toast.error(e instanceof Error ? e.message : 'Save failed')
   248→    } finally {
   249→      setSaving(false)
   250→    }
   251→  }
   252→
   253→  async function confirmDelete() {
   254→    if (!deleteTarget) return
   255→    try {
   256→      setDeleting(true)
   257→      const res = await fetch(`/api/devices/${deleteTarget.id}`, {
   258→        method: 'DELETE',
   259→      })
   260→      if (!res.ok) {
   261→        const j = await res.json().catch(() => ({}))
   262→        throw new Error(j.error ?? 'Delete failed')
   263→      }
   264→      toast.success('ลบอุปกรณ์แล้ว')
   265→      setDeleteTarget(null)
   266→      await qc.invalidateQueries({ queryKey: ['devices'] })
   267→      await qc.invalidateQueries({ queryKey: ['dashboard'] })
   268→    } catch (e) {
   269→      toast.error(e instanceof Error ? e.message : 'Delete failed')
   270→    } finally {
   271→      setDeleting(false)
   272→    }
   273→  }
   274→
   275→  async function exportCsv() {
   276→    try {
   277→      setExporting(true)
   278→      const params = new URLSearchParams()
   279→      if (search) params.set('search', search)
   280→      if (statusFilter !== 'all') params.set('status', statusFilter)
   281→      if (siteFilter !== 'all') params.set('site', siteFilter)
   282→      const res = await fetch(`/api/devices?${params.toString()}`)
   283→      if (!res.ok) throw new Error('Failed to export')
   284→      const json = await res.json()
   285→      const rows = (json.devices ?? []) as Device[]
   286→      downloadCsv(`devices-${dateStamp()}.csv`, rows, DEVICE_CSV_HEADERS)
   287→      toast.success(`ส่งออก ${rows.length} รายการแล้ว`)
   288→    } catch (e) {
   289→      toast.error(e instanceof Error ? e.message : 'Export failed')
   290→    } finally {
   291→      setExporting(false)
   292→    }
   293→  }
   294→
   295→  return (
   296→    <div className="space-y-4 p-4 md:p-6">
   297→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   298→        <div>
   299→          <h1 className="text-2xl font-bold text-slate-800 dark:text-slate-100">จัดการอุปกรณ์</h1>
   300→          <p className="text-sm text-slate-500 dark:text-slate-400">
   301→            เพิ่ม / แก้ไข / ลบ อุปกรณ์ IT ในระบบ
   302→          </p>
   303→        </div>
   304→      </div>
   305→
   306→      <Card className="dark:border-slate-800 dark:bg-slate-900">
   307→        <CardContent className="p-4">
   308→          {/* Toolbar */}
   309→          <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   310→            <div className="flex flex-1 flex-col gap-2 sm:flex-row sm:items-center">
   311→              <div className="relative flex-1">
   312→                <Search className="pointer-events-none absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
   313→                <Input
   314→                  placeholder="ค้นหารหัส / ชื่อ / SN / แบรนด์..."
   315→                  value={search}
   316→                  onChange={(e) => setSearch(e.target.value)}
   317→                  className="pl-9"
   318→                />
   319→              </div>
   320→              <Select value={statusFilter} onValueChange={setStatusFilter}>
   321→                <SelectTrigger className="w-full sm:w-40">
   322→                  <SelectValue placeholder="สถานะ" />
   323→                </SelectTrigger>
   324→                <SelectContent>
   325→                  <SelectItem value="all">สถานะทั้งหมด</SelectItem>
   326→                  {DEVICE_STATUS_OPTIONS.map((o) => (
   327→                    <SelectItem key={o.value} value={o.value}>
   328→                      {o.label}
   329→                    </SelectItem>
   330→                  ))}
   331→                </SelectContent>
   332→              </Select>
   333→              <Select value={siteFilter} onValueChange={setSiteFilter}>
   334→                <SelectTrigger className="w-full sm:w-40">
   335→                  <SelectValue placeholder="สาขา" />
   336→                </SelectTrigger>
   337→                <SelectContent>
   338→                  <SelectItem value="all">สาขาทั้งหมด</SelectItem>
   339→                  {(sites ?? []).map((s) => (
   340→                    <SelectItem key={s.code} value={s.code}>
   341→                      {s.name}
   342→                    </SelectItem>
   343→                  ))}
   344→                </SelectContent>
   345→              </Select>
   346→            </div>
   347→            <div className="flex items-center gap-2">
   348→              <Button
   349→                onClick={openAdd}
   350→                className="bg-[#f97316] text-white hover:bg-[#ea580c] focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   351→              >
   352→                <Plus className="h-4 w-4" />
   353→                เพิ่มอุปกรณ์
   354→              </Button>
   355→              <Button
   356→                variant="outline"
   357→                onClick={() => setImportOpen(true)}
   358→                className="focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   359→              >
   360→                <Upload className="h-4 w-4" />
   361→                นำเข้า CSV
   362→              </Button>
   363→              <Button
   364→                variant="outline"
   365→                onClick={exportCsv}
   366→                disabled={exporting}
   367→                className="focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   368→              >
   369→                <Download className="h-4 w-4" />
   370→                {exporting ? 'กำลังส่งออก...' : 'ส่งออก CSV'}
   371→              </Button>
   372→              <Button
   373→                variant="outline"
   374→                onClick={() => setStickerOpen(true)}
   375→                disabled={(devices ?? []).length === 0}
   376→                className="focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   377→              >
   378→                <Tag className="h-4 w-4" />
   379→                พิมพ์สติกเกอร์
   380→              </Button>
   381→              <Button
   382→                variant="outline"
   383→                onClick={() => qc.invalidateQueries({ queryKey: ['devices'] })}
   384→                className="focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   385→              >
   386→                <RefreshCw className="h-4 w-4" />
   387→                รีเฟรช
   388→              </Button>
   389→            </div>
   390→          </div>
   391→
   392→          {/* Table */}
   393→          <div className="itam-scroll mt-4 max-h-[60vh] overflow-auto rounded-md border border-slate-200 dark:border-slate-800">
   394→            <Table>
   395→              <TableHeader className="sticky top-0 z-10 bg-slate-50/80 backdrop-blur-sm dark:bg-slate-900/80">
   396→                <TableRow>
   397→                  <TableHead className="text-slate-600 dark:text-slate-300">รหัส</TableHead>
   398→                  <TableHead className="text-slate-600 dark:text-slate-300">ชื่อ</TableHead>
   399→                  <TableHead className="text-slate-600 dark:text-slate-300">แบรนด์</TableHead>
   400→                  <TableHead className="text-slate-600 dark:text-slate-300">รุ่น</TableHead>
   401→                  <TableHead className="text-slate-600 dark:text-slate-300">ประเภท</TableHead>
   402→                  <TableHead className="text-slate-600 dark:text-slate-300">สถานะ</TableHead>
   403→                  <TableHead className="text-slate-600 dark:text-slate-300">สาขา</TableHead>
   404→                  <TableHead className="text-slate-600 dark:text-slate-300">แผนก</TableHead>
   405→                  <TableHead className="text-slate-600 dark:text-slate-300">รหัสแผนก</TableHead>
   406→                  <TableHead className="text-right text-slate-600 dark:text-slate-300">การจัดการ</TableHead>
   407→                </TableRow>
   408→              </TableHeader>
   409→              <TableBody>
   410→                {isLoading ? (
   411→                  Array.from({ length: 6 }).map((_, i) => (
   412→                    <TableRow key={`sk-${i}`}>
   413→                      <TableCell colSpan={10}>
   414→                        <Skeleton className="h-6 w-full dark:bg-slate-800" />
   415→                      </TableCell>
   416→                    </TableRow>
   417→                  ))
   418→                ) : (devices ?? []).length === 0 ? (
   419→                  <TableRow>
   420→                    <TableCell colSpan={10} className="py-12">
   421→                      <div className="flex flex-col items-center justify-center gap-2 text-slate-400 dark:text-slate-500">
   422→                        <div className="flex h-14 w-14 items-center justify-center rounded-full bg-slate-100 dark:bg-slate-800">
   423→                          <PackageOpen className="h-7 w-7 text-slate-300 dark:text-slate-600" />
   424→                        </div>
   425→                        <div className="text-sm font-semibold text-slate-500 dark:text-slate-400">
   426→                          {search || statusFilter !== 'all' || siteFilter !== 'all'
   427→                            ? 'ไม่พบอุปกรณ์ที่ตรงกับเงื่อนไข'
   428→                            : 'ยังไม่มีอุปกรณ์ในระบบ'}
   429→                        </div>
   430→                        <div className="text-xs text-slate-400 dark:text-slate-500">
   431→                          {search || statusFilter !== 'all' || siteFilter !== 'all'
   432→                            ? 'ลองปรับตัวกรองหรือคำค้นหา หรือล้างตัวกรองเพื่อดูทั้งหมด'
   433→                            : 'เริ่มต้นโดยการเพิ่มอุปกรณ์เครื่องแรกของคุณ'}
   434→                        </div>
   435→                        {(search || statusFilter !== 'all' || siteFilter !== 'all') ? (
   436→                          <Button
   437→                            variant="outline"
   438→                            size="sm"
   439→                            onClick={() => {
   440→                              setSearch('')
   441→                              setStatusFilter('all')
   442→                              setSiteFilter('all')
   443→                            }}
   444→                            className="mt-2 focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   445→                          >
   446→                            ล้างตัวกรอง
   447→                          </Button>
   448→                        ) : (
   449→                          <Button
   450→                            size="sm"
   451→                            onClick={openAdd}
   452→                            className="mt-2 bg-[#f97316] text-white hover:bg-[#ea580c] focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   453→                          >
   454→                            <Plus className="h-4 w-4" />
   455→                            เพิ่มอุปกรณ์
   456→                          </Button>
   457→                        )}
   458→                      </div>
   459→                    </TableCell>
   460→                  </TableRow>
   461→                ) : (
   462→                  (devices ?? []).map((d) => (
   463→                    <TableRow
   464→                      key={d.id}
   465→                      className="cursor-pointer transition-colors hover:bg-slate-50/70 dark:hover:bg-slate-800/50"
   466→                      onClick={() => setDetailDeviceId(d.id)}
   467→                    >
   468→                      <TableCell className="font-mono text-xs font-medium text-slate-700 dark:text-slate-200">
   469→                        {d.assetCode}
   470→                      </TableCell>
   471→                      <TableCell className="max-w-[200px] truncate text-slate-700 dark:text-slate-200">
   472→                        {d.name}
   473→                      </TableCell>
   474→                      <TableCell className="text-slate-700 dark:text-slate-200">{d.brand}</TableCell>
   475→                      <TableCell className="max-w-[160px] truncate text-slate-700 dark:text-slate-200">
   476→                        {d.model}
   477→                      </TableCell>
   478→                      <TableCell className="text-slate-700 dark:text-slate-200">{d.type}</TableCell>
   479→                      <TableCell>
   480→                        <Badge className={statusBadgeClass(d.status)}>
   481→                          {statusLabel(d.status)}
   482→                        </Badge>
   483→                      </TableCell>
   484→                      <TableCell className="text-slate-700 dark:text-slate-200">{d.site}</TableCell>
   485→                      <TableCell className="max-w-[160px] truncate text-slate-600 dark:text-slate-300">
   486→                        {d.department ?? '-'}
   487→                      </TableCell>
   488→                      <TableCell className="font-mono text-xs text-slate-500 dark:text-slate-400">
   489→                        {d.departmentCode ?? '-'}
   490→                      </TableCell>
   491→                      <TableCell
   492→                        className="text-right"
   493→                        onClick={(e) => e.stopPropagation()}
   494→                      >
   495→                        <div className="flex justify-end gap-1">
   496→                          <Button
   497→                            size="icon"
   498→                            variant="ghost"

   769→// A combobox-like input: lets the user either type or pick from a dropdown list
   770→function SelectValueInput({
   771→  value,
   772→  onChange,
   773→  options,
   774→  placeholder,
   775→}: {
   776→  value: string
   777→  onChange: (v: string) => void
   778→  options: { value: string; label: string }[]
   779→  placeholder?: string
   780→}) {
   781→  const [open, setOpen] = React.useState(false)
   782→  const inputRef = React.useRef<HTMLInputElement>(null)
   783→  return (
   784→    <div className="relative">
   785→      <Input
   786→        ref={inputRef}
   787→        value={value}
   788→        onChange={(e) => onChange(e.target.value)}
   789→        onFocus={() => setOpen(true)}
   790→        onBlur={() => setTimeout(() => setOpen(false), 150)}
   791→        placeholder={placeholder}
   792→      />
   793→      {open && options.length > 0 && (
   794→        <div className="itam-scroll absolute z-50 mt-1 max-h-48 w-full overflow-y-auto rounded-md border bg-white shadow-lg">
   795→          {options.map((o) => (
   796→            <button
   797→              key={o.value}
   798→              type="button"
   799→              className="block w-full px-3 py-1.5 text-left text-sm hover:bg-slate-100"
   800→              onMouseDown={(e) => {
   801→                e.preventDefault()
   802→                onChange(o.value)
   803→                setOpen(false)
   804→              }}
   805→            >
   806→              {o.label}
   807→            </button>
   808→          ))}
   809→        </div>
   810→      )}
   811→    </div>
   812→  )
   813→}
