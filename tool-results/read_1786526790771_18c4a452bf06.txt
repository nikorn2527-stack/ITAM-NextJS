    41→// ============== Count-up hook ==============
    42→function useCountUp(value: number, durationMs = 600): number {
    43→  const [display, setDisplay] = React.useState(value)
    44→  const fromRef = React.useRef(value)
    45→  const rafRef = React.useRef<number | null>(null)
    46→  React.useEffect(() => {
    47→    const from = fromRef.current
    48→    const to = value
    49→    if (from === to) return
    50→    const start = performance.now()
    51→    const tick = (now: number) => {
    52→      const t = Math.min(1, (now - start) / durationMs)
    53→      const eased = 1 - Math.pow(1 - t, 3)
    54→      setDisplay(Math.round(from + (to - from) * eased))
    55→      if (t < 1) rafRef.current = requestAnimationFrame(tick)
    56→      else fromRef.current = to
    57→    }
    58→    rafRef.current = requestAnimationFrame(tick)
    59→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current) }
    60→  }, [value, durationMs])
    61→  return display
    62→}
