   404→// 10) Stock Transaction — รายการรับเข้า/เบิกออก
   405→// ============================================================
   406→model StockTransaction {
   407→  id              String   @id @default(cuid())
   408→  txnNumber       String?  // STX-YYYYMMDD-NNN — maps from ReceiptNo (IN) or IssueNo (OUT)
   409→  stockItemId     String
   410→  productCode     String?  // maps from ProductCode — denormalized for CSV import
   411→  productName     String?  // maps from ProductName — denormalized
   412→  type            String   // IN | OUT | ADJUST
   413→  quantity        Int      // maps from Quantity
   414→  unit            String?  // maps from Unit
   415→  balanceAfter    Int      @default(0)
   416→  reason          String?
   417→  requester       String?  // maps from Requester (StockOut)
   418→  department      String?  // maps from Department (StockOut)
   419→  purpose         String?  // maps from Purpose (StockOut)
   420→  approver        String?  // maps from Approver (StockOut)
   421→  approvedAt      String?  // maps from ApprovedAt (StockOut)
   422→  workOrderId     String?
   423→  workOrderNo     String?  // maps from WorkOrderNo — for CSV import
   424→  deviceId        String?
   425→  cost            Float?   // maps from TotalValue
   426→  unitCost        Float?   // maps from UnitPrice
   427→  vendor          String?  // maps from Supplier
   428→  receiver        String?  // maps from Receiver (StockIn)
   429→  purchaseOrderNo String?  // maps from PurchaseOrderNo (StockIn)
   430→  txnDate         String   // maps from Date
   431→  performedBy     String?  // maps from PerformedBy
   432→  remark          String?  // maps from Remark
   433→  sourceKey       String?  // maps from source_key
   434→  processedFlag   String?  // maps from processed_flag
   435→  // ── Approval workflow (PART 1: Stock Pending Approval System) ──
   436→  // null = immediate (legacy/IN-OUT-ADJUST processed at once)
   437→  // PENDING = waiting for approval (stock NOT yet reduced)
   438→  // APPROVED = approved (stock reduced on approval)
   439→  // REJECTED = rejected (no stock change)
