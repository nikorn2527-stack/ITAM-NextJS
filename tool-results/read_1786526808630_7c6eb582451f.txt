    93→/**
    94→ * Animated count-up hook — tweens from the previous value to the next
    95→ * over 500ms using requestAnimationFrame.
    96→ */
    97→function useCountUp(target: number, duration = 500) {
    98→  const [display, setDisplay] = React.useState(target)
    99→  const fromRef = React.useRef(target)
   100→  const rafRef = React.useRef<number | null>(null)
   101→  const startRef = React.useRef<number | null>(null)
   102→
   103→  React.useEffect(() => {
   104→    if (typeof window === 'undefined') return
   105→    const from = fromRef.current
   106→    if (from === target) {
   107→      setDisplay(target)
   108→      return
   109→    }
   110→    startRef.current = null
   111→    const step = (ts: number) => {
   112→      if (startRef.current === null) startRef.current = ts
   113→      const elapsed = ts - startRef.current
   114→      const t = Math.min(1, elapsed / duration)
   115→      // easeOutCubic
   116→      const eased = 1 - Math.pow(1 - t, 3)
   117→      const next = Math.round(from + (target - from) * eased)
   118→      setDisplay(next)
   119→      if (t < 1) {
   120→        rafRef.current = window.requestAnimationFrame(step)
   121→      } else {
   122→        fromRef.current = target
   123→        setDisplay(target)
   124→      }
   125→    }
   126→    rafRef.current = window.requestAnimationFrame(step)
   127→    return () => {
   128→      if (rafRef.current !== null) {
   129→        window.cancelAnimationFrame(rafRef.current)
   130→      }
   131→      // remember where we landed so the next tween starts smoothly
   132→      fromRef.current = display
   133→    }
   134→  }, [target, duration])
   135→
   136→  return display
   137→}
