   343→function formatDateTime(iso: string | null): string {
   344→  if (!iso) return '—'
   345→  try {
   346→    return new Date(iso).toLocaleString('th-TH', {
   347→      day: '2-digit',
   348→      month: '2-digit',
   349→      year: 'numeric',
   350→      hour: '2-digit',
   351→      minute: '2-digit',
   352→    })
   353→  } catch {
   354→    return iso
   355→  }
   356→}

   358→function parseExternalMeta(raw: string | null): ExternalMeta | null {
   359→  if (!raw) return null
   360→  try {
   361→    const obj = JSON.parse(raw)
   362→    if (!obj || typeof obj !== 'object') return null
   363→    return obj as ExternalMeta
   364→  } catch {
   365→    return null
   366→  }

   420→const MAX_PIC_BYTES = 1_500_000
   421→
   422→function readFileAsDataUrl(file: File): Promise<string> {
   423→  return new Promise((resolve, reject) => {
   424→    const reader = new FileReader()
   425→    reader.onload = () => resolve(String(reader.result))
   426→    reader.onerror = reject
   427→    reader.readAsDataURL(file)
   428→  })
   429→}
   430→
   431→/**
   432→ * Compress an image file to a JPEG data URL.
   433→ *
   434→ * - Max dimension: 1024px (preserves aspect ratio)
   435→ * - JPEG quality starts at 0.7 and steps down to ~0.4 if the output is still
   436→ *   larger than `maxBytes`
   437→ *
   438→ * Falls back to the original data URL if canvas drawing fails.
   439→ */
   440→async function compressImage(file: File, maxBytes = MAX_PIC_BYTES): Promise<string> {
   441→  const dataUrl = await readFileAsDataUrl(file)
   442→  // For non-image inputs (shouldn't happen given accept="image/*"), skip canvas
   443→  if (!dataUrl.startsWith('data:image')) return dataUrl
   444→  return new Promise<string>((resolve) => {
   445→    const img = new window.Image()
   446→    img.onload = () => {
   447→      const maxDim = 1024
   448→      let { width, height } = img
   449→      if (width > maxDim || height > maxDim) {
   450→        const ratio = Math.min(maxDim / width, maxDim / height)
   451→        width = Math.round(width * ratio)
   452→        height = Math.round(height * ratio)
   453→      }
   454→      const canvas = document.createElement('canvas')
   455→      canvas.width = width
   456→      canvas.height = height
   457→      const ctx = canvas.getContext('2d')
   458→      if (!ctx) {
   459→        resolve(dataUrl)
   460→        return
   461→      }
   462→      // White background to avoid black-on-transparent PNGs turning into black
   463→      // when re-encoded as JPEG.
   464→      ctx.fillStyle = '#ffffff'
   465→      ctx.fillRect(0, 0, width, height)
   466→      ctx.drawImage(img, 0, 0, width, height)
   467→      let quality = 0.7
   468→      let out = canvas.toDataURL('image/jpeg', quality)
   469→      while (out.length > maxBytes && quality > 0.4) {
   470→        quality -= 0.1
   471→        out = canvas.toDataURL('image/jpeg', quality)
   472→      }
   473→      resolve(out)
   474→    }
   475→    img.onerror = () => resolve(dataUrl)
   476→    img.src = dataUrl
   477→  })
   478→}

  1034→  const fileInputRef = React.useRef<HTMLInputElement>(null)
  1035→  const [picBusy, setPicBusy] = React.useState(false)
  1036→
  1037→  // Group subjects for the dropdown

  1047→  // Device lookup (internal mode only)

  1105→  // Multi-image upload (up to 9). Reads every selected file, compresses each,
  1106→  // then appends to form.picBeforeImages while respecting the cap.
  1107→  async function handlePicBeforeChange(e: React.ChangeEvent<HTMLInputElement>) {
  1108→    const files = e.target.files
  1109→    if (!files || files.length === 0) return
  1110→    try {
  1111→      setPicBusy(true)
  1112→      const remaining = MAX_IMAGES_PER_STAGE - form.picBeforeImages.length
  1113→      if (remaining <= 0) {
  1114→        toast.error(`เพิ่มรูปได้สูงสุด ${MAX_IMAGES_PER_STAGE} รูป`)
  1115→        return
  1116→      }
  1117→      const list = Array.from(files).slice(0, remaining)
  1118→      const compressed: string[] = []
  1119→      for (const f of list) {
  1120→        try {
  1121→          const dataUrl = await compressImage(f)
  1122→          compressed.push(dataUrl)
  1123→        } catch {
  1124→          /* skip bad file */
  1125→        }
  1126→      }
  1127→      if (compressed.length === 0) {
  1128→        toast.error('อ่านไฟล์รูปไม่สำเร็จ')
  1129→        return
  1130→      }
  1131→      setForm((s) => ({
  1132→        ...s,
  1133→        picBeforeImages: [...s.picBeforeImages, ...compressed].slice(
  1134→          0,
  1135→          MAX_IMAGES_PER_STAGE,
  1136→        ),
  1137→      }))
  1138→      if (list.length < files.length) {
  1139→        toast.message(`เพิ่มได้สูงสุด ${MAX_IMAGES_PER_STAGE} รูป — เพิ่ม ${list.length} รูปแรก`)
  1140→      }
  1141→    } finally {
  1142→      setPicBusy(false)
  1143→      if (fileInputRef.current) fileInputRef.current.value = ''
  1144→    }
  1145→  }
