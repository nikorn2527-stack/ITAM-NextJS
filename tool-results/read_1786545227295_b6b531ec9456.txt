    92→    92→/**
    93→    93→ * Animated count-up hook — tweens from the previous value to the next
    94→    94→ * over 500ms using requestAnimationFrame.
    95→    95→ */
    96→    96→function useCountUp(target: number, duration = 500) {
    97→    97→  const [display, setDisplay] = React.useState(target)
    98→    98→  const fromRef = React.useRef(target)
    99→    99→  const rafRef = React.useRef<number | null>(null)
   100→   100→  const startRef = React.useRef<number | null>(null)
   101→   101→
   102→   102→  React.useEffect(() => {
   103→   103→    if (typeof window === 'undefined') return
   104→   104→    const from = fromRef.current
   105→   105→    if (from === target) {
   106→   106→      setDisplay(target)
   107→   107→      return
   108→   108→    }
   109→   109→    startRef.current = null
   110→   110→    const step = (ts: number) => {
   111→   111→      if (startRef.current === null) startRef.current = ts
   112→   112→      const elapsed = ts - startRef.current
   113→   113→      const t = Math.min(1, elapsed / duration)
   114→   114→      // easeOutCubic
   115→   115→      const eased = 1 - Math.pow(1 - t, 3)
   116→   116→      const next = Math.round(from + (target - from) * eased)
   117→   117→      setDisplay(next)
   118→   118→      if (t < 1) {
   119→   119→        rafRef.current = window.requestAnimationFrame(step)
   120→   120→      } else {
   121→   121→        fromRef.current = target
   122→   122→        setDisplay(target)
   123→   123→      }
   124→   124→    }
   125→   125→    rafRef.current = window.requestAnimationFrame(step)
   126→   126→    return () => {
   127→   127→      if (rafRef.current !== null) {
   128→   128→        window.cancelAnimationFrame(rafRef.current)
   129→   129→      }
   130→   130→      // remember where we landed so the next tween starts smoothly
   131→   131→      fromRef.current = display
   132→   132→    }
   133→   133→  }, [target, duration])
   134→   134→
   135→   135→  return display
   136→   136→}
