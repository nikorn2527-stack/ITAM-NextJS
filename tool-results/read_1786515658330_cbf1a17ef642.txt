    92→/**
    93→ * Animated count-up hook — tweens from the previous value to the next
    94→ * over 500ms using requestAnimationFrame.
    95→ */
    96→function useCountUp(target: number, duration = 500) {
    97→  const [display, setDisplay] = React.useState(target)
    98→  const fromRef = React.useRef(target)
    99→  const rafRef = React.useRef<number | null>(null)
   100→  const startRef = React.useRef<number | null>(null)
   101→
   102→  React.useEffect(() => {
   103→    if (typeof window === 'undefined') return
   104→    const from = fromRef.current
   105→    if (from === target) {
   106→      setDisplay(target)
   107→      return
   108→    }
   109→    startRef.current = null
   110→    const step = (ts: number) => {
   111→      if (startRef.current === null) startRef.current = ts
   112→      const elapsed = ts - startRef.current
   113→      const t = Math.min(1, elapsed / duration)
   114→      // easeOutCubic
   115→      const eased = 1 - Math.pow(1 - t, 3)
   116→      const next = Math.round(from + (target - from) * eased)
   117→      setDisplay(next)
   118→      if (t < 1) {
   119→        rafRef.current = window.requestAnimationFrame(step)
   120→      } else {
   121→        fromRef.current = target
   122→        setDisplay(target)
   123→      }
   124→    }
   125→    rafRef.current = window.requestAnimationFrame(step)
   126→    return () => {
   127→      if (rafRef.current !== null) {
   128→        window.cancelAnimationFrame(rafRef.current)
   129→      }
   130→      // remember where we landed so the next tween starts smoothly
   131→      fromRef.current = display
   132→    }
   133→  }, [target, duration])
   134→
   135→  return display
   136→}
