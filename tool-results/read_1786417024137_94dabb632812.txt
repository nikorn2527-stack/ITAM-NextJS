   132→/* ---------- App tab ---------- */
   133→function AppTab() {
   134→  const qc = useQueryClient()
   135→  const { data: settings, isLoading } = useQuery<Record<string, string>>({
   136→    queryKey: ['settings'],
   137→    queryFn: async () => {
   138→      const res = await fetch('/api/settings')
   139→      if (!res.ok) throw new Error('Failed to load settings')
   140→      const json = await res.json()
   141→      return json.settings as Record<string, string>
   142→    },
   143→  })
   144→
   145→  const [form, setForm] = React.useState<Record<string, string>>({})
   146→  const [saving, setSaving] = React.useState(false)
   147→
   148→  React.useEffect(() => {
   149→    if (settings) setForm({ ...settings })
   150→  }, [settings])
   151→
   152→  async function save() {
   153→    try {
   154→      setSaving(true)
   155→      const res = await fetch('/api/settings', {
   156→        method: 'PUT',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify(form),
   159→      })
   160→      if (!res.ok) throw new Error('Save failed')
   161→      toast.success('บันทึกการตั้งค่าแล้ว')
   162→      await qc.invalidateQueries({ queryKey: ['settings'] })
   163→    } catch (e) {
   164→      toast.error(e instanceof Error ? e.message : 'Save failed')
   165→    } finally {
   166→      setSaving(false)
   167→    }
   168→  }
   169→
   170→  if (isLoading) {
   171→    return (
   172→      <Card className="dark:border-slate-800 dark:bg-slate-900">
   173→        <CardContent className="p-6">
   174→          <Skeleton className="h-64 w-full dark:bg-slate-800" />
   175→        </CardContent>
   176→      </Card>
   177→    )
   178→  }
   179→
   180→  return (
   181→    <Card className="dark:border-slate-800 dark:bg-slate-900">
   182→      <CardHeader>
   183→        <CardTitle className="flex items-center gap-2 text-base text-slate-800 dark:text-slate-100">
   184→          <SettingsIcon className="h-4 w-4 text-[#f97316]" />
   185→          ตั้งค่าทั่วไป
   186→        </CardTitle>
   187→      </CardHeader>
   188→      <CardContent className="space-y-4">
   189→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   190→          <div className="space-y-1.5">
   191→            <Label className="text-xs font-medium text-slate-600">
   192→              ชื่อองค์กร
   193→            </Label>
   194→            <Input
   195→              value={form.orgName ?? ''}
   196→              onChange={(e) => setForm({ ...form, orgName: e.target.value })}
   197→            />
   198→          </div>
   199→          <div className="space-y-1.5">
   200→            <Label className="text-xs font-medium text-slate-600">
   201→              สาขาเริ่มต้น
   202→            </Label>
   203→            <Input
   204→              value={form.defaultSite ?? ''}
   205→              onChange={(e) =>
   206→                setForm({ ...form, defaultSite: e.target.value })
   207→              }
   208→            />
   209→          </div>

   211→
   212→        <div className="flex items-center justify-between rounded-md border border-slate-200 bg-slate-50 px-4 py-3 dark:border-slate-800 dark:bg-slate-800/40">

   271→/* ---------- Master tab ---------- */
   272→function MasterTab() {
   273→  const qc = useQueryClient()
   274→  const [categoryFilter, setCategoryFilter] = React.useState<string>('all')
   275→  const [modalOpen, setModalOpen] = React.useState(false)
   276→  const [editTarget, setEditTarget] = React.useState<MasterItem | null>(null)
   277→  const [deleteTarget, setDeleteTarget] = React.useState<MasterItem | null>(null)
   278→  const [deleting, setDeleting] = React.useState(false)
   279→  const [syncing, setSyncing] = React.useState<string | null>(null)
   280→
   281→  const { data: items, isLoading } = useQuery<MasterItem[]>({
   282→    queryKey: ['master', categoryFilter],
   283→    queryFn: async () => {
   284→      const params = new URLSearchParams()
   285→      if (categoryFilter !== 'all') params.set('category', categoryFilter)
   286→      const res = await fetch(`/api/master?${params.toString()}`)
   287→      if (!res.ok) throw new Error('Failed to load master')
   288→      const json = await res.json()
   289→      return json.items as MasterItem[]
   290→    },
   291→  })
   292→
   293→  function openAdd() {
   294→    setEditTarget(null)
   295→    setModalOpen(true)
   296→  }
   297→  function openEdit(item: MasterItem) {
   298→    setEditTarget(item)
   299→    setModalOpen(true)
   300→  }
   301→
   302→  async function runSync(type: 'model' | 'dept' | 'labels') {
   303→    try {
   304→      setSyncing(type)
   305→      const res = await fetch(`/api/master/sync?type=${type}`, {
   306→        method: 'POST',
   307→      })
   308→      if (!res.ok) {
   309→        const j = await res.json().catch(() => ({}))
   310→        throw new Error(j.error ?? 'Sync failed')
   311→      }
   312→      const json = await res.json()
   313→      toast.success(
   314→        `ซิงค์ข้อมูลสำเร็จ (${type}) — อัปเดต ${json.updated ?? 0} รายการ`,
   315→      )
   316→      await qc.invalidateQueries({ queryKey: ['master'] })
   317→      await qc.invalidateQueries({ queryKey: ['devices'] })
   318→    } catch (e) {
   319→      toast.error(e instanceof Error ? e.message : 'Sync failed')
   320→    } finally {
   321→      setSyncing(null)
   322→    }
   323→  }
   324→
   325→  async function confirmDelete() {
   326→    if (!deleteTarget) return
   327→    try {
   328→      setDeleting(true)
   329→      const res = await fetch(`/api/master/${deleteTarget.id}`, {
   330→        method: 'DELETE',
   331→      })
   332→      if (!res.ok) {
   333→        const j = await res.json().catch(() => ({}))
   334→        throw new Error(j.error ?? 'Delete failed')
   335→      }
   336→      toast.success('ลบรายการแล้ว')
   337→      setDeleteTarget(null)
   338→      await qc.invalidateQueries({ queryKey: ['master'] })
   339→    } catch (e) {
   340→      toast.error(e instanceof Error ? e.message : 'Delete failed')
   341→    } finally {
   342→      setDeleting(false)
   343→    }
   344→  }
   345→
   346→  return (
   347→    <Card className="dark:border-slate-800 dark:bg-slate-900">
   348→      <CardHeader>
   349→        <CardTitle className="flex items-center gap-2 text-base text-slate-800 dark:text-slate-100">
   350→          <Database className="h-4 w-4 text-[#f97316]" />
   351→          ข้อมูลมาตรฐาน (Master Data)
   352→        </CardTitle>
   353→      </CardHeader>
   354→      <CardContent className="space-y-4">
   355→        {/* Toolbar */}
   356→        <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   357→          <div className="flex flex-1 flex-col gap-2 sm:flex-row sm:items-center">
   358→            <Select
   359→              value={categoryFilter}
   360→              onValueChange={setCategoryFilter}
   361→            >
   362→              <SelectTrigger className="w-full sm:w-44">
   363→                <SelectValue placeholder="หมวดหมู่" />
   364→              </SelectTrigger>
   365→              <SelectContent>
   366→                <SelectItem value="all">หมวดหมู่ทั้งหมด</SelectItem>
   367→                {MASTER_CATEGORIES.map((c) => (
   368→                  <SelectItem key={c} value={c}>
   369→                    {c}
   370→                  </SelectItem>
   371→                ))}
   372→              </SelectContent>
   373→            </Select>
   374→          </div>
   375→          <div className="flex flex-wrap items-center gap-2">
   376→            <Button
   377→              variant="outline"
   378→              size="sm"
   379→              onClick={() => runSync('model')}
   380→              disabled={syncing !== null}
   381→            >
   382→              <Link2 className="h-3.5 w-3.5" />
   383→              {syncing === 'model' ? '...' : 'Model'}

   386→              variant="outline"
   387→              size="sm"

   390→            >
   391→              <Link2 className="h-3.5 w-3.5" />
   392→              {syncing === 'dept' ? '...' : 'Dept'}
   393→            </Button>
   394→            <Button
   395→              variant="outline"
