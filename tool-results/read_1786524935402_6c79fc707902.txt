   168→function formatBaht(n: number | null | undefined): string {
   169→  if (n == null || isNaN(n)) return '—'
   170→  return '฿' + n.toLocaleString('th-TH', { minimumFractionDigits: 2, maximumFractionDigits: 2 })
   171→}
   172→
   173→function formatDate(iso: string | null | undefined): string {
   174→  if (!iso) return '—'
   175→  try {
   176→    // Accept both ISO datetime and yyyy-mm-dd
   177→    const d = new Date(iso.length <= 10 ? `${iso}T00:00:00` : iso)
   178→    if (isNaN(d.getTime())) return iso
   179→    return d.toLocaleDateString('th-TH', { day: '2-digit', month: 'short', year: 'numeric' })
   180→  } catch {
   181→    return iso
   182→  }
   183→}
   184→
   185→function todayISO(): string {
   186→  // Local date in yyyy-mm-dd (avoid UTC drift)
   187→  const d = new Date()
   188→  const y = d.getFullYear()
   189→  const m = String(d.getMonth() + 1).padStart(2, '0')
   190→  const day = String(d.getDate()).padStart(2, '0')
   191→  return `${y}-${m}-${day}`
   192→}
   193→
   194→function isThisMonth(iso: string | null | undefined): boolean {
   195→  if (!iso) return false
   196→  try {
   197→    const d = new Date(iso.length <= 10 ? `${iso}T00:00:00` : iso)
   198→    if (isNaN(d.getTime())) return false
   199→    const now = new Date()
   200→    return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth()
   201→  } catch {
   202→    return false
   203→  }
   204→}
   205→
   206→function truncate(s: string | null | undefined, n = 40): string {
   207→  if (!s) return '—'
   208→  return s.length > n ? s.slice(0, n) + '…' : s
   209→}
