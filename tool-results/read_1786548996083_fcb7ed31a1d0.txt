    73→function formatThaiDate(iso: string | null): string {
    74→  if (!iso) return '—'
    75→  try {
    76→    return new Date(iso).toLocaleDateString('th-TH', {
    77→      day: '2-digit',
    78→      month: '2-digit',
    79→      year: 'numeric',
    80→    })
    81→  } catch {
    82→    return iso
    83→  }
    84→}

    86→/** Pretty-print a JSON string for display; returns the original on parse error. */
    87→function prettyJson(raw: string): string {
    88→  try {
    89→    return JSON.stringify(JSON.parse(raw), null, 2)
    90→  } catch {
    91→    return raw
    92→  }
    93→}

    95→// ---------- Default-template seeding ----------
    96→// On first load, if a type has zero templates, auto-create the default one.

    98→function useSeedDefaults() {
    99→  const queryClient = useQueryClient()
   100→  const [seeding, setSeeding] = React.useState(false)
   101→  const seededRef = React.useRef(false)
   102→
   103→  const { data: allTemplates, isLoading } = useQuery<DocumentTemplate[]>({
   104→    queryKey: ['templates', 'all'],
   105→    queryFn: async () => {
   106→      const res = await fetch('/api/templates')
   107→      if (!res.ok) throw new Error('Failed to load templates')
   108→      const j = await res.json()
   109→      return (j.templates ?? []) as DocumentTemplate[]
   110→    },
   111→    staleTime: 30_000,
   112→  })
   113→
   114→  React.useEffect(() => {
   115→    if (isLoading || seededRef.current || !allTemplates) return
   116→    const missing = TEMPLATE_TYPES.filter(
   117→      (t) => !allTemplates.some((tpl) => tpl.type === t),
   118→    )
   119→    if (missing.length === 0) {
   120→      seededRef.current = true
   121→      return
   122→    }
   123→    seededRef.current = true
   124→    setSeeding(true)
   125→    Promise.all(
   126→      missing.map(async (t) => {
   127→        const def = DEFAULT_TEMPLATES[t]
   128→        const res = await fetch('/api/templates', {
   129→          method: 'POST',
   130→          headers: { 'Content-Type': 'application/json' },
   131→          body: JSON.stringify({
   132→            name: def.name,
   133→            type: t,
   134→            content: def.content,
   135→            isActive: true,
   136→            isDefault: true,
   137→          }),
   138→        })
   139→        if (!res.ok) {
   140→          const j = await res.json().catch(() => ({}))
   141→          throw new Error(j.error ?? `สร้างเทมเพลตเริ่มต้นสำหรับ ${t} ล้มเหลว`)
   142→        }
   143→        return res.json()
   144→      }),
   145→    )
   146→      .then(() => {
   147→        queryClient.invalidateQueries({ queryKey: ['templates'] })
   148→        toast.success('สร้างเทมเพลตเริ่มต้นเรียบร้อยแล้ว')
   149→      })
   150→      .catch((err: unknown) => {
   151→        toast.error(
   152→          err instanceof Error ? err.message : 'สร้างเทมเพลตเริ่มต้นล้มเหลว',
   153→        )
   154→      })
   155→      .finally(() => setSeeding(false))
   156→  }, [allTemplates, isLoading, queryClient])
   157→
   158→  return { seeding }
   159→}

   188→function EditorDialog({ open, onOpenChange, type, initial }: EditorDialogProps) {
   189→  const queryClient = useQueryClient()
   190→  const [form, setForm] = React.useState<EditorState>(initial)
   191→  const [jsonError, setJsonError] = React.useState<string | null>(null)
   192→
   193→  // Sync form state whenever the dialog opens or the initial value changes.
   194→  React.useEffect(() => {
   195→    if (open) {
   196→      setForm(initial)
   197→      setJsonError(null)
   198→    }
   199→  }, [open, initial])
   200→
   201→  const isEditing = form.id !== null
   202→
   203→  const saveMutation = useMutation({
   204→    mutationFn: async () => {
   205→      // Validate JSON content (must be parseable).
   206→      let contentStr = form.content
   207→      try {
   208→        // Re-stringify to normalise — store a compact-but-valid JSON.
   209→        const parsed = JSON.parse(form.content)
   210→        contentStr = JSON.stringify(parsed)
   211→      } catch {
   212→        throw new Error('เนื้อหา (content) ต้องเป็น JSON ที่ถูกต้อง')
   213→      }
   214→
   215→      const payload = {
   216→        name: form.name,
   217→        type, // type is fixed by the selected card
   218→        category: form.category.trim() || null,
   219→        content: contentStr,
   220→        isActive: form.isActive,
   221→        isDefault: form.isDefault,
   222→      }
   223→
   224→      if (isEditing && form.id) {
   225→        const res = await fetch(`/api/templates/${form.id}`, {
   226→          method: 'PUT',
   227→          headers: { 'Content-Type': 'application/json' },
   228→          body: JSON.stringify(payload),
   229→        })
   230→        if (!res.ok) {
   231→          const j = await res.json().catch(() => ({}))
   232→          throw new Error(j.error ?? 'บันทึกไม่สำเร็จ')
   233→        }
   234→        return res.json()
   235→      } else {
   236→        const res = await fetch('/api/templates', {
   237→          method: 'POST',
   238→          headers: { 'Content-Type': 'application/json' },
   239→          body: JSON.stringify(payload),
   240→        })
   241→        if (!res.ok) {
   242→          const j = await res.json().catch(() => ({}))
   243→          throw new Error(j.error ?? 'สร้างไม่สำเร็จ')
   244→        }
   245→        return res.json()
   246→      }
   247→    },
   248→    onSuccess: () => {
   249→      queryClient.invalidateQueries({ queryKey: ['templates'] })
   250→      toast.success(isEditing ? 'บันทึกเทมเพลตเรียบร้อย' : 'สร้างเทมเพลตเรียบร้อย')
   251→      onOpenChange(false)
   252→    },
   253→    onError: (err: unknown) => {
   254→      toast.error(err instanceof Error ? err.message : 'บันทึกไม่สำเร็จ')
   255→    },
   256→  })
   257→
   258→  // Live JSON validation as the user types.
   259→  React.useEffect(() => {
   260→    if (!open) return
   261→    try {
   262→      JSON.parse(form.content)
   263→      setJsonError(null)
   264→    } catch (e) {
   265→      setJsonError(e instanceof Error ? e.message : 'JSON ไม่ถูกต้อง')
   266→    }
   267→  }, [form.content, open])
   268→
   269→  function handleSubmit(e: React.FormEvent) {
   270→    e.preventDefault()
   271→    if (!form.name.trim()) {
   272→      toast.error('กรุณาระบุชื่อเทมเพลต')
   273→      return
   274→    }
   275→    saveMutation.mutate()
   276→  }
   277→
   278→  return (
   279→    <Dialog open={open} onOpenChange={onOpenChange}>
   280→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-2xl">
   281→        <DialogHeader>
   282→          <DialogTitle>
   283→            {isEditing ? 'แก้ไขเทมเพลต' : 'สร้างเทมเพลตใหม่'}
   284→          </DialogTitle>
   285→          <DialogDescription>
   286→            ประเภท:{' '}
   287→            <span className="font-medium text-foreground">
   288→              {TEMPLATE_TYPE_META.find((m) => m.value === type)?.label ?? type}
   289→            </span>
   290→          </DialogDescription>
   291→        </DialogHeader>
   292→
   293→        <form onSubmit={handleSubmit} className="space-y-4">
   294→          {/* name */}
   295→          <div className="space-y-2">
   296→            <Label htmlFor="tpl-name">
   297→              ชื่อเทมเพลต <span className="text-destructive">*</span>
   298→            </Label>
   299→            <Input
   300→              id="tpl-name"
   301→              value={form.name}
   302→              onChange={(e) => setForm((s) => ({ ...s, name: e.target.value }))}
   303→              placeholder="เช่น สติกเกอร์อุปกรณ์ IT"
   304→              autoFocus
   305→            />
   306→          </div>
   307→
   308→          {/* category */}
   309→          <div className="space-y-2">
   310→            <Label htmlFor="tpl-category">หมวดหมู่ (ไม่บังคับ)</Label>
   311→            <Input
   312→              id="tpl-category"
   313→              value={form.category}
   314→              onChange={(e) =>
   315→                setForm((s) => ({ ...s, category: e.target.value }))
   316→              }
   317→              placeholder="เช่น asset, meter, location"
   318→            />
   319→          </div>
   320→
   321→          {/* content (JSON) */}
   322→          <div className="space-y-2">
   323→            <div className="flex items-center justify-between">
   324→              <Label htmlFor="tpl-content">
   325→                เนื้อหา (JSON) <span className="text-destructive">*</span>
   326→              </Label>
   327→              <span
   328→                className={
   329→                  jsonError
   330→                    ? 'text-xs font-medium text-rose-600 dark:text-rose-400'
   331→                    : 'text-xs font-medium text-emerald-600 dark:text-emerald-400'
   332→                }
   333→              >
   334→                {jsonError ? '⚠ JSON ไม่ถูกต้อง' : '✓ JSON ถูกต้อง'}
   335→              </span>
   336→            </div>
   337→            <Textarea
   338→              id="tpl-content"
   339→              value={form.content}
   340→              onChange={(e) =>
   341→                setForm((s) => ({ ...s, content: e.target.value }))
   342→              }
   343→              className="min-h-[240px] font-mono text-xs"
   344→              spellCheck={false}
   345→            />
   346→            {jsonError && (
   347→              <p className="text-xs text-rose-600 dark:text-rose-400">
   348→                {jsonError}
   349→              </p>
   350→            )}
   351→          </div>
   352→
   353→          {/* toggles */}
   354→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   355→            <div className="flex items-center justify-between rounded-lg border p-3">
   356→              <div className="space-y-0.5">
   357→                <Label htmlFor="tpl-active" className="cursor-pointer">
   358→                  ใช้งาน
   359→                </Label>
   360→                <p className="text-xs text-muted-foreground">
   361→                  ปิดชั่วคราวเพื่อซ่อนจากรายการที่ใช้ได้
   362→                </p>
   363→              </div>
   364→              <Switch
   365→                id="tpl-active"
   366→                checked={form.isActive}
   367→                onCheckedChange={(v) =>
   368→                  setForm((s) => ({ ...s, isActive: v }))
   369→                }
   370→              />
   371→            </div>
   372→            <div className="flex items-center justify-between rounded-lg border p-3">
   373→              <div className="space-y-0.5">
   374→                <Label htmlFor="tpl-default" className="cursor-pointer">
   375→                  ตั้งเป็นค่าเริ่มต้น
   376→                </Label>
   377→                <p className="text-xs text-muted-foreground">
   378→                  ใช้เป็นเทมเพลตหลักของประเภทนี้
   379→                </p>
   380→              </div>
   381→              <Switch
   382→                id="tpl-default"
   383→                checked={form.isDefault}
   384→                onCheckedChange={(v) =>
   385→                  setForm((s) => ({ ...s, isDefault: v }))
   386→                }
   387→              />
   388→            </div>
   389→          </div>
   390→
   391→          <DialogFooter>
   392→            <Button
   393→              type="button"
   394→              variant="outline"
   395→              onClick={() => onOpenChange(false)}
   396→              disabled={saveMutation.isPending}
   397→            >
   398→              ยกเลิก
   399→            </Button>
   400→            <Button type="submit" disabled={saveMutation.isPending}>
   401→              {saveMutation.isPending ? (
   402→                <>
   403→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   404→                  กำลังบันทึก…
   405→                </>
   406→              ) : (
   407→                'บันทึก'
   408→              )}
   409→            </Button>
   410→          </DialogFooter>
   411→        </form>
   412→      </DialogContent>
   413→    </Dialog>
   414→  )
   415→}
