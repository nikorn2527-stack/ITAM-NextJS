    84→/**
    85→ * Animated count-up hook — tweens from the previous value to the next
    86→ * over 500ms using requestAnimationFrame.
    87→ */
    88→function useCountUp(target: number, duration = 500) {
    89→  const [display, setDisplay] = React.useState(target)
    90→  const fromRef = React.useRef(target)
    91→  const rafRef = React.useRef<number | null>(null)
    92→  const startRef = React.useRef<number | null>(null)
    93→
    94→  React.useEffect(() => {
    95→    if (typeof window === 'undefined') return
    96→    const from = fromRef.current
    97→    if (from === target) {
    98→      setDisplay(target)
    99→      return
   100→    }
   101→    startRef.current = null
   102→    const step = (ts: number) => {
   103→      if (startRef.current === null) startRef.current = ts
   104→      const elapsed = ts - startRef.current
   105→      const t = Math.min(1, elapsed / duration)
   106→      // easeOutCubic
   107→      const eased = 1 - Math.pow(1 - t, 3)
   108→      const next = Math.round(from + (target - from) * eased)
   109→      setDisplay(next)
   110→      if (t < 1) {
   111→        rafRef.current = window.requestAnimationFrame(step)
   112→      } else {
   113→        fromRef.current = target
   114→        setDisplay(target)
   115→      }
   116→    }
   117→    rafRef.current = window.requestAnimationFrame(step)
   118→    return () => {
   119→      if (rafRef.current !== null) {
   120→        window.cancelAnimationFrame(rafRef.current)
   121→      }
   122→      // remember where we landed so the next tween starts smoothly
   123→      fromRef.current = display
   124→    }
   125→  }, [target, duration])
   126→
   127→  return display
   128→}
