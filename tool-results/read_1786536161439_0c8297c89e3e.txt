   415→/* ---------- Master tab ---------- */
   416→function MasterTab() {
   417→  const qc = useQueryClient()
   418→  const [categoryFilter, setCategoryFilter] = React.useState<string>('all')
   419→  const [modalOpen, setModalOpen] = React.useState(false)
   420→  const [editTarget, setEditTarget] = React.useState<MasterItem | null>(null)
   421→  const [deleteTarget, setDeleteTarget] = React.useState<MasterItem | null>(null)
   422→  const [deleting, setDeleting] = React.useState(false)
   423→  const [syncing, setSyncing] = React.useState<string | null>(null)
   424→
   425→  const { data: items, isLoading } = useQuery<MasterItem[]>({
   426→    queryKey: ['master', categoryFilter],
   427→    queryFn: async () => {
   428→      const params = new URLSearchParams()
   429→      if (categoryFilter !== 'all') params.set('category', categoryFilter)
   430→      const res = await fetch(`/api/master?${params.toString()}`)
   431→      if (!res.ok) throw new Error('Failed to load master')
   432→      const json = await res.json()
   433→      return json.items as MasterItem[]
   434→    },
   435→  })
   436→
   437→  function openAdd() {
   438→    setEditTarget(null)
   439→    setModalOpen(true)
   440→  }
   441→  function openEdit(item: MasterItem) {
   442→    setEditTarget(item)
   443→    setModalOpen(true)
   444→  }
   445→
   446→  async function runSync(type: 'model' | 'dept' | 'labels') {
   447→    try {
   448→      setSyncing(type)
   449→      const res = await fetch(`/api/master/sync?type=${type}`, {
   450→        method: 'POST',
   451→      })
   452→      if (!res.ok) {
   453→        const j = await res.json().catch(() => ({}))
   454→        throw new Error(j.error ?? 'Sync failed')
   455→      }
   456→      const json = await res.json()
   457→      toast.success(
   458→        `ซิงค์ข้อมูลสำเร็จ (${type}) — อัปเดต ${json.updated ?? 0} รายการ`,
   459→      )
   460→      await qc.invalidateQueries({ queryKey: ['master'] })
   461→      await qc.invalidateQueries({ queryKey: ['devices'] })
   462→    } catch (e) {
   463→      toast.error(e instanceof Error ? e.message : 'Sync failed')
   464→    } finally {
   465→      setSyncing(null)
   466→    }
   467→  }
   468→
   469→  async function confirmDelete() {
   470→    if (!deleteTarget) return
   471→    try {
   472→      setDeleting(true)
   473→      const res = await fetch(`/api/master/${deleteTarget.id}`, {
   474→        method: 'DELETE',
   475→      })
   476→      if (!res.ok) {
   477→        const j = await res.json().catch(() => ({}))
   478→        throw new Error(j.error ?? 'Delete failed')
   479→      }
   480→      toast.success('ลบรายการแล้ว')
   481→      setDeleteTarget(null)
   482→      await qc.invalidateQueries({ queryKey: ['master'] })
   483→    } catch (e) {
   484→      toast.error(e instanceof Error ? e.message : 'Delete failed')
   485→    } finally {
   486→      setDeleting(false)
   487→    }
   488→  }
   489→
   490→  return (
   491→    <Card className="dark:border-slate-800 dark:bg-slate-900">
   492→      <CardHeader>
   493→        <CardTitle className="flex items-center gap-2 text-base text-slate-800 dark:text-slate-100">
   494→          <Database className="h-4 w-4 text-[#f97316]" />
   495→          ข้อมูลมาตรฐาน (Master Data)
   496→        </CardTitle>
   497→      </CardHeader>
   498→      <CardContent className="space-y-4">
   499→        {/* Toolbar */}
   500→        <div className="flex flex-col gap-3 lg:flex-row lg:items-center lg:justify-between">
   501→          <div className="flex flex-1 flex-col gap-2 sm:flex-row sm:items-center">
   502→            <Select
   503→              value={categoryFilter}
   504→              onValueChange={setCategoryFilter}
   505→            >
   506→              <SelectTrigger className="w-full sm:w-44">
   507→                <SelectValue placeholder="หมวดหมู่" />
   508→              </SelectTrigger>
   509→              <SelectContent>
   510→                <SelectItem value="all">หมวดหมู่ทั้งหมด</SelectItem>
   511→                {MASTER_CATEGORIES.map((c) => (
   512→                  <SelectItem key={c} value={c}>
   513→                    {c}
   514→                  </SelectItem>
   515→                ))}
   516→              </SelectContent>
