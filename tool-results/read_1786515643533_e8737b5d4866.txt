     8→function incrementUsageGroup(group, key, pages, deviceCountMap, assetNo) {
     9→  if (!group[key]) group[key] = { pages:0, devices:0, deviceSet:{} };
    10→  group[key].pages += pages;
    11→  if (!group[key].deviceSet[assetNo]) { group[key].deviceSet[assetNo]=true; group[key].devices++; }
    12→}

    14→function toUsageArray(groupMap, limit) {
    15→  return Object.keys(groupMap).map(function(k){return {name:k, pages:Math.round(groupMap[k].pages), devices:groupMap[k].devices};})
    16→    .sort(function(a,b){return b.pages-a.pages;}).slice(0, limit||999);
    17→}

    19→function getPaperUsageStats(devices, meterRows, authToken) {
    20→  var allowedSites = getUserAllowedSites(authToken);
    21→  var isSuper = allowedSites.includes('ALL');
    22→  var deviceMap = {};
    23→  (devices||[]).forEach(function(d){deviceMap[formatAssetNo(d.AssetNo)]=d;});
    24→
    25→  var rows = (meterRows||readMeterRowsCached()).filter(function(r){
    26→    // ✨ FIX: ข้าม INITIAL/RESET (ไม่มี pages) + FINAL/SEND_REPAIR/CHECKOUT (lifecycle event, ไม่ใช่รายเดือน)
    27→    var rt = String(r.Reading_Type||'').trim().toUpperCase();
    28→    if (rt==='INITIAL'||rt==='RESET'||rt==='FINAL'||rt==='SEND_REPAIR'||rt==='CHECKOUT') return false;
    29→    var assetNo=formatAssetNo(r.Asset_No);
    30→    var device=deviceMap[assetNo];
    31→    if (!isSuper && !device) return false;
    32→    if (!isSuper && device) {
    33→      var rowSite=device.Site||r.Site_At_Reading||'';
    34→      if (!allowedSites.includes(rowSite)) return false;
    35→    }
    36→    return true;
    37→  }).map(function(r){
    38→    var d=deviceMap[formatAssetNo(r.Asset_No)]||{};
    39→    return { assetNo:formatAssetNo(r.Asset_No), month:normalizeReadingMonth(r.Reading_Month||r.Reading_Date),
    40→      pages:parseNumberValue(r.Pages_BW)+parseNumberValue(r.Pages_Color), brand:d.Brand||'', model:d.Model||'',
    41→      building:d.Building||'', floor:d.Floor||'', department:d.Department||'', departmentCode:d.DepartmentCode||'',
    42→      site:d.Site||'', serial:d.Serial||'', location:d.Location||'', readingType:r.Reading_Type||'',
    43→      buildingFloor:(d.Building||'')+(d.Floor?' / ชั้น '+d.Floor:'') };
    44→  });
    45→
    46→  var byMonth={}, byDept={}, byBF={}, byDevice={};
    47→  var allMonths={};
    48→  rows.forEach(function(r){
    49→    var m=r.month; if(!m)return;
    50→    allMonths[m]=true;
    51→    incrementUsageGroup(byMonth, m, r.pages, null, r.assetNo);
    52→    var deptKey=r.departmentCode||r.department||'ไม่ระบุ';
    53→    incrementUsageGroup(byDept, deptKey, r.pages, null, r.assetNo);
    54→    incrementUsageGroup(byBF, r.buildingFloor||'ไม่ระบุ', r.pages, null, r.assetNo);
    55→    incrementUsageGroup(byDevice, r.assetNo, r.pages, null, r.assetNo);
    56→  });
    57→
    58→  var months = Object.keys(allMonths).sort();
    59→  var totalPages = rows.reduce(function(s,r){return s+r.pages;},0);
    60→  var latestMonth = months.length?months[months.length-1]:'';
    61→  var prevMonth = months.length>1?months[months.length-2]:'';
    62→  var latestPages = latestMonth?(byMonth[latestMonth]||{}).pages||0:0;
    63→  var prevPages = prevMonth?(byMonth[prevMonth]||{}).pages||0:0;
    64→  var momChange = (latestMonth && prevMonth && prevPages>0) ? Math.round((latestPages-prevPages)/prevPages*1000)/10 : null;
    65→
    66→  return {
    67→    totalPages: Math.round(totalPages), latestMonth:latestMonth, latestMonthPages:Math.round(latestPages),
    68→    momChange:momChange, avgPages:months.length?Math.round(totalPages/months.length):0,
    69→    monthlyTrend: months.map(function(m){return {month:m, pages:Math.round((byMonth[m]||{}).pages||0)};}),
    70→    byDepartment: toUsageArray(byDept, 10).map(function(d){return {name:d.name, pages:d.pages, devices:d.devices};}),
    71→    byBuildingFloor: toUsageArray(byBF, 10),
    72→    topDept: toUsageArray(byDept,1)[0]||null,
    73→    topDevice: null,
    74→    detailRows: rows
    75→  };
    76→}

    79→function getDashboardStats(authToken) {
    80→  try {

    83→    var safeToken = authToken ? String(authToken).substring(0,15) : 'guest';
    84→    var cacheKey = 'dashboard_v15_'+safeToken;
    85→    try { var c=cache.get(cacheKey); if(c) return JSON.parse(c); } catch(e){}
    86→
    87→    var devices = getAllDevicesCached(authToken);
    88→    var total=devices.length;
    89→    var active=devices.filter(function(d){return d.Status==='Active';}).length;
    90→    var byType={}, byBuilding={}, byStatus={};
    91→    // ✨ ข้อ 3: bySite — แยกจำนวนอุปกรณ์ + สถานะ + จดมิเตอร์ครบ ตามแต่ละ Site
    92→    var bySite = {};
    93→    devices.forEach(function(d){
    94→      var t=d.Type||'ไม่ระบุ', b=d.Building||'ไม่ระบุ', s=d.Status||'ไม่ระบุ';
    95→      var site=d.Site||'ไม่ระบุ';
    96→      byType[t]=(byType[t]||0)+1; byBuilding[b]=(byBuilding[b]||0)+1; byStatus[s]=(byStatus[s]||0)+1;
    97→      if (!bySite[site]) bySite[site] = { total:0, active:0, byStatus:{} };
    98→      bySite[site].total++;
    99→      if (s === 'Active') bySite[site].active++;
   100→      bySite[site].byStatus[s] = (bySite[site].byStatus[s]||0) + 1;
   101→    });
   102→
   103→    // ✨ Phase: โหลด paper usage เฉพาะเดือนปัจจุบัน (เร็วกว่าโหลดทั้ง 14,000 rows)
   104→    var nowMonth = Utilities.formatDate(new Date(),CONFIG.TIMEZONE,'yyyy-MM');
   105→    var paperUsage = {};
   106→    try {
   107→      paperUsage = getPaperUsageStatsFast(devices, nowMonth, authToken);
   108→    } catch(pe) {
   109→      // ✨ ถ้า paper usage พัง → ไม่ทำให้ทั้ง dashboard พัง — ส่ง empty object
   110→      console.warn('getPaperUsageStatsFast error (non-fatal): ' + pe.message);
   111→      paperUsage = { latestMonth: '', latestMonthPages: 0, detailRows: [], byDepartment: [], byBuildingFloor: [], monthlyTrend: [], insights: [], meterCompleteness: { readCount: 0, activeCount: 0, percent: 0 } };
   112→    }
   113→
   114→    // ✨ ข้อ 3: เพิ่ม meter completeness by Site ใช้ข้อมูลจาก paperUsage.meterCompleteness.devicesWithReading
   115→    var completenessMonth = (paperUsage.meterCompleteness && paperUsage.meterCompleteness.month) || nowMonth;
   116→    var devicesWithReadingSet = (paperUsage.meterCompleteness && paperUsage.meterCompleteness.devicesWithReading) || {};
   117→    Object.keys(bySite).forEach(function(site){
   118→      var siteDevices = devices.filter(function(d){ return (d.Site||'ไม่ระบุ') === site; });
   119→      var siteMeterRequired = 0;
   120→      var siteRead = 0;
   121→      siteDevices.forEach(function(d){
   122→        if (d.Status === 'Active' && typeof isMeterRequiredDevice === 'function' && isMeterRequiredDevice(d)) {
   123→          siteMeterRequired++;
   124→          if (devicesWithReadingSet[formatAssetNo(d.AssetNo)]) siteRead++;
   125→        }
   126→      });
   127→      bySite[site].meterRequired = siteMeterRequired;
   128→      bySite[site].meterRead = siteRead;
   129→      bySite[site].meterPercent = siteMeterRequired > 0 ? Math.round(siteRead / siteMeterRequired * 1000) / 10 : 0;
   130→      bySite[site].meterRemaining = siteMeterRequired - siteRead;
   131→    });
   132→
   133→    var result = { total:total, active:active, inactive:total-active, byType:byType, byBuilding:byBuilding, byStatus:byStatus, bySite:bySite, paperUsage:paperUsage };
   134→    try { cache.put(cacheKey, JSON.stringify(result), CONFIG.DASHBOARD_CACHE_TTL); } catch(e){}
   135→    return result;
   136→  } catch(e) { throw new Error('getDashboardStats error: '+e.message); }

   188→// ✨ เวอร์ชันเร็ว — กรองเฉพาะเดือนปัจจุบัน + เดือนก่อนหน้า (เทียบ MoM)
   189→// ✨ Phase: เพิ่ม cost analytics (BW/Color split + rate per site) + Top 10 devices + completeness + YoY
   190→function getPaperUsageStatsFast(devices, currentMonth, authToken) {
   191→  var allowedSites = getUserAllowedSites(authToken);
   192→  var isSuper = allowedSites.includes('ALL');
   193→  var deviceMap = {};
   194→  (devices||[]).forEach(function(d){deviceMap[formatAssetNo(d.AssetNo)]=d;});
   195→
   196→  // ✨ กรองเฉพาะเดือนปัจจุบัน + เดือนก่อนหน้า (ไม่อ่านทั้ง 14,000)
   197→  var prevMonth = '';
   198→  var ym = currentMonth.split('-');
   199→  if (ym.length===2) {
   200→    var y=parseInt(ym[0],10), m=parseInt(ym[1],10);
   201→    var pm = m-1; if (pm<1) {pm=12; y--;}
   202→    prevMonth = y+'-'+String(pm).padStart(2,'0');
   203→  }
   204→
   205→  // ✨ YoY: เดือนเดียวกันของปีก่อน
   206→  var yoyMonth = '';
   207→  if (ym.length===2) {
   208→    var yoyY = parseInt(ym[0],10) - 1;
   209→    yoyMonth = yoyY + '-' + ym[1];
   210→  }
   211→
   212→  var rows = readMeterRowsCached().filter(function(r){
   213→    // ✨ FIX: ข้าม INITIAL/RESET (ไม่มี pages) + FINAL/SEND_REPAIR/CHECKOUT (lifecycle event)
   214→    var rt = String(r.Reading_Type||'').trim().toUpperCase();
   215→    if (rt==='INITIAL'||rt==='RESET'||rt==='FINAL'||rt==='SEND_REPAIR'||rt==='CHECKOUT') return false;
   216→    var rowMonth = normalizeReadingMonth(r.Reading_Month||r.Reading_Date);
   217→    if (rowMonth !== currentMonth && rowMonth !== prevMonth && rowMonth !== yoyMonth) return false; // ✨ 3 เดือน: current + prev + YoY
   218→    var assetNo=formatAssetNo(r.Asset_No);
   219→    var device=deviceMap[assetNo];
   220→    if (!isSuper && !device) return false;
   221→    if (!isSuper && device) {
   222→      var rowSite=device.Site||r.Site_At_Reading||'';
   223→      if (!allowedSites.includes(rowSite)) return false;
   224→    }
   225→    return true;
   226→  }).map(function(r){
   227→    var d=deviceMap[formatAssetNo(r.Asset_No)]||{};
   228→    var pagesBW = parseNumberValue(r.Pages_BW);
   229→    var pagesColor = parseNumberValue(r.Pages_Color);
   230→    var siteName = d.Site || r.Site_At_Reading || '';
   231→    // ✨ คำนวณ cost ด้วย effective rate ของ site นั้น
   232→    var bwRate = siteName ? getEffectivePaperRate(siteName, 'BW') : getEffectivePaperRate(null, 'BW');
   233→    var colorRate = siteName ? getEffectivePaperRate(siteName, 'Color') : getEffectivePaperRate(null, 'Color');
   234→    var costBW = pagesBW * bwRate;
   235→    var costColor = pagesColor * colorRate;
   236→    return { assetNo:formatAssetNo(r.Asset_No), month:normalizeReadingMonth(r.Reading_Month||r.Reading_Date),
   237→      pages: pagesBW + pagesColor, pagesBW: pagesBW, pagesColor: pagesColor,
   238→      costBW: costBW, costColor: costColor, costTotal: costBW + costColor,
   239→      bwRate: bwRate, colorRate: colorRate,
   240→      brand:d.Brand||'', model:d.Model||'',
   241→      building:d.Building||'', floor:d.Floor||'', department:d.Department||'', departmentCode:d.DepartmentCode||'',
   242→      site: siteName, serial:d.Serial||'', location:d.Location||'', readingType:r.Reading_Type||'',
   243→      buildingFloor:(d.Building||'')+(d.Floor?' / ชั้น '+d.Floor:'') };
   244→  });
   245→
   246→  var byMonth={}, byDept={}, byBF={}, byDevice={};
   247→  var byMonthBW={}, byMonthColor={}, byMonthCost={};
   248→  var allMonths={};
   249→  // ✨ Top device tracking — เก็บข้อมูลละเอียดสำหรับ Top 10
   250→  var topDeviceMap = {};
   251→  rows.forEach(function(r){
   252→    var m=r.month; if(!m)return;
   253→    allMonths[m]=true;
   254→    incrementUsageGroup(byMonth, m, r.pages, null, r.assetNo);
   255→    incrementUsageGroup(byMonthBW, m, r.pagesBW, null, r.assetNo);
   256→    incrementUsageGroup(byMonthColor, m, r.pagesColor, null, r.assetNo);
   257→    incrementUsageGroup(byMonthCost, m, r.costTotal, null, r.assetNo);
   258→    var deptKey=r.departmentCode||r.department||'ไม่ระบุ';
   259→    incrementUsageGroup(byDept, deptKey, r.pages, null, r.assetNo);
   260→    incrementUsageGroup(byBF, r.buildingFloor||'ไม่ระบุ', r.pages, null, r.assetNo);
   261→    incrementUsageGroup(byDevice, r.assetNo, r.pages, null, r.assetNo);
   262→    // ✨ Top device: aggregate cost + BW + Color + site + brand/model
   263→    //    ใช้ latestMonth แทน currentMonth (เพราะ currentMonth อาจยังไม่มีข้อมูล)
   264→    //    latestMonth ยังไม่ถูกคำนวณตอนนี้ → เก็บไว้ใน temp map แล้วกรองทีหลัง

   276→    topDeviceMap[r.assetNo].pagesColor += r.pagesColor;
   277→    topDeviceMap[r.assetNo].costTotal += r.costTotal;

   279→  });

   281→  var months = Object.keys(allMonths).sort();
   282→  var totalPages = rows.reduce(function(s,r){return s+r.pages;},0);
   283→  var latestMonth = months.length?months[months.length-1]:'';
   284→  var prevMonthIdx = months.length>1?months[months.length-2]:'';
   285→  var latestPages = latestMonth?(byMonth[latestMonth]||{}).pages||0:0;
   286→  var prevPages = prevMonthIdx?(byMonth[prevMonthIdx]||{}).pages||0:0;
   287→  var momChange = (latestMonth && prevMonthIdx && prevPages>0) ? Math.round((latestPages-prevPages)/prevPages*1000)/10 : null;
   288→
   289→  // ✨ Filter topDeviceMap — เก็บเฉพาะ devices ที่มีข้อมูลใน latestMonth
   290→  //    และ reset ค่า pages/cost ให้เป็นเฉพาะเดือน latestMonth (ไม่รวมเดือนอื่น)
   291→  var topDeviceLatestMap = {};
   292→  if (latestMonth) {
   293→    rows.forEach(function(r) {
   294→      if (r.month === latestMonth && r.assetNo) {
   295→        if (!topDeviceLatestMap[r.assetNo]) {
   296→          var orig = topDeviceMap[r.assetNo] || {};
   297→          topDeviceLatestMap[r.assetNo] = {
   298→            assetNo: r.assetNo, brand: orig.brand || r.brand, model: orig.model || r.model,

   302→            pages: 0, pagesBW: 0, pagesColor: 0, costTotal: 0
   303→          };
   304→        }
   305→        topDeviceLatestMap[r.assetNo].pages += r.pages;
   306→        topDeviceLatestMap[r.assetNo].pagesBW += r.pagesBW;
   307→        topDeviceLatestMap[r.assetNo].pagesColor += r.pagesColor;
   308→        topDeviceLatestMap[r.assetNo].costTotal += r.costTotal;
   309→      }
   310→    });
   311→  }
   312→  topDeviceMap = topDeviceLatestMap;
   313→
   314→  // ✨ Cost summary สำหรับเดือนปัจจุบัน — แยก BW/Color โดยตรงจาก byMonthCost maps
   315→  //    ไม่ใช้ byMonthCost (เป็น total) → ต้องคำนวณ costBW/costColor แยกจาก detailRows ของเดือนนั้น
   316→  var latestBW = latestMonth ? (byMonthBW[latestMonth]||{}).pages||0 : 0;
   317→  var latestColor = latestMonth ? (byMonthColor[latestMonth]||{}).pages||0 : 0;
   318→  var latestCost = latestMonth ? (byMonthCost[latestMonth]||{}).pages||0 : 0;
   319→  // ✨ Split cost BW/Color สำหรับเดือนปัจจุบัน (สำหรับ frontend ไม่ต้อง re-filter detailRows)
   320→  var latestCostBW = 0, latestCostColor = 0;
   321→  if (latestMonth) {
   322→    rows.forEach(function(r){
   323→      if (r.month === latestMonth) {
   324→        latestCostBW += Number(r.costBW || 0);
   325→        latestCostColor += Number(r.costColor || 0);
   326→      }
   327→    });
   328→  }
   329→  // ✨ YoY cost — เทียบค่าใช้จ่ายกับเดือนเดียวกันปีก่อน
   330→  var yoyCost = 0;
   331→  if (yoyMonth) {
   332→    rows.forEach(function(r){
   333→      if (r.month === yoyMonth) yoyCost += Number(r.costTotal || 0);
   334→    });
   335→  }
   336→  var yoyCostChange = (latestCost > 0 && yoyCost > 0) ? Math.round((latestCost - yoyCost) / yoyCost * 1000) / 10 : null;
   337→
   338→  // ✨ Cost forecast — linear projection สำหรับเดือนหน้า จาก trend 3 เดือนล่าสุด
   339→  var costForecast = null;
   340→  if (months.length >= 2) {
   341→    var recentMonths = months.slice(-3); // ใช้ 3 เดือนล่าสุด (หรือ 2 ถ้ามีแค่ 2)
   342→    var recentCosts = recentMonths.map(function(m){ return (byMonthCost[m]||{}).pages || 0; });
   343→    var recentDates = recentMonths.map(function(m){
   344→      var parts = m.split('-');
   345→      return parseInt(parts[0],10) * 12 + parseInt(parts[1],10); // sequential month number
   346→    });
   347→    // Simple linear regression: y = a + b*x
   348→    var n = recentCosts.length;

   705→    function calcDeviceTrend(monthData, monthKeys) {
   706→      var values = monthKeys.map(function(m){return monthData[m]||0;});
   707→      if(values.length<2) return {trend:'stable',slope:0,avg:values[0]||0};
   708→      var n=values.length, sumX=0, sumY=0, sumXY=0, sumX2=0;
   709→      for(var i=0;i<n;i++){sumX+=i;sumY+=values[i];sumXY+=i*values[i];sumX2+=i*i;}
   710→      var slope=(n*sumXY-sumX*sumY)/(n*sumX2-sumX*sumX);
   711→      var avg=Math.round(sumY/n);
   712→      var trend='stable';
   713→      if(avg>0 && slope/avg>0.2) trend='increasing';
   714→      if(avg>0 && slope/avg<-0.2) trend='decreasing';
   715→      return {trend:trend, slope:Math.round(slope), avg:avg};
   716→    }

   878→      if(momInsight&&!momInsight.recommendation) {
   879→        momInsight.recommendation=momChange>0?'ตรวจสอบว่ามีงานพิมพ์ก้อนใหญ่ หรือมีเครื่องใหม่เพิ่ม':'ตรวจสอบว่าจดมิเตอร์ครบ หรือมีเครื่องเสีย/ย้ายออก';
   880→      }
   881→    }
   882→
   883→    // 8. สรุปภาพรวม (summary)
   884→    if(monthsArr.length>=2 && totalPages>0) {
   885→      var summaryParts = [];
   886→      summaryParts.push('รวม '+totalPages.toLocaleString()+" แผ่น ใน "+monthsArr.length+" เดือน");
   887→      if(deptArr.length>0) summaryParts.push(deptArr.length+' แผนก');
   888→      if(devArr.length>0) summaryParts.push(devArr.length+' เครื่อง');
   889→      smartInsights.push({type:'SUMMARY',tone:'info',title:'สรุปภาพรวม',text:summaryParts.join(' | ')});
   890→    }
   891→
   892→    return {totalPages:Math.round(totalPages),avgPages:avgPages,momChange:momChange,months:monthsArr,allMonths:months,byDepartment:deptArr,byBuildingFloor:bfArr,byDevice:devArr,latestMonth:latestMonth,prevMonth:prevMonth,topDept:deptArr[0]||null,topDevice:devArr[0]||null,smartInsights:smartInsights,detailRows:enriched,fromMonth:fromMonth,toMonth:toMonth};
   893→  } catch(e) { throw new Error('getPaperAnalyticsData error: '+e.message); }
   894→}

   897→function getPaperExportRows(filters, authToken) {
   898→  try {
   899→    filters=filters||{};
   900→    var requestedFormat=String(filters.format||'').trim().toLowerCase();
   901→    requirePermission(requestedFormat==='print'?'PRINT':'EXPORT_PRINT', authToken);
   902→    var toMonth=String(filters.toMonth||'').trim();
   903→    var siteFilter=String(filters.site||'').trim();
   904→    var buildingFilters=Array.isArray(filters.building)?filters.building:(filters.building?[String(filters.building)]:[]);
   905→    var deptCodeFilters=Array.isArray(filters.departmentCode)?filters.departmentCode:(filters.departmentCode||filters.department?[String(filters.departmentCode||filters.department)]:[]);
   906→    var allowedSites=getUserAllowedSites(authToken);
   907→    var isSuper=allowedSites.includes('ALL');
   908→    var devices=getAllDevicesCached(authToken);
   909→    var deviceMap={};
   910→    devices.forEach(function(d){deviceMap[formatAssetNo(d.AssetNo)]=d;});
   911→
   912→    var allMeterRows=readMeterRowsCached();
   913→    // ✨ FIX: กรองเฉพาะ MONTHLY + RETURN (รายงานรายเดือนเท่านั้น)
   914→    var filteredMeters=allMeterRows.filter(function(row){
   915→      var a=formatAssetNo(row.Asset_No), d=deviceMap[a];
   916→      if(!isSuper&&!d)return false;
   917→      var rs=row.Site_At_Reading||(d?d.Site:'');
   918→      if(!isSuper&&!allowedSites.includes(rs))return false;
   919→      // ✨ FIX: ข้าม INITIAL/RESET/FINAL/SEND_REPAIR/CHECKOUT — เอาเฉพาะรายจดมิเตอร์รายเดือน
   920→      var rt=String(row.Reading_Type||'').trim().toUpperCase();
   921→      if(rt==='INITIAL'||rt==='RESET'||rt==='FINAL'||rt==='SEND_REPAIR'||rt==='CHECKOUT')return false;
   922→      return true;
   923→    });
   924→
   925→    var months=[...new Set(filteredMeters.map(function(r){return normalizeReadingMonth(r.Reading_Month||r.Reading_Date);}).filter(Boolean))].sort();
   926→    var latestMonth=toMonth||(months.length?months[months.length-1]:'');
   927→    var previousMonth=getPreviousMonthKey(latestMonth);
   928→    var agg={};
   929→
   930→    filteredMeters.forEach(function(row){
   931→      var month=normalizeReadingMonth(row.Reading_Month||row.Reading_Date);
   932→      if(!month)return;
   933→      if(month!==latestMonth&&month!==previousMonth)return;
   934→      var assetNo=formatAssetNo(row.Asset_No), key=assetNo+'|'+month;
   935→      // ✨ FIX: แยก BW และ Color — เดิมใช้ || รวมกัน (Prev_Meter_BW || Prev_Meter_Color)
   936→      //   ถ้า BW=0 จะใช้ Color (ผิด — คนละ meter)
   937→      //   ใหม่: ใช้ BW + Color แยก (sum รวมเป็น meterStart/meterEnd)
   938→      var prevBW=Number(row.Prev_Meter_BW)||0, prevColor=Number(row.Prev_Meter_Color)||0;
   939→      var currBW=Number(row.Meter_BW)||0, currColor=Number(row.Meter_Color)||0;
   940→      var prev=prevBW+prevColor;
   941→      var curr=currBW+currColor;
   942→      var pages=Number(row.Pages_BW||0)+Number(row.Pages_Color||0);
   943→      if(!agg[key])agg[key]={pages:0,meterStart:prev,meterEnd:curr,remark:''};
   944→      agg[key].pages+=pages;
   945→      if(prev>0)agg[key].meterStart=agg[key].meterStart>0?Math.min(agg[key].meterStart,prev):prev;
   946→      if(curr>0)agg[key].meterEnd=Math.max(agg[key].meterEnd,curr);
   947→      if(!agg[key].remark&&row.Remark)agg[key].remark=String(row.Remark);
   948→    });
   949→
   950→    return devices.filter(function(d){
   951→      var a=formatAssetNo(d.AssetNo), isReq=d.MeterRequired===true||String(d.MeterRequired).toUpperCase()==='TRUE';
   952→      return isReq||agg[a+'|'+latestMonth]||agg[a+'|'+previousMonth];
   953→    }).filter(function(d){return !siteFilter||String(d.Site||'')===siteFilter;})
   954→    .filter(function(d){return buildingFilters.length===0||buildingFilters.indexOf(String(d.Building||''))!==-1;})
   955→    .filter(function(d){return deptCodeFilters.length===0||deptCodeFilters.indexOf(String(d.DepartmentCode||''))!==-1;})
   956→    .map(function(d){
   957→      var a=formatAssetNo(d.AssetNo);
   958→      var curr=agg[a+'|'+latestMonth]||{pages:0,meterStart:0,meterEnd:0,remark:''};
   959→      var prev=agg[a+'|'+previousMonth]||{pages:0,meterStart:0,meterEnd:0};
   960→      var pc=Number(curr.pages||0), pp=Number(prev.pages||0), diff=pc-pp;
   961→      var pct=pp>0?(diff/pp)*100:0;
   962→      var me=Number(curr.meterEnd||0), ms=Number(curr.meterStart||0);
   963→      if(ms===0&&Number(prev.meterEnd||0)>0)ms=Number(prev.meterEnd);
   964→      if(ms===0&&me>0&&pc>0)ms=Math.max(0,me-pc);
   965→      if(me===0&&ms>0&&pc>0)me=ms+pc;
   966→      // ✨ FIX: คำนวณต้นทุนตาม site rate ของอุปกรณ์นั้น
   967→      var dSite=d.Site||'';
   968→      var bwRate=getEffectivePaperRate(dSite,'BW');
   969→      var colorRate=getEffectivePaperRate(dSite,'Color');
   970→      var rowCost=pc*bwRate; // BW only (ส่วนใหญ่ UDH ใช้ BW)
   971→      return{brand:d.Brand||'',model:d.Model||'',serial:d.Serial||'',buildingFloor:d.Building||'',floor:d.Floor||'',
   972→        department:d.Department||d.DepartmentCode||'',site:dSite,meterStart:ms,meterEnd:me,pagesCurrent:pc,pagesPrevious:pp,
   973→        pagesDiff:diff,momPercent:Math.round(pct*100)/100,status:d.Status||'',remark:curr.remark||'',
   974→        rowCost:rowCost,bwRate:bwRate,colorRate:colorRate};
   975→    });
   976→  } catch(e) { throw new Error('getPaperExportRows error: '+e.message); }
   977→}

   988→function getCompletedMeterReadings(params, authToken) {
   989→  try {
   990→    params = params || {};
   991→    requirePermission('METER_WRITE', authToken);
   992→    var month = String(params.month || '').trim();
   993→    if (!month) return { success: false, message: 'กรุณาระบุเดือน' };
   994→    var siteFilter = String(params.site || '').trim();
   995→    var buildingFilter = String(params.building || '').trim();
   996→    var deptFilter = String(params.department || '').trim();
   997→
   998→    var allowedSites = getUserAllowedSites(authToken);
   999→    var isSuper = allowedSites.includes('ALL');
  1000→    var devices = getAllDevicesCached(authToken);
  1001→    var deviceMap = {};

  1003→
  1004→    var allMeterRows = readMeterRowsCached();

  1009→    allMeterRows.forEach(function(row) {
  1010→      var rowMonth = normalizeReadingMonth(row.Reading_Month || row.Reading_Date);
  1011→      if (!rowMonth) return;
  1012→      if (rowMonth !== month && rowMonth !== previousMonth) return;
  1013→      // ✨ FIX: ข้าม INITIAL/RESET/FINAL/SEND_REPAIR/CHECKOUT — เอาเฉพาะรายจดรายเดือน
  1014→      var rt = String(row.Reading_Type || '').trim().toUpperCase();
  1015→      if (rt === 'INITIAL' || rt === 'RESET' || rt === 'FINAL' || rt === 'SEND_REPAIR' || rt === 'CHECKOUT') return;
  1016→      var assetNo = formatAssetNo(row.Asset_No);
  1017→      var key = assetNo + '|' + rowMonth;
  1018→      // ✨ FIX: ใช้ + แยก BW กับ Color (เดิมใช้ || ทำให้เสีย BW=0 ตกไปใช้ Color)
  1019→      var prev = (Number(row.Prev_Meter_BW) || 0) + (Number(row.Prev_Meter_Color) || 0);
  1020→      var curr = (Number(row.Meter_BW) || 0) + (Number(row.Meter_Color) || 0);
  1021→      var pages = Number(row.Pages_BW || 0) + Number(row.Pages_Color || 0);
  1022→      if (!agg[key]) agg[key] = { pages: 0, meterStart: prev, meterEnd: curr, remark: '' };
  1023→      agg[key].pages += pages;
  1024→      if (prev > 0) agg[key].meterStart = agg[key].meterStart > 0 ? Math.min(agg[key].meterStart, prev) : prev;
  1025→      if (curr > 0) agg[key].meterEnd = Math.max(agg[key].meterEnd, curr);
  1026→      if (!agg[key].remark && row.Remark) agg[key].remark = String(row.Remark);
  1027→    });

  1029→    // ✨ กรองเฉพาะอุปกรณ์ที่ "จดแล้ว" ในเดือนปัจจุบัน (มี reading ในเดือนที่เลือก)
  1030→    var result = devices
  1031→      .filter(function(d) {
  1032→        var a = formatAssetNo(d.AssetNo);
  1033→        return agg[a + '|' + month]; // ✨ ต้องมี reading เดือนนี้

  1043→        if (!deptFilter) return true;
  1044→        return String(d.Department || '') === deptFilter || String(d.DepartmentCode || '') === deptFilter;

  1060→          model: d.Model || '',
  1061→          serial: d.Serial || '',
  1062→          building: d.Building || '',
  1063→          floor: d.Floor || '',
  1064→          department: d.Department || d.DepartmentCode || '',

  1066→          endMeter: me,
  1067→          pagesCurrent: pc,
  1068→          pagesPrevious: pp,
