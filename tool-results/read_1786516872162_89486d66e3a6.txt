    58→interface ReminderDevice {
    59→  id: string
    60→  assetCode: string
    61→  name: string
    62→  brand: string
    63→  model: string
    64→  site: string
    65→  lastMeterReading: number
    66→}
    67→interface ReminderEntry {
    68→  device: ReminderDevice
    69→  lastReadingDate: string | null
    70→  daysOverdue: number
    71→}
    72→interface RemindersData {
    73→  hasActiveCycle: boolean
    74→  cycle: { id: string; name: string; startDate: string; endDate: string; status: string } | null
    75→  reminders: ReminderEntry[]
    76→  totalRead: number
    77→  totalUnread: number
    78→}
    79→
    80→function todayISO(): string {
    81→  return new Date().toISOString().slice(0, 10)
    82→}
    83→
    84→function daysBetween(a: string, b: string): number {
    85→  const ms = new Date(b).getTime() - new Date(a).getTime()
    86→  return Math.round(ms / (1000 * 60 * 60 * 24))
    87→}
