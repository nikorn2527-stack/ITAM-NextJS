   331→  async function save() {
   332→    if (!form.assetCode || !form.name || !form.brand || !form.model || !form.type) {
   333→      toast.error('กรุณากรอกข้อมูลที่จำเป็น (รหัส, ชื่อ, แบรนด์, รุ่น, ประเภท)')
   334→      return
   335→    }
   336→    try {
   337→      setSaving(true)
   338→      const payload = {
   339→        ...form,
   340→        serialNumber: form.serialNumber || null,
   341→        department: form.department || null,
   342→        departmentCode: form.departmentCode || null,
   343→        parentRef: form.parentRef || null,
   344→        displayLabel: form.displayLabel || null,
   345→        location: form.location || null,
   346→        purchaseDate: form.purchaseDate || null,
   347→        warrantyMonths: Number(form.warrantyMonths) || 12,
   348→        purchasePrice:
   349→          form.purchasePrice === '' ? null : Number(form.purchasePrice),
   350→        salvageValue:
   351→          form.salvageValue === '' ? 0 : Number(form.salvageValue),
   352→        usefulLife:
   353→          form.usefulLife === '' ? null : Number(form.usefulLife),
   354→      }
   355→      const isEdit = Boolean(form.id)
   356→      const url = isEdit ? `/api/devices/${form.id}` : '/api/devices'
   357→      const method = isEdit ? 'PUT' : 'POST'
   358→      const res = await fetch(url, {
   359→        method,
   360→        headers: { 'Content-Type': 'application/json' },
   361→        body: JSON.stringify(payload),
   362→      })
   363→      if (!res.ok) {
   364→        const j = await res.json().catch(() => ({}))
   365→        throw new Error(j.error ?? 'Save failed')
   366→      }
   367→      toast.success(isEdit ? 'แก้ไขอุปกรณ์แล้ว' : 'เพิ่มอุปกรณ์ใหม่แล้ว')
   368→      setDialogOpen(false)
   369→      await qc.invalidateQueries({ queryKey: ['devices'] })
   370→      await qc.invalidateQueries({ queryKey: ['dashboard'] })
   371→    } catch (e) {
   372→      toast.error(e instanceof Error ? e.message : 'Save failed')
   373→    } finally {
   374→      setSaving(false)
   375→    }
   376→  }

   458→  async function applyBulkStatus() {
   459→    if (!bulkStatus || selectedIds.size === 0) return
   460→    setBulkAction(true)
   461→    const ids = Array.from(selectedIds)
   462→    const results = await Promise.allSettled(
   463→      ids.map((id) =>
   464→        fetch(`/api/devices/${id}`, {
   465→          method: 'PUT',
   466→          headers: { 'Content-Type': 'application/json' },
   467→          body: JSON.stringify({ status: bulkStatus }),
   468→        }),
   469→      ),
   470→    )
   471→    const ok = results.filter((r) => r.status === 'fulfilled').length
   472→    const fail = results.length - ok
   473→    const label = statusLabel(bulkStatus)
   474→    if (fail === 0) {
   475→      toast.success(`อัปเดต ${ok} เครื่องเป็น "${label}" แล้ว`)
   476→    } else {
   477→      toast.warning(`อัปเดตสำเร็จ ${ok} เครื่อง, ล้มเหลว ${fail} เครื่อง`)
   478→    }
   479→    await logBulkAction('BULK_UPDATE', `เปลี่ยนสถานะอุปกรณ์ ${ok} เครื่องเป็น ${label}`, {
   480→      status: bulkStatus,
   481→      count: ok,
   482→      failed: fail,
   483→      deviceIds: ids,
   484→    })
   485→    setBulkStatus('')
   486→    clearSelection()
   487→    await qc.invalidateQueries({ queryKey: ['devices'] })
   488→    await qc.invalidateQueries({ queryKey: ['dashboard'] })
   489→    await qc.invalidateQueries({ queryKey: ['audit'] })
   490→    setBulkAction(false)
   491→  }

   493→  async function applyBulkTransfer() {
   494→    if (!bulkSite || selectedIds.size === 0) return
   495→    setBulkAction(true)
   496→    const ids = Array.from(selectedIds)
   497→    const today = new Date().toISOString().slice(0, 10)
   498→    const results = await Promise.allSettled(
   499→      ids.map((id) =>
   500→        fetch(`/api/devices/${id}/transfer`, {
   501→          method: 'POST',
   502→          headers: { 'Content-Type': 'application/json' },
   503→          body: JSON.stringify({ toSite: bulkSite, transferDate: today }),
   504→        }),
   505→      ),
   506→    )
   507→    const ok = results.filter((r) => r.status === 'fulfilled').length
   508→    const fail = results.length - ok
   509→    const siteName =
   510→      (sites ?? []).find((s) => s.code === bulkSite)?.name ?? bulkSite
   511→    if (fail === 0) {
   512→      toast.success(`ย้าย ${ok} เครื่องไปสาขา ${siteName} แล้ว`)
   513→    } else {
   514→      toast.warning(`ย้ายสำเร็จ ${ok} เครื่อง, ล้มเหลว ${fail} เครื่อง`)
   515→    }
   516→    await logBulkAction('BULK_TRANSFER', `ย้ายอุปกรณ์ ${ok} เครื่องไปสาขา ${siteName}`, {
   517→      toSite: bulkSite,
   518→      count: ok,
   519→      failed: fail,
   520→      deviceIds: ids,
   521→    })
   522→    setBulkSite('')
   523→    clearSelection()
   524→    await qc.invalidateQueries({ queryKey: ['devices'] })
   525→    await qc.invalidateQueries({ queryKey: ['dashboard'] })
   526→    await qc.invalidateQueries({ queryKey: ['audit'] })
   527→    setBulkAction(false)
   528→  }

   530→  async function applyBulkDelete() {
   531→    if (selectedIds.size === 0) return
   532→    setBulkAction(true)
   533→    const ids = Array.from(selectedIds)
   534→    const results = await Promise.allSettled(
   535→      ids.map((id) => fetch(`/api/devices/${id}`, { method: 'DELETE' })),
   536→    )
   537→    const ok = results.filter((r) => r.status === 'fulfilled').length
   538→    const fail = results.length - ok
   539→    if (fail === 0) {

  1328→// A combobox-like input: lets the user either type or pick from a dropdown list
  1329→function SelectValueInput({
  1330→  value,
  1331→  onChange,
  1332→  options,
  1333→  placeholder,
  1334→}: {
  1335→  value: string
  1336→  onChange: (v: string) => void
  1337→  options: { value: string; label: string }[]
  1338→  placeholder?: string
  1339→}) {
  1340→  const [open, setOpen] = React.useState(false)
  1341→  const inputRef = React.useRef<HTMLInputElement>(null)
  1342→  return (
  1343→    <div className="relative">
  1344→      <Input
  1345→        ref={inputRef}
  1346→        value={value}
  1347→        onChange={(e) => onChange(e.target.value)}
  1348→        onFocus={() => setOpen(true)}
  1349→        onBlur={() => setTimeout(() => setOpen(false), 150)}
  1350→        placeholder={placeholder}
  1351→      />
  1352→      {open && options.length > 0 && (
  1353→        <div className="itam-scroll absolute z-50 mt-1 max-h-48 w-full overflow-y-auto rounded-md border bg-white shadow-lg">
  1354→          {options.map((o) => (
  1355→            <button
  1356→              key={o.value}
  1357→              type="button"
  1358→              className="block w-full px-3 py-1.5 text-left text-sm hover:bg-slate-100"
  1359→              onMouseDown={(e) => {
  1360→                e.preventDefault()
  1361→                onChange(o.value)
  1362→                setOpen(false)
  1363→              }}
  1364→            >
  1365→              {o.label}
  1366→            </button>
  1367→          ))}
  1368→        </div>
  1369→      )}
  1370→    </div>
  1371→  )
