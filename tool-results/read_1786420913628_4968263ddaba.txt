    57→interface ReminderDevice {
    58→  id: string
    59→  assetCode: string
    60→  name: string
    61→  brand: string
    62→  model: string
    63→  site: string
    64→  lastMeterReading: number
    65→}
    66→interface ReminderEntry {
    67→  device: ReminderDevice
    68→  lastReadingDate: string | null
    69→  daysOverdue: number
    70→}
    71→interface RemindersData {
    72→  hasActiveCycle: boolean
    73→  cycle: { id: string; name: string; startDate: string; endDate: string; status: string } | null
    74→  reminders: ReminderEntry[]
    75→  totalRead: number
    76→  totalUnread: number
    77→}
    78→
    79→function todayISO(): string {
    80→  return new Date().toISOString().slice(0, 10)
    81→}
    82→
    83→function daysBetween(a: string, b: string): number {
    84→  const ms = new Date(b).getTime() - new Date(a).getTime()
    85→  return Math.round(ms / (1000 * 60 * 60 * 24))
    86→}
