    60→// ============ Cascading dropdown hook ============
    61→// Fetches distinct values for a location field, filtered by previous selections.
    62→function useCascadingOptions(
    63→  field: 'building' | 'floor' | 'department' | 'location',
    64→  parents: { site?: string; building?: string; floor?: string; department?: string },
    65→): string[] {
    66→  const [values, setValues] = React.useState<string[]>([])
    67→  const { site, building, floor, department } = parents
    68→  React.useEffect(() => {
    69→    let cancelled = false
    70→    const params = new URLSearchParams({ field })
    71→    if (site) params.set('site', site)
    72→    if (building) params.set('building', building)
    73→    if (floor) params.set('floor', floor)
    74→    if (department) params.set('department', department)
    75→    fetch(`/api/itam/devices/cascading?${params}`)
    76→      .then((r) => r.ok ? r.json() : Promise.reject(r))
    77→      .then((j: { values: string[] }) => { if (!cancelled) setValues(j.values ?? []) })
    78→      .catch(() => { if (!cancelled) setValues([]) })
    79→    return () => { cancelled = true }
    80→  }, [field, site, building, floor, department])
    81→  return values
    82→}

   143→// ============ Search debounce hook (200ms) ============
   144→function useDebounced<T>(value: T, delayMs = 200): [T, boolean] {
   145→  const [debounced, setDebounced] = React.useState<T>(value)
   146→  const [isPending, setIsPending] = React.useState(false)
   147→  React.useEffect(() => {
   148→    setIsPending(true)
   149→    const t = setTimeout(() => {
   150→      setDebounced(value)
   151→      setIsPending(false)
   152→    }, delayMs)
   153→    return () => clearTimeout(t)
   154→  }, [value, delayMs])
   155→  return [debounced, isPending]
   156→}

   158→// ============ Highlight matched text ============
   159→function Highlight({ text, query }: { text: string; query: string }) {
   160→  if (!query.trim()) return <>{text || '—'}</>
   161→  const q = query.trim()
   162→  // Escape regex special chars
   163→  const esc = q.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
   164→  const parts = (text || '').split(new RegExp(`(${esc})`, 'ig'))
   165→  return (
   166→    <>
   167→      {parts.map((p, i) =>
   168→        p.toLowerCase() === q.toLowerCase()
   169→          ? <mark key={i} className="rounded bg-orange-200 px-0.5 text-slate-900 dark:bg-orange-500/60 dark:text-white">{p}</mark>
   170→          : <React.Fragment key={i}>{p}</React.Fragment>,
   171→      )}
   172→    </>
   173→  )
   174→}

   176→// ============ Recent searches (localStorage) ============
   177→const RECENT_KEY = 'itam-recent-searches'
   178→function loadRecent(): string[] {
   179→  if (typeof window === 'undefined') return []
   180→  try {
   181→    const raw = window.localStorage.getItem(RECENT_KEY)
   182→    if (!raw) return []
   183→    const arr = JSON.parse(raw)
   184→    return Array.isArray(arr) ? arr.filter((x) => typeof x === 'string').slice(0, 6) : []
   185→  } catch {
   186→    return []
   187→  }
   188→}
   189→function saveRecent(q: string) {
   190→  if (typeof window === 'undefined' || !q.trim()) return
   191→  try {
   192→    const cur = loadRecent()
   193→    const next = [q.trim(), ...cur.filter((x) => x !== q.trim())].slice(0, 6)
   194→    window.localStorage.setItem(RECENT_KEY, JSON.stringify(next))
   195→  } catch { /* ignore */ }
   196→}
