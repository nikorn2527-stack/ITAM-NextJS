    20→import { ItamDeviceDetailSheet } from './itam-device-detail-sheet'
    21→
    22→interface Device {
    23→  id: string; assetNo: string; deviceType: string | null; brand: string | null
    24→  model: string | null; serial: string | null; status: string; site: string | null
    25→  department: string | null; building: string | null; floor: string | null
    26→  location: string | null; meterRequired: boolean; _count?: { meterReadings: number; locationHistories: number; assignments: number; maintenanceLogs: number }
    27→}
    28→interface DevicesResponse {
    29→  devices: Device[]; pagination: { page: number; limit: number; total: number; totalPages: number }
    30→}
    31→
    32→const STATUS_BADGE: Record<string, string> = {
    33→  Active: 'bg-emerald-100 text-emerald-700 border-emerald-200 dark:bg-emerald-950 dark:text-emerald-300 dark:border-emerald-800',
    34→  Inactive: 'bg-slate-100 text-slate-600 border-slate-200 dark:bg-slate-800 dark:text-slate-300',
    35→  'In Stock': 'bg-amber-100 text-amber-700 border-amber-200 dark:bg-amber-950 dark:text-amber-300 dark:border-amber-800',
    36→  'Pending Repair': 'bg-orange-100 text-orange-700 border-orange-200 dark:bg-orange-950 dark:text-orange-300 dark:border-orange-800',
    37→  Retired: 'bg-rose-100 text-rose-700 border-rose-200 dark:bg-rose-950 dark:text-rose-300 dark:border-rose-800',
    38→}
    39→
    40→const STATUS_OPTIONS = ['Active', 'In Stock', 'Pending Repair', 'Inactive', 'Retired']
    41→
    42→type DeviceForm = {
    43→  assetNo: string; deviceType: string; brand: string; model: string; serial: string
    44→  building: string; floor: string; department: string; location: string; departmentCode: string
    45→  status: string; site: string; contractNo: string; ip: string; mac: string; remoteId: string
    46→  vendor: string; installDate: string; warrantyEnd: string; deviceGroup: string; costCenter: string
    47→  meterRequired: boolean; meterMode: string; assetSiteCode: string; remark: string
    48→}
    49→
    50→const EMPTY_FORM: DeviceForm = {
    51→  assetNo: '', deviceType: '', brand: '', model: '', serial: '',
    52→  building: '', floor: '', department: '', location: '', departmentCode: '',
    53→  status: 'Active', site: '', contractNo: '', ip: '', mac: '', remoteId: '',
    54→  vendor: '', installDate: '', warrantyEnd: '', deviceGroup: '', costCenter: '',
    55→  meterRequired: false, meterMode: 'TOTAL', assetSiteCode: '', remark: '',
    56→}
    57→

    59→  { key: 'assetNo', label: 'รหัสสินทรัพย์' },
    60→  { key: 'deviceType', label: 'ประเภท' },
    61→  { key: 'brand', label: 'แบรนด์' },
    62→  { key: 'model', label: 'รุ่น' },
    63→  { key: 'serial', label: 'Serial' },

    80→  { key: 'meterRequired', label: 'ต้องจดมิเตอร์' },
    81→  { key: 'meterMode', label: 'โหมดมิเตอร์' },

    83→  { key: 'remark', label: 'หมายเหตุ' },
    84→]
    85→
    86→export function ItamDevices() {
    87→  const qc = useQueryClient()
    88→  const [search, setSearch] = React.useState('')
    89→  const [status, setStatus] = React.useState('all')
    90→  const [page, setPage] = React.useState(1)
    91→  const [limit] = React.useState(20)
    92→
    93→  // Detail sheet
    94→  const [detailAssetNo, setDetailAssetNo] = React.useState<string | null>(null)
    95→  const [detailOpen, setDetailOpen] = React.useState(false)
    96→
    97→  // CRUD dialog
    98→  const [crudOpen, setCrudOpen] = React.useState(false)
    99→  const [editAssetNo, setEditAssetNo] = React.useState<string | null>(null)
   100→  const [form, setForm] = React.useState<DeviceForm>(EMPTY_FORM)
   101→  const [saving, setSaving] = React.useState(false)
   102→
   103→  // Delete confirm
   104→  const [deleteAssetNo, setDeleteAssetNo] = React.useState<string | null>(null)
   105→  const [deleting, setDeleting] = React.useState(false)
   106→
   107→  // CSV export
   108→  const [exporting, setExporting] = React.useState(false)
   109→
   110→  const queryKey = React.useMemo(() => ['itam-devices', search, status, page, limit], [search, status, page, limit])
   111→
   112→  const { data, isLoading } = useQuery<DevicesResponse>({
   113→    queryKey,
   114→    queryFn: async () => {
   115→      const params = new URLSearchParams({ page: String(page), limit: String(limit) })
   116→      if (search) params.set('search', search)
   117→      if (status !== 'all') params.set('status', status)
   118→      const res = await fetch(`/api/itam/devices?${params}`)
   119→      if (!res.ok) throw new Error('Failed')
   120→      return res.json()
   121→    },
   122→  })
   123→
   124→  function onSearch(val: string) {
   125→    setSearch(val)
   126→    setPage(1)
   127→  }
   128→
   129→  function openAdd() {
   130→    setEditAssetNo(null)
   131→    setForm(EMPTY_FORM)
   132→    setCrudOpen(true)
   133→  }
   134→
   135→  function openEdit(assetNo: string) {
   136→    setEditAssetNo(assetNo)
   137→    // Fetch current device fields then open dialog
   138→    fetch(`/api/itam/devices/${assetNo}`)
   139→      .then(r => r.ok ? r.json() : Promise.reject(r))
   140→      .then((j: { device: Device & Record<string, unknown> }) => {
   141→        const d = j.device
   142→        setForm({
   143→          assetNo: d.assetNo, deviceType: d.deviceType || '', brand: d.brand || '', model: d.model || '',
   144→          serial: d.serial || '', building: d.building || '', floor: d.floor || '',
   145→          department: d.department || '', location: d.location || '', departmentCode: d.departmentCode || '',
   146→          status: d.status, site: d.site || '', contractNo: (d as { contractNo?: string }).contractNo || '',
   147→          ip: (d as { ip?: string }).ip || '', mac: (d as { mac?: string }).mac || '',
   148→          remoteId: (d as { remoteId?: string }).remoteId || '',
   149→          vendor: (d as { vendor?: string }).vendor || '',
   150→          installDate: (d as { installDate?: string }).installDate || '',
   151→          warrantyEnd: (d as { warrantyEnd?: string }).warrantyEnd || '',
   152→          deviceGroup: (d as { deviceGroup?: string }).deviceGroup || '',
   153→          costCenter: (d as { costCenter?: string }).costCenter || '',
   154→          meterRequired: d.meterRequired, meterMode: (d as { meterMode?: string }).meterMode || 'TOTAL',
   155→          assetSiteCode: (d as { assetSiteCode?: string }).assetSiteCode || '',
   156→          remark: d.remark || '',
   157→        })
   158→        setCrudOpen(true)
   159→      })
   160→      .catch(() => toast.error('โหลดข้อมูลไม่สำเร็จ'))
   161→  }
   162→
   163→  async function saveDevice() {
   164→    if (!form.assetNo.trim()) {
   165→      toast.error('กรุณากรอกรหัสสินทรัพย์')
   166→      return
   167→    }
   168→    try {
   169→      setSaving(true)
   170→      const payload = {
   171→        ...form,
   172→        meterRequired: form.meterRequired,
   173→      }
   174→      if (editAssetNo) {
   175→        const res = await fetch(`/api/itam/devices/${editAssetNo}`, {
   176→          method: 'PUT', headers: { 'Content-Type': 'application/json' },
   177→          body: JSON.stringify(payload),
   178→        })
   179→        if (!res.ok) {
   180→          const j = await res.json().catch(() => ({}))
   181→          throw new Error(j.error || 'Failed')
   182→        }
   183→        toast.success('แก้ไขอุปกรณ์แล้ว')
   184→      } else {
   185→        const res = await fetch('/api/itam/devices', {
   186→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   187→          body: JSON.stringify(payload),
   188→        })
   189→        if (!res.ok) {
   190→          const j = await res.json().catch(() => ({}))
   191→          throw new Error(j.error || 'Failed')
   192→        }
   193→        toast.success('เพิ่มอุปกรณ์แล้ว')
   194→      }
   195→      setCrudOpen(false)
   196→      await qc.invalidateQueries({ queryKey: ['itam-devices'] })
   197→      await qc.invalidateQueries({ queryKey: ['itam-dashboard'] })
   198→    } catch (e) {
   199→      toast.error(e instanceof Error ? e.message : 'บันทึกไม่สำเร็จ')
   200→    } finally {
   201→      setSaving(false)
   202→    }
   203→  }
   204→
   205→  async function confirmDelete() {
   206→    if (!deleteAssetNo) return
   207→    try {
   208→      setDeleting(true)
   209→      const res = await fetch(`/api/itam/devices/${deleteAssetNo}`, { method: 'DELETE' })
   210→      if (!res.ok) {
   211→        const j = await res.json().catch(() => ({}))
   212→        throw new Error(j.error || 'Failed')
   213→      }
   214→      toast.success(`ลบ ${deleteAssetNo} แล้ว`)
   215→      setDeleteAssetNo(null)
   216→      await qc.invalidateQueries({ queryKey: ['itam-devices'] })
   217→      await qc.invalidateQueries({ queryKey: ['itam-dashboard'] })
   218→    } catch (e) {
   219→      toast.error(e instanceof Error ? e.message : 'ลบไม่สำเร็จ')
   220→    } finally {
   221→      setDeleting(false)
   222→    }
   223→  }
   224→
   225→  async function exportCsv() {
   226→    try {
   227→      setExporting(true)
   228→      toast.info('กำลังดึงข้อมูลทั้งหมด...')
   229→      const res = await fetch('/api/itam/devices?limit=100')
   230→      if (!res.ok) throw new Error('Failed')
   231→      const j: DevicesResponse = await res.json()
   232→      const rows = j.devices.map(d => ({
   233→        assetNo: d.assetNo, deviceType: d.deviceType || '', brand: d.brand || '',
   234→        model: d.model || '', serial: d.serial || '', status: d.status,
   235→        site: d.site || '', building: d.building || '', floor: d.floor || '',
   236→        department: d.department || '', departmentCode: '', location: d.location || '',
   237→        deviceGroup: '', costCenter: '', contractNo: '', vendor: '',
   238→        ip: '', mac: '', remoteId: '', installDate: '', warrantyEnd: '',
   239→        meterRequired: d.meterRequired ? 'Yes' : 'No', meterMode: '', assetSiteCode: '',
   240→        remark: '',
   241→      }))
   242→      downloadCsv(`devices-${dateStamp()}.csv`, rows, CSV_HEADERS)
   243→      toast.success(`ส่งออก ${rows.length} เครื่อง`)
   244→    } catch (e) {
   245→      toast.error(e instanceof Error ? e.message : 'ส่งออกไม่สำเร็จ')
   246→    } finally {
   247→      setExporting(false)
   248→    }
   249→  }
   250→
   251→  const devices = data?.devices ?? []
   252→  const total = data?.pagination.total ?? 0
   253→  const totalPages = data?.pagination.totalPages ?? 0
   254→
   255→  return (
   256→    <div className="space-y-4 p-4 md:p-6">
   257→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   258→        <div>
   259→          <h1 className="text-2xl font-bold text-slate-800 dark:text-slate-100">จัดการอุปกรณ์ (Real DB)</h1>
   260→          <p className="text-sm text-slate-500 dark:text-slate-400">ข้อมูลจริง {total.toLocaleString()} เครื่อง</p>
   261→        </div>
   262→        <div className="flex flex-wrap gap-2">
   263→          <Button variant="outline" onClick={exportCsv} disabled={exporting} className="dark:bg-slate-800 dark:border-slate-700">
   264→            <Download className="h-4 w-4" /> ส่งออก CSV
   265→          </Button>
   266→          <Button variant="outline" onClick={() => qc.invalidateQueries({ queryKey })} className="dark:bg-slate-800 dark:border-slate-700">
   267→            <RefreshCw className="h-4 w-4" /> รีเฟรช
   268→          </Button>
   269→          <Button className="bg-[#f97316] text-white hover:bg-[#ea580c]" onClick={openAdd}>
   270→            <Plus className="h-4 w-4" /> เพิ่มอุปกรณ์
   271→          </Button>
   272→        </div>
   273→      </div>
   274→
   275→      {/* Toolbar */}
   276→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center">
   277→        <div className="relative flex-1">
   278→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
   279→          <Input
   280→            placeholder="ค้นหารหัส, แบรนด์, รุ่น, SN..."
   281→            value={search}
   282→            onChange={(e) => onSearch(e.target.value)}
   283→            className="pl-9 dark:bg-slate-800 dark:border-slate-700"
   284→          />
   285→        </div>
   286→        <Select value={status} onValueChange={(v) => { setStatus(v); setPage(1) }}>
   287→          <SelectTrigger className="w-full sm:w-48 dark:bg-slate-800 dark:border-slate-700">
   288→            <SelectValue placeholder="สถานะ" />
   289→          </SelectTrigger>
   290→          <SelectContent>
   291→            <SelectItem value="all">สถานะทั้งหมด</SelectItem>
   292→            <SelectItem value="Active">ใช้งานอยู่</SelectItem>
   293→            <SelectItem value="In Stock">สำรอง</SelectItem>
   294→            <SelectItem value="Pending Repair">ส่งซ่อม</SelectItem>
   295→            <SelectItem value="Inactive">ไม่ใช้งาน</SelectItem>
   296→            <SelectItem value="Retired">ตัดของออก</SelectItem>
   297→          </SelectContent>
   298→        </Select>
   299→      </div>
   300→
   301→      {/* Table */}
   302→      <Card className="shadow-sm dark:border-slate-800 dark:bg-slate-900">
   303→        <CardContent className="p-0">
   304→          <div className="itam-scroll max-h-[60vh] overflow-auto">
   305→            <Table>
   306→              <TableHeader className="sticky top-0 z-10 bg-slate-50/80 backdrop-blur-sm dark:bg-slate-900/80">
   307→                <TableRow>
   308→                  <TableHead className="w-16">รหัส</TableHead>
   309→                  <TableHead>ประเภท</TableHead>
   310→                  <TableHead>แบรนด์/รุ่น</TableHead>
   311→                  <TableHead>สถานะ</TableHead>
   312→                  <TableHead>สาขา</TableHead>
   313→                  <TableHead>แผนก</TableHead>
   314→                  <TableHead className="text-center">มิเตอร์</TableHead>
   315→                  <TableHead className="text-right w-32">จัดการ</TableHead>
   316→                </TableRow>
   317→              </TableHeader>
   318→              <TableBody>
   319→                {isLoading ? (
   320→                  Array.from({ length: 10 }).map((_, i) => (
   321→                    <TableRow key={`sk-${i}`}>
   322→                      <TableCell colSpan={8}><Skeleton className="h-6 w-full" /></TableCell>
   323→                    </TableRow>
   324→                  ))
   325→                ) : devices.length === 0 ? (
   326→                  <TableRow>
   327→                    <TableCell colSpan={8} className="py-12">
   328→                      <div className="flex flex-col items-center gap-2 text-slate-400">
   329→                        <Package className="h-10 w-10" />
   330→                        <span className="text-sm">ไม่พบอุปกรณ์</span>
   331→                      </div>
   332→                    </TableCell>
   333→                  </TableRow>
   334→                ) : (
   335→                  devices.map((d) => (
   336→                    <TableRow
   337→                      key={d.id}
   338→                      className="cursor-pointer hover:bg-slate-50 dark:hover:bg-slate-800/50"
   339→                      onClick={() => { setDetailAssetNo(d.assetNo); setDetailOpen(true) }}
   340→                    >
   341→                      <TableCell className="font-mono text-xs font-medium text-slate-700 dark:text-slate-300">{d.assetNo}</TableCell>
   342→                      <TableCell className="text-xs text-slate-600 dark:text-slate-400">{d.deviceType || '—'}</TableCell>
   343→                      <TableCell className="text-sm">{d.brand} {d.model}</TableCell>
   344→                      <TableCell>
   345→                        <Badge className={STATUS_BADGE[d.status] || 'bg-slate-100 text-slate-600'}>{d.status}</Badge>
   346→                      </TableCell>
   347→                      <TableCell className="text-xs">{(d.site || '').substring(0, 20)}</TableCell>
   348→                      <TableCell className="text-xs">{(d.department || '').substring(0, 20) || '—'}</TableCell>
   349→                      <TableCell className="text-center">
   350→                        {d.meterRequired ? <Badge className="bg-teal-50 text-teal-700 border-teal-200 dark:bg-teal-950 dark:text-teal-300">✓</Badge> : '—'}
   351→                      </TableCell>
   352→                      <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
   353→                        <div className="flex justify-end gap-1">
   354→                          <Button size="sm" variant="ghost" title="ดู" onClick={() => { setDetailAssetNo(d.assetNo); setDetailOpen(true) }}>
   355→                            <Eye className="h-3.5 w-3.5" />
   356→                          </Button>
   357→                          <Button size="sm" variant="ghost" title="แก้ไข" onClick={() => openEdit(d.assetNo)}>
   358→                            <Pencil className="h-3.5 w-3.5" />
   359→                          </Button>
   360→                          <Button size="sm" variant="ghost" title="ลบ" className="text-rose-500 hover:bg-rose-50 dark:hover:bg-rose-950/30" onClick={() => setDeleteAssetNo(d.assetNo)}>
   361→                            <Trash2 className="h-3.5 w-3.5" />
   362→                          </Button>
   363→                        </div>
   364→                      </TableCell>
   365→                    </TableRow>
   366→                  ))
   367→                )}
   368→              </TableBody>
   369→            </Table>
   370→          </div>
   371→        </CardContent>
   372→      </Card>
   373→
   374→      {/* Pagination */}
   375→      {totalPages > 1 && (
   376→        <div className="flex items-center justify-between">
   377→          <span className="text-xs text-slate-500">หน้า {page} / {totalPages} ({total.toLocaleString()} รายการ)</span>
   378→          <div className="flex gap-2">
   379→            <Button size="sm" variant="outline" disabled={page <= 1} onClick={() => setPage(p => p - 1)}>
   380→              <ChevronLeft className="h-4 w-4" /> ก่อนหน้า
   381→            </Button>
   382→            <Button size="sm" variant="outline" disabled={page >= totalPages} onClick={() => setPage(p => p + 1)}>
   383→              ถัดไป <ChevronRight className="h-4 w-4" />
   384→            </Button>
   385→          </div>
   386→        </div>
   387→      )}
   388→
   389→      {/* Detail Sheet */}
   390→      <ItamDeviceDetailSheet
   391→        assetNo={detailAssetNo}
   392→        open={detailOpen}
   393→        onOpenChange={setDetailOpen}
   394→        onEdit={(an) => {
   395→          setDetailOpen(false)
   396→          openEdit(an)
   397→        }}
   398→      />
   399→
   400→      {/* CRUD Dialog */}
   401→      <Dialog open={crudOpen} onOpenChange={setCrudOpen}>
   402→        <DialogContent className="max-h-[92vh] overflow-y-auto sm:max-w-3xl dark:border-slate-800 dark:bg-slate-900">
   403→          <DialogHeader>
   404→            <DialogTitle className="flex items-center gap-2 text-slate-800 dark:text-slate-100">
   405→              <Package className="h-5 w-5 text-[#f97316]" />
   406→              {editAssetNo ? `แก้ไขอุปกรณ์: ${editAssetNo}` : 'เพิ่มอุปกรณ์ใหม่'}
   407→            </DialogTitle>
   408→          </DialogHeader>
   409→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 md:grid-cols-3">
   410→            <div className="space-y-1.5">
   411→              <Label className="text-xs">รหัสสินทรัพย์ *</Label>
   412→              <Input value={form.assetNo} onChange={(e) => setForm({ ...form, assetNo: e.target.value })} disabled={!!editAssetNo} className="dark:bg-slate-800 dark:border-slate-700" />
   413→            </div>
   414→            <div className="space-y-1.5">
   415→              <Label className="text-xs">ประเภท</Label>
   416→              <Input value={form.deviceType} onChange={(e) => setForm({ ...form, deviceType: e.target.value })} placeholder="เช่น PRINTER" className="dark:bg-slate-800 dark:border-slate-700" />
   417→            </div>
