   137→function getAuthHeaders(): HeadersInit {
   138→  if (typeof window !== 'undefined') {
   139→    const token = window.localStorage.getItem('itam.token')

   141→      return {

   152→function formatBaht(n: number): string {
   153→  const v = typeof n === 'number' && isFinite(n) ? n : 0
   154→  return (
   155→    '฿' +
   156→    v.toLocaleString('th-TH', {
   157→      minimumFractionDigits: 2,
   158→      maximumFractionDigits: 2,
   159→    })
   160→  )
   161→}
   162→
   163→function formatThaiDate(iso: string): string {
   164→  try {
   165→    const d = new Date(iso)
   166→    if (isNaN(d.getTime())) return iso
   167→    return d.toLocaleString('th-TH', {
   168→      year: 'numeric',
   169→      month: 'short',
   170→      day: 'numeric',
   171→      hour: '2-digit',
   172→      minute: '2-digit',
   173→    })
   174→  } catch {
   175→    return iso
   176→  }
   177→}

   489→            </Button>
   490→          </div>
   491→        </div>

   493→    </div>
   494→  )
   495→}

   499→function SnapshotDetailTab({
   500→  snapshotId,
   501→  onBack,
   502→}: {
   503→  snapshotId: string
   504→  onBack: () => void
   505→}) {
   506→  const [verifyResult, setVerifyResult] = React.useState<VerifyResult | null>(
   507→    null,
   508→  )
   509→  const [verifying, setVerifying] = React.useState(false)
   510→  const [verifyError, setVerifyError] = React.useState<string | null>(null)
   511→  const [copied, setCopied] = React.useState(false)
   512→
   513→  const { data, isLoading, isError, error, refetch } = useQuery({
   514→    queryKey: ['v1-snapshot', snapshotId],
   515→    queryFn: async (): Promise<DetailResponse> => {
   516→      const res = await fetch(
   517→        `/api/v1/snapshots/${encodeURIComponent(snapshotId)}?include=rows`,
   518→        { headers: getAuthHeaders() },
   519→      )
   520→      if (!res.ok) {
   521→        const j = (await res.json().catch(() => null)) as
   522→          | { error?: { message?: string } }
   523→          | null
   524→        throw new Error(j?.error?.message || `HTTP ${res.status}`)
   525→      }
   526→      return (await res.json()) as DetailResponse
   527→    },
   528→  })
   529→
   530→  // Reset verify state when snapshot changes (react-query v5 dropped onSuccess)
   531→  React.useEffect(() => {
   532→    setVerifyResult(null)
   533→    setVerifyError(null)
   534→  }, [snapshotId])
   535→
   536→  async function handleVerify() {
   537→    setVerifying(true)
   538→    setVerifyError(null)
   539→    setVerifyResult(null)
   540→    try {
   541→      const res = await fetch(
   542→        `/api/v1/snapshots/${encodeURIComponent(snapshotId)}/verify`,
   543→        {
   544→          method: 'POST',
   545→          headers: getAuthHeaders(),
   546→        },
   547→      )
   548→      const j = (await res.json().catch(() => null)) as
   549→        | { data?: VerifyResult; error?: { message?: string } }
   550→        | null
   551→      if (!res.ok) {
   552→        throw new Error(j?.error?.message || `HTTP ${res.status}`)
   553→      }
   554→      if (j?.data) {
   555→        setVerifyResult(j.data)
   556→        if (j.data.verified) {
   557→          toast.success('ข้อมูลถูกต้อง — hash ตรงกัน')
   558→        } else {
   559→          toast.error('ตรวจพบการแก้ไข — hash ไม่ตรงกัน')
   560→        }
   561→      }
   562→    } catch (e) {
   563→      setVerifyError((e as Error).message)
   564→    } finally {
   565→      setVerifying(false)
   566→    }
   567→  }
   568→
   569→  async function copyHash() {
   570→    const hash = data?.data?.snapshot?.contentHash
   571→    if (!hash) return
   572→    try {
   573→      await navigator.clipboard.writeText(hash)
   574→      setCopied(true)
   575→      toast.success('คัดลอก hash แล้ว')
   576→      setTimeout(() => setCopied(false), 1500)
   577→    } catch {
   578→      toast.error('คัดลอกไม่สำเร็จ')
   579→    }
   580→  }
   581→
   582→  if (isLoading) {
   583→    return (
   584→      <div className="flex flex-1 items-center justify-center py-12">
   585→        <Loader2 className="h-6 w-6 animate-spin text-[#f97316]" />
   586→        <span className="ml-2 text-sm text-slate-500">กำลังโหลด...</span>
   587→      </div>
   588→    )
   589→  }
   590→
   591→  if (isError) {
   592→    return (
   593→      <div className="flex flex-1 flex-col items-center justify-center gap-3 py-12">
   594→        <div className="text-sm text-rose-600 dark:text-rose-400">
   595→          โหลดข้อมูลไม่สำเร็จ: {(error as Error).message}
   596→        </div>
   597→        <Button variant="outline" size="sm" onClick={() => refetch()}>
   598→          <RefreshCw className="h-3.5 w-3.5" /> ลองอีกครั้ง
   599→        </Button>
   600→      </div>
   601→    )
   602→  }
   603→
   604→  const snapshot = data?.data?.snapshot
   605→  const rows = data?.data?.rows ?? []
   606→
   607→  if (!snapshot) return null
   608→
   609→  return (
   610→    <div className="flex min-h-0 flex-1 flex-col gap-3 overflow-hidden">
   611→      {/* Top action bar */}
   612→      <div className="flex flex-shrink-0 items-center justify-between gap-2">
   613→        <Button variant="ghost" size="sm" onClick={onBack}>
   614→          <ArrowLeft className="h-4 w-4" /> กลับ
   615→        </Button>
   616→        <Button variant="outline" size="sm" onClick={() => refetch()}>
   617→          <RefreshCw className="h-3.5 w-3.5" /> รีเฟรช
   618→        </Button>
   619→      </div>
   620→
   621→      {/* Metadata card */}
   622→      <div className="flex-shrink-0 rounded-md border border-slate-200 bg-card p-4 dark:border-slate-700">
   623→        <div className="grid grid-cols-2 gap-3 text-sm md:grid-cols-3">
   624→          <Info
   625→            label="Snapshot ID"
   626→            value={
   627→              <span className="break-all font-mono text-xs">
   628→                {snapshot.snapshotId}
   629→              </span>
   630→            }
   631→          />
   632→          <Info label="รอบเดือน" value={snapshot.cycleMonth} />
   633→          <Info label="Revision" value={`R${snapshot.revision}`} />
   634→          <Info label="สถานะ" value={statusBadge(snapshot.status)} />
   635→          <Info
   636→            label="Rule Version"
   637→            value={snapshot.ruleVersion || '—'}
   638→          />
   639→          <Info
   640→            label="จำนวนแถว"
   641→            value={`${snapshot.rowCount.toLocaleString()} แถว`}
   642→          />
   643→          <Info
   644→            label="หน้าขาวดำ"
   645→            value={`${snapshot.totalPagesBw.toLocaleString()} แผ่น`}
   646→          />
   647→          <Info
   648→            label="หน้าสี"
   649→            value={`${snapshot.totalPagesColor.toLocaleString()} แผ่น`}
   650→          />
   651→          <Info
   652→            label="ค่าใช้จ่ายรวม"
   653→            value={<span className="font-semibold">{formatBaht(snapshot.totalCost)}</span>}
   654→          />
   655→          <Info label="สร้างโดย" value={snapshot.createdBy || '—'} />
   656→          <Info label="สร้างเมื่อ" value={formatThaiDate(snapshot.createdAt)} />
   657→          <div className="col-span-2 md:col-span-1">
   658→            <div className="mb-0.5 text-[11px] text-slate-500">
   659→              Content Hash (SHA-256)
   660→            </div>
   661→            <div className="flex items-start gap-1">
   662→              <code className="break-all rounded bg-slate-100 px-1.5 py-0.5 font-mono text-[10px] dark:bg-slate-800">
   663→                {snapshot.contentHash}
   664→              </code>
   665→              <Button
   666→                size="sm"
   667→                variant="ghost"
   668→                onClick={copyHash}
   669→                className="h-7 w-7 flex-shrink-0 p-0"
   670→                aria-label="คัดลอก hash"
   671→              >
   672→                {copied ? (
   673→                  <Check className="h-3.5 w-3.5 text-emerald-600" />
   674→                ) : (
   675→                  <Copy className="h-3.5 w-3.5" />
   676→                )}
   677→              </Button>
   678→            </div>
   679→          </div>
   680→        </div>
   681→
   682→        {/* Verify section */}
   683→        <div className="mt-4 border-t border-slate-200 pt-4 dark:border-slate-700">
   684→          <div className="flex flex-wrap items-center justify-between gap-2">
   685→            <div className="flex items-center gap-2">
   686→              <Shield className="h-4 w-4 text-[#f97316]" />
   687→              <span className="text-sm font-medium">การตรวจสอบความถูกต้อง</span>
   688→              <span className="text-xs text-slate-500">
   689→                (เปรียบเทียบ SHA-256 hash กับข้อมูลปัจจุบัน)
   690→              </span>
   691→            </div>
   692→            <Button size="sm" onClick={handleVerify} disabled={verifying}>
   693→              {verifying ? (
   694→                <Loader2 className="h-3.5 w-3.5 animate-spin" />
   695→              ) : (
   696→                <ShieldCheck className="h-3.5 w-3.5" />
   697→              )}
   698→              ตรวจสอบความถูกต้อง
   699→            </Button>
   700→          </div>
   701→
   702→          {verifyError ? (
   703→            <div className="mt-2 text-sm text-rose-600 dark:text-rose-400">
   704→              ตรวจสอบไม่สำเร็จ: {verifyError}
   705→            </div>
   706→          ) : null}
   707→
   708→          {verifyResult ? (
   709→            <div className="mt-2">
   710→              {verifyResult.verified ? (
   711→                <div className="flex items-start gap-2 rounded-md border border-emerald-200 bg-emerald-50 p-3 text-sm text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950 dark:text-emerald-300">
   712→                  <ShieldCheck className="mt-0.5 h-5 w-5 flex-shrink-0" />
   713→                  <div>
   714→                    <div className="font-medium">✓ ข้อมูลถูกต้อง</div>
   715→                    <div className="mt-0.5 text-xs">
   716→                      hash ตรงกัน — ไม่พบการแก้ไขหลังสร้าง snapshot
   717→                    </div>
   718→                  </div>
   719→                </div>
   720→              ) : (
   721→                <div className="flex items-start gap-2 rounded-md border border-rose-200 bg-rose-50 p-3 text-sm text-rose-700 dark:border-rose-800 dark:bg-rose-950 dark:text-rose-300">
   722→                  <ShieldAlert className="mt-0.5 h-5 w-5 flex-shrink-0" />
   723→                  <div className="flex-1">
   724→                    <div className="font-medium">✗ ตรวจพบการแก้ไข</div>
   725→                    <div className="mt-1 space-y-0.5 text-xs">
   726→                      <div>
   727→                        Stored hash:{' '}
   728→                        <code className="font-mono">
   729→                          {verifyResult.storedHash}
   730→                        </code>
   731→                      </div>
   732→                      <div>
   733→                        Computed hash:{' '}
   734→                        <code className="font-mono">
   735→                          {verifyResult.computedHash}
   736→                        </code>
   737→                      </div>
   738→                    </div>
   739→                  </div>
   740→                </div>
   741→              )}
   742→            </div>
   743→          ) : null}
   744→        </div>
   745→      </div>
   746→
   747→      {/* Frozen rows */}
   748→      <div className="flex min-h-0 flex-1 flex-col gap-2 overflow-hidden">
   749→        <div className="flex-shrink-0 text-sm font-medium">
   750→          รายการแถวที่ถูกแช่แข็ง ({rows.length.toLocaleString()} แถว)
   751→        </div>
   752→        <div className="min-h-0 flex-1 overflow-auto rounded-md border border-slate-200 dark:border-slate-700">
   753→          <Table>
   754→            <TableHeader className="sticky top-0 z-10 bg-slate-50 dark:bg-slate-900">
   755→              <TableRow>
   756→                <TableHead className="whitespace-nowrap">เลขครุภัณฑ์</TableHead>
   757→                <TableHead className="whitespace-nowrap">ยี่ห้อ / รุ่น</TableHead>
   758→                <TableHead className="whitespace-nowrap">วันที่จด</TableHead>
   759→                <TableHead className="text-center whitespace-nowrap">ประเภท</TableHead>
   760→                <TableHead className="text-right whitespace-nowrap">มิเตอร์ BW</TableHead>
   761→                <TableHead className="text-right whitespace-nowrap">มิเตอร์สี</TableHead>
   762→                <TableHead className="text-right whitespace-nowrap">หน้า BW</TableHead>
   763→                <TableHead className="text-right whitespace-nowrap">หน้าสี</TableHead>
   764→                <TableHead className="text-right whitespace-nowrap">ค่า BW</TableHead>

   766→                <TableHead className="whitespace-nowrap">สาขา</TableHead>

   768→            </TableHeader>
   769→            <TableBody>
   770→              {rows.length === 0 ? (
   771→                <TableRow>
   772→                  <TableCell
   773→                    colSpan={11}
   774→                    className="py-8 text-center text-sm text-slate-500"
   775→                  >
   776→                    ไม่มีแถวข้อมูลใน snapshot นี้
   777→                  </TableCell>
   778→                </TableRow>
   779→              ) : (
   780→                rows.map((r) => (
   781→                  <TableRow key={r.id}>
   782→                    <TableCell className="whitespace-nowrap font-mono text-xs">
   783→                      {r.assetNo}
   784→                    </TableCell>
   785→                    <TableCell className="whitespace-nowrap text-xs">
   786→                      {[r.brand, r.model].filter(Boolean).join(' ') || '—'}
   787→                    </TableCell>
   788→                    <TableCell className="whitespace-nowrap text-xs">
   789→                      {r.readingDate || '—'}
   790→                    </TableCell>
   791→                    <TableCell className="text-center">
   792→                      {readingTypeBadge(r.readingType)}
   793→                    </TableCell>
   794→                    <TableCell className="text-right tabular-nums">
   795→                      {r.meterBw.toLocaleString()}
   796→                    </TableCell>

   798→                      {r.meterColor.toLocaleString()}

   800→                    <TableCell className="text-right tabular-nums">
   801→                      {r.pagesBw.toLocaleString()}
   802→                    </TableCell>
   803→                    <TableCell className="text-right tabular-nums">

   807→                      {formatBaht(r.costBw)}
   808→                    </TableCell>
   809→                    <TableCell className="text-right tabular-nums whitespace-nowrap">
   810→                      {formatBaht(r.costColor)}
   811→                    </TableCell>
   812→                    <TableCell className="whitespace-nowrap text-xs">
   813→                      {r.siteAtReading || '—'}
   814→                    </TableCell>
   815→                  </TableRow>
   816→                ))
   817→              )}
   818→            </TableBody>
   819→          </Table>
   820→        </div>
   821→      </div>
   822→    </div>
   823→  )
   824→}
