   315→function formatDateTime(iso: string | null): string {
   316→  if (!iso) return '—'
   317→  try {
   318→    return new Date(iso).toLocaleString('th-TH', {
   319→      day: '2-digit',
   320→      month: '2-digit',
   321→      year: 'numeric',
   322→      hour: '2-digit',
   323→      minute: '2-digit',
   324→    })
   325→  } catch {
   326→    return iso
   327→  }
   328→}

   330→function parseExternalMeta(raw: string | null): ExternalMeta | null {
   331→  if (!raw) return null
   332→  try {
   333→    const obj = JSON.parse(raw)
   334→    if (!obj || typeof obj !== 'object') return null
   335→    return obj as ExternalMeta
   336→  } catch {
   337→    return null
   338→  }
   339→}

   391→// 1 MB hard cap to keep SQLite payload sane
   392→const MAX_PIC_BYTES = 1_000_000
   393→
   394→function readFileAsDataUrl(file: File): Promise<string> {
   395→  return new Promise((resolve, reject) => {
   396→    const reader = new FileReader()
   397→    reader.onload = () => resolve(String(reader.result))
   398→    reader.onerror = reject
   399→    reader.readAsDataURL(file)
   400→  })
   401→}
   402→
   403→async function compressImage(file: File, maxBytes = MAX_PIC_BYTES): Promise<string> {
   404→  const dataUrl = await readFileAsDataUrl(file)
   405→  if (dataUrl.length <= maxBytes) return dataUrl
   406→  // Try shrinking via canvas
   407→  return new Promise<string>((resolve) => {
   408→    const img = new window.Image()
   409→    img.onload = () => {
   410→      const maxDim = 1280
   411→      let { width, height } = img
   412→      if (width > maxDim || height > maxDim) {
   413→        const ratio = Math.min(maxDim / width, maxDim / height)
   414→        width = Math.round(width * ratio)
   415→        height = Math.round(height * ratio)
   416→      }
   417→      const canvas = document.createElement('canvas')
   418→      canvas.width = width
   419→      canvas.height = height
   420→      const ctx = canvas.getContext('2d')
   421→      if (!ctx) {
   422→        resolve(dataUrl)
   423→        return
   424→      }
   425→      ctx.drawImage(img, 0, 0, width, height)
   426→      let quality = 0.8
   427→      let out = canvas.toDataURL('image/jpeg', quality)
   428→      while (out.length > maxBytes && quality > 0.3) {
   429→        quality -= 0.15
   430→        out = canvas.toDataURL('image/jpeg', quality)
   431→      }
   432→      resolve(out)
   433→    }
   434→    img.onerror = () => resolve(dataUrl)
   435→    img.src = dataUrl
   436→  })
   437→}

   446→  const [form, setForm] = React.useState<NewFormState>(EMPTY_FORM)

   448→  const [detailId, setDetailId] = React.useState<string | null>(null)
   449→

   452→    queryKey: ['wo-options'],
   453→    queryFn: async () => {
   454→      const res = await fetch('/api/settings/options')

   465→      setDebouncedSearch(search)

   470→

   472→    setPage(1)
   473→  }, [statusFilter, priorityFilter])
   474→

   476→    queryKey: [
   477→      'work-orders',

   489→      params.set('pageSize', String(PAGE_SIZE))
   490→      const res = await fetch(`/api/work-orders?${params.toString()}`)

   514→    }

   517→        toast.error('กรุณาระบุชื่อลูกค้าสำหรับงานนอก')

   525→      }
   526→    }
   527→    try {

   575→    }

   583→          <h1 className="flex items-center gap-2 text-xl font-bold md:text-2xl">
   584→            <Wrench className="h-6 w-6 text-orange-500" />
   585→            แจ้งซ่อม
   586→          </h1>
   587→          <p className="text-sm text-muted-foreground">
   588→            ระบบแจ้งซ่อมครบวงจร — แจ้ง → รับงาน → ซ่อม → ปิดงาน (รองรับลูกค้าภายนอก)
   589→          </p>
   590→        </div>
   591→        <div className="flex items-center gap-2">
   592→          <Button
   593→            variant="outline"
   594→            size="sm"
   595→            onClick={() => listQuery.refetch()}
   596→            disabled={listQuery.isFetching}
   597→            aria-label="รีเฟรช"
   598→          >
   599→            <RefreshCw
   600→              className={listQuery.isFetching ? 'h-4 w-4 animate-spin' : 'h-4 w-4'}
   601→            />
   602→            <span className="hidden sm:inline">รีเฟรช</span>
   603→          </Button>
   604→          <Button size="sm" onClick={openCreate} className="bg-orange-500 hover:bg-orange-600">
   605→            <Plus className="h-4 w-4" />
   606→            แจ้งซ่อมใหม่
   607→          </Button>
   608→        </div>
   609→      </div>
   610→
   611→      {/* KPI stats bar */}
   612→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   613→        <KpiCard
   614→          label="รอดำเนินการ"
   615→          value={stats.PENDING ?? 0}
   616→          icon={<Clock className="h-5 w-5" />}
   617→          color="amber"
   618→        />
