    51→// ============ Cascading dropdown hook ============
    52→// Fetches distinct values for a location field, filtered by previous selections.
    53→function useCascadingOptions(
    54→  field: 'building' | 'floor' | 'department' | 'location',
    55→  parents: { site?: string; building?: string; floor?: string; department?: string },
    56→): string[] {
    57→  const [values, setValues] = React.useState<string[]>([])
    58→  const { site, building, floor, department } = parents
    59→  React.useEffect(() => {
    60→    let cancelled = false
    61→    const params = new URLSearchParams({ field })
    62→    if (site) params.set('site', site)
    63→    if (building) params.set('building', building)
    64→    if (floor) params.set('floor', floor)
    65→    if (department) params.set('department', department)
    66→    fetch(`/api/itam/devices/cascading?${params}`)
    67→      .then((r) => r.ok ? r.json() : Promise.reject(r))
    68→      .then((j: { values: string[] }) => { if (!cancelled) setValues(j.values ?? []) })
    69→      .catch(() => { if (!cancelled) setValues([]) })
    70→    return () => { cancelled = true }
    71→  }, [field, site, building, floor, department])
    72→  return values
    73→}

   134→// ============ Search debounce hook (200ms) ============
   135→function useDebounced<T>(value: T, delayMs = 200): [T, boolean] {
   136→  const [debounced, setDebounced] = React.useState<T>(value)
   137→  const [isPending, setIsPending] = React.useState(false)
   138→  React.useEffect(() => {
   139→    setIsPending(true)
   140→    const t = setTimeout(() => {
   141→      setDebounced(value)
   142→      setIsPending(false)
   143→    }, delayMs)
   144→    return () => clearTimeout(t)
   145→  }, [value, delayMs])
   146→  return [debounced, isPending]
   147→}

   149→// ============ Highlight matched text ============
   150→function Highlight({ text, query }: { text: string; query: string }) {
   151→  if (!query.trim()) return <>{text || '—'}</>
   152→  const q = query.trim()
   153→  // Escape regex special chars
   154→  const esc = q.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
   155→  const parts = (text || '').split(new RegExp(`(${esc})`, 'ig'))
   156→  return (
   157→    <>
   158→      {parts.map((p, i) =>
   159→        p.toLowerCase() === q.toLowerCase()
   160→          ? <mark key={i} className="rounded bg-orange-200 px-0.5 text-slate-900 dark:bg-orange-500/60 dark:text-white">{p}</mark>
   161→          : <React.Fragment key={i}>{p}</React.Fragment>,
   162→      )}
   163→    </>
   164→  )
   165→}

   167→// ============ Recent searches (localStorage) ============
   168→const RECENT_KEY = 'itam-recent-searches'
   169→function loadRecent(): string[] {
   170→  if (typeof window === 'undefined') return []
   171→  try {
   172→    const raw = window.localStorage.getItem(RECENT_KEY)
   173→    if (!raw) return []
   174→    const arr = JSON.parse(raw)
   175→    return Array.isArray(arr) ? arr.filter((x) => typeof x === 'string').slice(0, 6) : []
   176→  } catch {
   177→    return []
   178→  }
   179→}
   180→function saveRecent(q: string) {
   181→  if (typeof window === 'undefined' || !q.trim()) return
   182→  try {
   183→    const cur = loadRecent()
   184→    const next = [q.trim(), ...cur.filter((x) => x !== q.trim())].slice(0, 6)
   185→    window.localStorage.setItem(RECENT_KEY, JSON.stringify(next))
   186→  } catch { /* ignore */ }
   187→}
