   306→        method: 'POST',

   546→/* ---------- Sites tab ---------- */
   547→function SitesTab() {
   548→  const qc = useQueryClient()
   549→  const { data: sites, isLoading } = useQuery<Site[]>({
   550→    queryKey: ['sites'],
   551→    queryFn: async () => {
   552→      const res = await fetch('/api/sites')
   553→      if (!res.ok) throw new Error('Failed to load sites')
   554→      const json = await res.json()
   555→      return json.sites as Site[]
   556→    },
   557→  })
   558→
   559→  const [code, setCode] = React.useState('')
   560→  const [name, setName] = React.useState('')
   561→  const [saving, setSaving] = React.useState(false)
   562→
   563→  async function addSite() {
   564→    if (!code || !name) {
   565→      toast.error('กรุณากรอกรหัสและชื่อสาขา')
   566→      return
   567→    }
   568→    try {
   569→      setSaving(true)
   570→      const res = await fetch('/api/sites', {
   571→        method: 'POST',
   572→        headers: { 'Content-Type': 'application/json' },
   573→        body: JSON.stringify({ code, name }),
   574→      })
   575→      if (!res.ok) {
   576→        const j = await res.json().catch(() => ({}))
