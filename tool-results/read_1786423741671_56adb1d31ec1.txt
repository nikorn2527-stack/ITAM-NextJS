    72→function InfoRow({ label, value }: { label: string; value: React.ReactNode }) {
    73→  return (

    87→function formatThaiDateTime(iso: string): string {
    88→  try {
    89→    return new Date(iso).toLocaleString('th-TH', {
    90→      day: '2-digit',
    91→      month: '2-digit',
    92→      year: 'numeric',
    93→      hour: '2-digit',
    94→      minute: '2-digit',
    95→    })
    96→  } catch {
    97→    return iso
    98→  }
    99→}
   100→
   101→function todayISO(): string {
   102→  return new Date().toISOString().slice(0, 10)
   103→}
