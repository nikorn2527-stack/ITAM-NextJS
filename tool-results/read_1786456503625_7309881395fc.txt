    86→    86→export function ItamDevices() {
    87→    87→  const qc = useQueryClient()
    88→    88→  const [search, setSearch] = React.useState('')
    89→    89→  const [status, setStatus] = React.useState('all')
    90→    90→  const [page, setPage] = React.useState(1)
    91→    91→  const [limit] = React.useState(20)
    92→    92→
    93→    93→  // Detail sheet
    94→    94→  const [detailAssetNo, setDetailAssetNo] = React.useState<string | null>(null)
    95→    95→  const [detailOpen, setDetailOpen] = React.useState(false)
    96→    96→
    97→    97→  // CRUD dialog
    98→    98→  const [crudOpen, setCrudOpen] = React.useState(false)
    99→    99→  const [editAssetNo, setEditAssetNo] = React.useState<string | null>(null)
   100→   100→  const [form, setForm] = React.useState<DeviceForm>(EMPTY_FORM)
   101→   101→  const [saving, setSaving] = React.useState(false)
   102→   102→
   103→   103→  // Delete confirm
   104→   104→  const [deleteAssetNo, setDeleteAssetNo] = React.useState<string | null>(null)
   105→   105→  const [deleting, setDeleting] = React.useState(false)
   106→   106→
   107→   107→  // CSV export
   108→   108→  const [exporting, setExporting] = React.useState(false)
   109→   109→
   110→   110→  const queryKey = React.useMemo(() => ['itam-devices', search, status, page, limit], [search, status, page, limit])
   111→   111→
   112→   112→  const { data, isLoading } = useQuery<DevicesResponse>({
   113→   113→    queryKey,
   114→   114→    queryFn: async () => {
   115→   115→      const params = new URLSearchParams({ page: String(page), limit: String(limit) })
   116→   116→      if (search) params.set('search', search)
   117→   117→      if (status !== 'all') params.set('status', status)
   118→   118→      const res = await fetch(`/api/itam/devices?${params}`)
   119→   119→      if (!res.ok) throw new Error('Failed')
   120→   120→      return res.json()
   121→   121→    },
   122→   122→  })
   123→   123→
   124→   124→  function onSearch(val: string) {
   125→   125→    setSearch(val)
   126→   126→    setPage(1)
   127→   127→  }
   128→   128→
   129→   129→  function openAdd() {
   130→   130→    setEditAssetNo(null)
   131→   131→    setForm(EMPTY_FORM)
   132→   132→    setCrudOpen(true)
   133→   133→  }
   134→   134→
   135→   135→  function openEdit(assetNo: string) {
   136→   136→    setEditAssetNo(assetNo)
   137→   137→    // Fetch current device fields then open dialog
   138→   138→    fetch(`/api/itam/devices/${assetNo}`)
   139→   139→      .then(r => r.ok ? r.json() : Promise.reject(r))
   140→   140→      .then((j: { device: Device & Record<string, unknown> }) => {
   141→   141→        const d = j.device
   142→   142→        setForm({
   143→   143→          assetNo: d.assetNo, deviceType: d.deviceType || '', brand: d.brand || '', model: d.model || '',
   144→   144→          serial: d.serial || '', building: d.building || '', floor: d.floor || '',
   145→   145→          department: d.department || '', location: d.location || '', departmentCode: d.departmentCode || '',
   146→   146→          status: d.status, site: d.site || '', contractNo: (d as { contractNo?: string }).contractNo || '',
   147→   147→          ip: (d as { ip?: string }).ip || '', mac: (d as { mac?: string }).mac || '',
   148→   148→          remoteId: (d as { remoteId?: string }).remoteId || '',
   149→   149→          vendor: (d as { vendor?: string }).vendor || '',
   150→   150→          installDate: (d as { installDate?: string }).installDate || '',
   151→   151→          warrantyEnd: (d as { warrantyEnd?: string }).warrantyEnd || '',
   152→   152→          deviceGroup: (d as { deviceGroup?: string }).deviceGroup || '',
   153→   153→          costCenter: (d as { costCenter?: string }).costCenter || '',
   154→   154→          meterRequired: d.meterRequired, meterMode: (d as { meterMode?: string }).meterMode || 'TOTAL',
   155→   155→          assetSiteCode: (d as { assetSiteCode?: string }).assetSiteCode || '',
   156→   156→          remark: d.remark || '',
   157→   157→        })
   158→   158→        setCrudOpen(true)
   159→   159→      })
   160→   160→      .catch(() => toast.error('โหลดข้อมูลไม่สำเร็จ'))
   161→   161→  }
   162→   162→
   163→   163→  async function saveDevice() {
   164→   164→    if (!form.assetNo.trim()) {
   165→   165→      toast.error('กรุณากรอกรหัสสินทรัพย์')
   166→   166→      return
   167→   167→    }
   168→   168→    try {
   169→   169→      setSaving(true)
   170→   170→      const payload = {
   171→   171→        ...form,
   172→   172→        meterRequired: form.meterRequired,
   173→   173→      }
   174→   174→      if (editAssetNo) {
   175→   175→        const res = await fetch(`/api/itam/devices/${editAssetNo}`, {
   176→   176→          method: 'PUT', headers: { 'Content-Type': 'application/json' },
   177→   177→          body: JSON.stringify(payload),
   178→   178→        })
   179→   179→        if (!res.ok) {
   180→   180→          const j = await res.json().catch(() => ({}))
   181→   181→          throw new Error(j.error || 'Failed')
   182→   182→        }
   183→   183→        toast.success('แก้ไขอุปกรณ์แล้ว')
   184→   184→      } else {
   185→   185→        const res = await fetch('/api/itam/devices', {
   186→   186→          method: 'POST', headers: { 'Content-Type': 'application/json' },
   187→   187→          body: JSON.stringify(payload),
   188→   188→        })
   189→   189→        if (!res.ok) {
   190→   190→          const j = await res.json().catch(() => ({}))
   191→   191→          throw new Error(j.error || 'Failed')
   192→   192→        }
   193→   193→        toast.success('เพิ่มอุปกรณ์แล้ว')
   194→   194→      }
   195→   195→      setCrudOpen(false)
   196→   196→      await qc.invalidateQueries({ queryKey: ['itam-devices'] })
   197→   197→      await qc.invalidateQueries({ queryKey: ['itam-dashboard'] })
   198→   198→    } catch (e) {
   199→   199→      toast.error(e instanceof Error ? e.message : 'บันทึกไม่สำเร็จ')
   200→   200→    } finally {
   201→   201→      setSaving(false)
   202→   202→    }
   203→   203→  }
   204→   204→
   205→   205→  async function confirmDelete() {
   206→   206→    if (!deleteAssetNo) return
   207→   207→    try {
   208→   208→      setDeleting(true)
   209→   209→      const res = await fetch(`/api/itam/devices/${deleteAssetNo}`, { method: 'DELETE' })
   210→   210→      if (!res.ok) {
   211→   211→        const j = await res.json().catch(() => ({}))
   212→   212→        throw new Error(j.error || 'Failed')
   213→   213→      }
   214→   214→      toast.success(`ลบ ${deleteAssetNo} แล้ว`)
   215→   215→      setDeleteAssetNo(null)
   216→   216→      await qc.invalidateQueries({ queryKey: ['itam-devices'] })
   217→   217→      await qc.invalidateQueries({ queryKey: ['itam-dashboard'] })
   218→   218→    } catch (e) {
   219→   219→      toast.error(e instanceof Error ? e.message : 'ลบไม่สำเร็จ')
   220→   220→    } finally {
   221→   221→      setDeleting(false)
   222→   222→    }
   223→   223→  }
   224→   224→
   225→   225→  async function exportCsv() {

   233→   233→        assetNo: d.assetNo, deviceType: d.deviceType || '', brand: d.brand || '',

   235→   235→        site: d.site || '', building: d.building || '', floor: d.floor || '',
   236→   236→        department: d.department || '', departmentCode: '', location: d.location || '',
   237→   237→        deviceGroup: '', costCenter: '', contractNo: '', vendor: '',
   238→   238→        ip: '', mac: '', remoteId: '', installDate: '', warrantyEnd: '',

   248→   248→    }
   249→   249→  }

   251→   251→  const devices = data?.devices ?? []

   264→   264→            <Download className="h-4 w-4" /> ส่งออก CSV

   266→   266→          <Button variant="outline" onClick={() => qc.invalidateQueries({ queryKey })} className="dark:bg-slate-800 dark:border-slate-700">
   267→   267→            <RefreshCw className="h-4 w-4" /> รีเฟรช
   268→   268→          </Button>
   269→   269→          <Button className="bg-[#f97316] text-white hover:bg-[#ea580c]" onClick={openAdd}>
   270→   270→            <Plus className="h-4 w-4" /> เพิ่มอุปกรณ์
   271→   271→          </Button>
   272→   272→        </div>
   273→   273→      </div>
   274→   274→
   275→   275→      {/* Toolbar */}
   276→   276→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center">

   278→   278→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
   279→   279→          <Input
   280→   280→            placeholder="ค้นหารหัส, แบรนด์, รุ่น, SN..."

   289→   289→          </SelectTrigger>
   290→   290→          <SelectContent>
   291→   291→            <SelectItem value="all">สถานะทั้งหมด</SelectItem>
   292→   292→            <SelectItem value="Active">ใช้งานอยู่</SelectItem>

   304→   304→          <div className="itam-scroll max-h-[60vh] overflow-auto">
   305→   305→            <Table>
   306→   306→              <TableHeader className="sticky top-0 z-10 bg-slate-50/80 backdrop-blur-sm dark:bg-slate-900/80">
   307→   307→                <TableRow>
   308→   308→                  <TableHead className="w-16">รหัส</TableHead>

   313→   313→                  <TableHead>แผนก</TableHead>
   314→   314→                  <TableHead className="text-center">มิเตอร์</TableHead>
   315→   315→                  <TableHead className="text-right w-32">จัดการ</TableHead>

   317→   317→              </TableHeader>
   318→   318→              <TableBody>

   340→   340→                    >
   341→   341→                      <TableCell className="font-mono text-xs font-medium text-slate-700 dark:text-slate-300">{d.assetNo}</TableCell>
   342→   342→                      <TableCell className="text-xs text-slate-600 dark:text-slate-400">{d.deviceType || '—'}</TableCell>

   359→   359→                          </Button>
