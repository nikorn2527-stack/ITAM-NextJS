   164→function formatBaht(value: number | null | undefined): string {
   165→  if (value === null || value === undefined || Number.isNaN(value)) return '—'
   166→  return `฿${value.toLocaleString('th-TH', {
   167→    minimumFractionDigits: 2,
   168→    maximumFractionDigits: 2,
   169→  })}`
   170→}
   171→
   172→function formatThaiDate(iso: string | null | undefined): string {
   173→  if (!iso) return '—'
   174→  try {
   175→    return new Date(iso.length > 10 ? iso : iso + 'T00:00:00').toLocaleDateString(
   176→      'th-TH',
   177→      { day: '2-digit', month: '2-digit', year: 'numeric' },
   178→    )
   179→  } catch {
   180→    return iso
   181→  }
   182→}

   472→        vendor: txnForm.vendor || undefined,
   473→        performedBy: txnForm.performedBy || undefined,

   478→        method: 'POST',
   479→        headers: { 'Content-Type': 'application/json' },
   480→        body: JSON.stringify(payload),
   481→      })
   482→      if (!res.ok) {
   483→        const j = await res.json().catch(() => ({}))
   484→        throw new Error(j.error ?? 'Transaction failed')
   485→      }
   486→      const json = await res.json()
   487→      const verb =
   488→        txnForm.type === 'IN'
   489→          ? 'รับเข้า'
   490→          : txnForm.type === 'OUT'
   491→          ? 'เบิกออก'
   492→          : 'ปรับปรุง'
   493→      toast.success(`${verb}สำเร็จ — คงเหลือ ${json.data.item.quantity} ${json.data.item.unit}`)

   495→      await qc.invalidateQueries({ queryKey: ['stock-items'] })
   496→      if (detailId === txnTarget.id) {
   497→        await qc.invalidateQueries({ queryKey: ['stock-item-detail', detailId] })
   498→      }
   499→    } catch (e) {
   500→      toast.error(e instanceof Error ? e.message : 'Transaction failed')

   502→      setSavingTxn(false)
   503→    }
   504→  }
