    56→interface ReminderDevice {
    57→  id: string
    58→  assetCode: string
    59→  name: string
    60→  brand: string
    61→  model: string
    62→  site: string
    63→  lastMeterReading: number
    64→}
    65→interface ReminderEntry {
    66→  device: ReminderDevice
    67→  lastReadingDate: string | null
    68→  daysOverdue: number
    69→}
    70→interface RemindersData {
    71→  hasActiveCycle: boolean
    72→  cycle: { id: string; name: string; startDate: string; endDate: string; status: string } | null
    73→  reminders: ReminderEntry[]
    74→  totalRead: number
    75→  totalUnread: number
    76→}
    77→
    78→function todayISO(): string {
    79→  return new Date().toISOString().slice(0, 10)
    80→}
    81→
    82→function daysBetween(a: string, b: string): number {
    83→  const ms = new Date(b).getTime() - new Date(a).getTime()
    84→  return Math.round(ms / (1000 * 60 * 60 * 24))
    85→}
    86→
    87→export function MeterPage() {
    88→  const qc = useQueryClient()
    89→  const tableRef = React.useRef<HTMLDivElement>(null)
    90→  const [highlightUnread, setHighlightUnread] = React.useState(false)
    91→  const [readingTarget, setReadingTarget] = React.useState<Device | null>(null)
    92→  const [newReading, setNewReading] = React.useState('')
    93→  const [readingDate, setReadingDate] = React.useState(todayISO())
    94→  const [remark, setRemark] = React.useState('')
    95→  const [saving, setSaving] = React.useState(false)
    96→  const [cycleDialogOpen, setCycleDialogOpen] = React.useState(false)
    97→  const [cycleName, setCycleName] = React.useState('')
    98→  const [cycleStart, setCycleStart] = React.useState(todayISO())
    99→  const [cycleEnd, setCycleEnd] = React.useState(todayISO())
   100→  const [creatingCycle, setCreatingCycle] = React.useState(false)
   101→  const [exporting, setExporting] = React.useState(false)
   102→
   103→  const { data: activeCycle } = useQuery<Cycle | null>({
   104→    queryKey: ['active-cycle'],
   105→    queryFn: async () => {
   106→      const res = await fetch('/api/cycles?status=active')
   107→      if (!res.ok) return null
   108→      const json = await res.json()
   109→      return (json.cycles?.[0] as Cycle | undefined) ?? null
   110→    },
   111→  })
   112→
   113→  const { data: devices, isLoading } = useQuery<Device[]>({
   114→    queryKey: ['devices-meter'],
   115→    queryFn: async () => {
   116→      const res = await fetch('/api/devices')
   117→      if (!res.ok) throw new Error('Failed to load devices')
   118→      const json = await res.json()
   119→      return (json.devices as Device[]).filter(
   120→        (d) => d.type === 'COPIER' || d.type === 'MFP' || d.type === 'PRINTER',
   121→      )
   122→    },
   123→  })
   124→
   125→  const { data: remindersData, isLoading: remindersLoading } = useQuery<RemindersData>({
   126→    queryKey: ['meter-reminders'],
   127→    queryFn: async () => {
   128→      const res = await fetch('/api/meter/reminders')
   129→      if (!res.ok) throw new Error('Failed to load reminders')
   130→      return res.json()
   131→    },
   132→  })
   133→
   134→  const meterableDevices = devices ?? []
   135→  // Map unread device IDs for the table badge column
   136→  const unreadDeviceIds = React.useMemo(
   137→    () => new Set((remindersData?.reminders ?? []).map((r) => r.device.id)),
   138→    [remindersData],
   139→  )
   140→
   141→  function openReadingDialog(d: Device) {
   142→    setReadingTarget(d)
   143→    setNewReading(String(d.lastMeterReading ?? 0))
   144→    setReadingDate(todayISO())
   145→    setRemark('')
   146→  }
   147→
   148→  function scrollToTable() {
   149→    setHighlightUnread(true)
   150→    tableRef.current?.scrollIntoView({ behavior: 'smooth', block: 'start' })
   151→    // Clear highlight after a moment
   152→    setTimeout(() => setHighlightUnread(false), 3000)
   153→  }
   154→
   155→  const prevReading = readingTarget?.lastMeterReading ?? 0
   156→  const newReadingNum = Number(newReading)
   157→  const delta = Number.isFinite(newReadingNum) ? newReadingNum - prevReading : 0
   158→  const isReset = newReadingNum < prevReading
   159→  const isExceed = delta > 20000
   160→  const needsRemark = isReset && !remark.trim()
   161→
   162→  async function saveReading() {
   163→    if (!readingTarget) return
   164→    if (!Number.isFinite(newReadingNum)) {
   165→      toast.error('กรุณากรอกค่ามิเตอร์เป็นตัวเลข')
   166→      return
   167→    }
   168→    if (needsRemark) {
   169→      toast.error('ค่าใหม่น้อยกว่าค่าก่อนหน้า กรุณาระบุหมายเหตุ (RESET)')
   170→      return
   171→    }
   172→    try {
   173→      setSaving(true)
   174→      const res = await fetch('/api/meter', {
   175→        method: 'POST',
   176→        headers: { 'Content-Type': 'application/json' },
   177→        body: JSON.stringify({
   178→          deviceId: readingTarget.id,
   179→          reading: newReadingNum,
   180→          date: readingDate,
   181→          remark: remark.trim() || null,
   182→          cycleId: activeCycle?.id ?? null,
   183→        }),
   184→      })
   185→      if (!res.ok) {
   186→        const j = await res.json().catch(() => ({}))
   187→        throw new Error(j.error ?? 'Save failed')
   188→      }
   189→      const json = await res.json()
   190→      if (json.warning) {
   191→        toast.warning(json.warning)
   192→      } else {
   193→        toast.success('บันทึกการจดมิเตอร์แล้ว')
   194→      }
   195→      setReadingTarget(null)
   196→      await qc.invalidateQueries({ queryKey: ['devices-meter'] })
   197→      await qc.invalidateQueries({ queryKey: ['dashboard'] })
   198→      await qc.invalidateQueries({ queryKey: ['active-cycle'] })
   199→      await qc.invalidateQueries({ queryKey: ['meter-reminders'] })
   200→    } catch (e) {
   201→      toast.error(e instanceof Error ? e.message : 'Save failed')
   202→    } finally {
   203→      setSaving(false)
   204→    }
   205→  }
   206→
   207→  async function createCycle() {
   208→    if (!cycleName || !cycleStart || !cycleEnd) {
   209→      toast.error('กรุณากรอกชื่อรอบและวันที่ให้ครบ')
   210→      return
   211→    }
   212→    try {
   213→      setCreatingCycle(true)
   214→      const res = await fetch('/api/cycles', {
   215→        method: 'POST',
   216→        headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({
   218→          name: cycleName,
   219→          startDate: cycleStart,
   220→          endDate: cycleEnd,
   221→          status: 'active',
   222→        }),
   223→      })
   224→      if (!res.ok) {
   225→        const j = await res.json().catch(() => ({}))
   226→        throw new Error(j.error ?? 'Create cycle failed')
   227→      }
   228→      toast.success('สร้างรอบจดมิเตอร์ใหม่แล้ว')
   229→      setCycleDialogOpen(false)
   230→      setCycleName('')
   231→      await qc.invalidateQueries({ queryKey: ['active-cycle'] })
   232→      await qc.invalidateQueries({ queryKey: ['meter-reminders'] })
   233→    } catch (e) {
   234→      toast.error(e instanceof Error ? e.message : 'Create cycle failed')
   235→    } finally {
   236→      setCreatingCycle(false)
   237→    }
   238→  }
   239→
   240→  const remainingDays = activeCycle
   241→    ? daysBetween(todayISO(), activeCycle.endDate)
   242→    : null
   243→
   244→  // Reminder progress
   245→  const totalMeterable = (remindersData?.totalRead ?? 0) + (remindersData?.totalUnread ?? 0)
   246→  const readCount = remindersData?.totalRead ?? 0
   247→  const readPct = totalMeterable > 0 ? Math.round((readCount / totalMeterable) * 100) : 0
   248→
   249→  async function exportCsv() {
   250→    try {
   251→      setExporting(true)
   252→      const res = await fetch('/api/meter')
   253→      if (!res.ok) throw new Error('Failed to export')
   254→      const json = await res.json()
   255→      const rows = ((json.readings ?? []) as MeterReading[]).map((r) => ({
   256→        date: r.date,
   257→        assetCode: r.device?.assetCode ?? '',
   258→        deviceName: r.device?.name ?? '',
   259→        brand: r.device?.brand ?? '',
   260→        model: r.device?.model ?? '',
   261→        prevReading: r.prevReading,
   262→        reading: r.reading,
   263→        delta: r.delta,
   264→        remark: r.remark ?? '',
   265→      }))
   266→      downloadCsv(`meter-readings-${dateStamp()}.csv`, rows, METER_CSV_HEADERS)
   267→      toast.success(`ส่งออก ${rows.length} รายการแล้ว`)
   268→    } catch (e) {
   269→      toast.error(e instanceof Error ? e.message : 'Export failed')
   270→    } finally {
   271→      setExporting(false)
   272→    }
   273→  }
   274→
   275→  return (
   276→    <div className="space-y-4 p-4 md:p-6">
   277→      <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   278→        <div>
   279→          <h1 className="text-2xl font-bold text-slate-800 dark:text-slate-100">จดมิเตอร์</h1>
   280→          <p className="text-sm text-slate-500 dark:text-slate-400">
   281→            บันทึกการอ่านค่ามิเตอร์เครื่องพิมพ์ / ถ่ายเอกสาร
   282→          </p>
   283→        </div>
   284→        <div className="flex items-center gap-2">
   285→          <Button
   286→            variant="outline"
   287→            onClick={() => setCycleDialogOpen(true)}
   288→            className="focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   289→          >
   290→            <Plus className="h-4 w-4" />
   291→            จัดการรอบ
   292→          </Button>
   293→          <Button
   294→            variant="outline"
   295→            onClick={exportCsv}
   296→            disabled={exporting}
   297→            className="focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   298→          >
   299→            <Download className="h-4 w-4" />
   300→            {exporting ? 'กำลังส่งออก...' : 'ส่งออก CSV'}
   301→          </Button>
   302→          <Button
   303→            variant="outline"
   304→            onClick={() => {
   305→              qc.invalidateQueries({ queryKey: ['devices-meter'] })
   306→              qc.invalidateQueries({ queryKey: ['meter-reminders'] })
   307→            }}
   308→            className="focus-visible:ring-2 focus-visible:ring-[#f97316] focus-visible:ring-offset-1 dark:focus-visible:ring-offset-slate-950"
   309→          >
   310→            <RefreshCw className="h-4 w-4" />
   311→            รีเฟรช
   312→          </Button>
   313→        </div>
   314→      </div>
   315→
   316→      {/* Cycle bento cards */}
   317→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   318→        <Card className="lg:col-span-2 dark:border-slate-800 dark:bg-slate-900">
   319→          <CardHeader>
   320→            <CardTitle className="flex items-center gap-2 text-base text-slate-800 dark:text-slate-100">
   321→              <CalendarClock className="h-4 w-4 text-[#f97316]" />
   322→              รอบจดมิเตอร์ปัจจุบัน
   323→            </CardTitle>
   324→          </CardHeader>
   325→          <CardContent>
   326→            {activeCycle ? (
   327→              <div className="space-y-2">
   328→                <div className="text-lg font-semibold text-slate-800 dark:text-slate-100">
   329→                  {activeCycle.name}
   330→                </div>
   331→                <div className="flex flex-wrap items-center gap-3 text-sm text-slate-500 dark:text-slate-400">
   332→                  <span>📅 {activeCycle.startDate} → {activeCycle.endDate}</span>
   333→                  <Badge className="border-emerald-200 bg-emerald-50 text-emerald-700 dark:border-emerald-800 dark:bg-emerald-950 dark:text-emerald-300">
   334→                    {activeCycle.status === 'active' ? 'กำลังดำเนินการ' : activeCycle.status}
   335→                  </Badge>
   336→                </div>
   337→              </div>
   338→            ) : (
   339→              <p className="text-sm text-slate-400 dark:text-slate-500">
   340→                ยังไม่มีรอบจดมิเตอร์ที่กำลังดำเนินการ
   341→              </p>
   342→            )}
   343→          </CardContent>
   344→        </Card>
   345→
   346→        <Card className="dark:border-slate-800 dark:bg-slate-900">
   347→          <CardContent className="flex h-full flex-col items-center justify-center p-6 text-center">
   348→            <Gauge className="mb-2 h-8 w-8 text-[#f97316]" />
   349→            <div className="text-xs font-medium text-slate-500 dark:text-slate-400">เหลือเวลา</div>
   350→            {remainingDays === null ? (
   351→              <div className="text-sm text-slate-400 dark:text-slate-500">—</div>
   352→            ) : (
   353→              <>
   354→                <div className="text-3xl font-bold text-slate-800 dark:text-slate-100">
   355→                  {remainingDays}
   356→                </div>
   357→                <div className="text-xs text-slate-500 dark:text-slate-400">วัน</div>
   358→              </>
   359→            )}
   360→          </CardContent>
   361→        </Card>
   362→
   363→        {/* Progress card — จดแล้ว X/Y */}
   364→        <Card className="dark:border-slate-800 dark:bg-slate-900">
   365→          <CardContent className="flex h-full flex-col justify-center p-5">
   366→            <div className="mb-2 flex items-center gap-2">
   367→              <ClipboardList className="h-4 w-4 text-[#0d9488]" />
   368→              <span className="text-xs font-medium text-slate-500 dark:text-slate-400">
   369→                ความคืบหน้ารอบ
   370→              </span>
   371→            </div>
   372→            {remindersLoading ? (
   373→              <Skeleton className="h-10 w-full dark:bg-slate-800" />
   374→            ) : !remindersData?.hasActiveCycle ? (
   375→              <p className="text-xs text-slate-400 dark:text-slate-500">
   376→                ยังไม่มีรอบที่กำลังดำเนินการ
   377→              </p>
   378→            ) : (
   379→              <>
   380→                <div className="flex items-baseline gap-1">
   381→                  <span className="text-2xl font-bold tabular-nums text-slate-800 dark:text-slate-100">
   382→                    {readCount}
   383→                  </span>
   384→                  <span className="text-sm text-slate-400 dark:text-slate-500">
   385→                    / {totalMeterable}
   386→                  </span>
   387→                </div>
   388→                <Progress
   389→                  value={readPct}
   390→                  className="mt-2 h-2 [&>div]:bg-[#0d9488]"
   391→                />
   392→                <div className="mt-1.5 text-xs text-slate-400 dark:text-slate-500">
   393→                  จดแล้ว {readPct}%
   394→                </div>
   395→              </>
   396→            )}
   397→          </CardContent>
   398→        </Card>
   399→      </div>
   400→
   401→      {/* Reminder banner */}
   402→      {remindersData?.hasActiveCycle && (
   403→        <>
   404→          {remindersData.totalUnread > 0 ? (
   405→            <div className="flex flex-col gap-3 rounded-lg border border-amber-200 bg-gradient-to-r from-amber-50 to-orange-50 p-4 dark:border-amber-900/60 dark:from-amber-950/40 dark:to-orange-950/40 sm:flex-row sm:items-center sm:justify-between">
   406→              <div className="flex items-start gap-3">
   407→                <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-300">
   408→                  <AlertTriangle className="h-5 w-5" />
   409→                </div>
   410→                <div>
   411→                  <div className="text-sm font-semibold text-amber-800 dark:text-amber-200">
   412→                    ⚠️ ยังไม่ได้จดมิเตอร์ {remindersData.totalUnread} จาก {totalMeterable} เครื่องในรอบปัจจุบัน
   413→                  </div>
   414→                  <div className="text-xs text-amber-700 dark:text-amber-300/80">
   415→                    รอบ: {remindersData.cycle?.name} · จดแล้ว {readCount} เครื่อง ({readPct}%)
   416→                  </div>
   417→                </div>
   418→              </div>
   419→              <Button
   420→                size="sm"
   421→                onClick={scrollToTable}
   422→                className="shrink-0 border-amber-300 bg-amber-100 text-amber-800 hover:bg-amber-200 hover:text-amber-900 dark:border-amber-800 dark:bg-amber-900/40 dark:text-amber-200 dark:hover:bg-amber-900/60"
   423→              >
   424→                ดูรายการ
   425→              </Button>
   426→            </div>
   427→          ) : totalMeterable > 0 ? (
   428→            <div className="flex items-center gap-3 rounded-lg border border-emerald-200 bg-emerald-50 p-4 dark:border-emerald-900/60 dark:bg-emerald-950/40">
   429→              <div className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300">
   430→                <CheckCircle2 className="h-5 w-5" />
   431→              </div>
   432→              <div>
   433→                <div className="text-sm font-semibold text-emerald-800 dark:text-emerald-200">
   434→                  ✅ จดมิเตอร์ครบทุกเครื่องในรอบปัจจุบันแล้ว
   435→                </div>
   436→                <div className="text-xs text-emerald-700 dark:text-emerald-300/80">
   437→                  รวม {totalMeterable} เครื่อง · รอบ {remindersData.cycle?.name}
   438→                </div>
   439→              </div>
   440→            </div>
   441→          ) : null}
   442→        </>
   443→      )}
   444→
   445→      {/* Devices table */}
   446→      <Card className="dark:border-slate-800 dark:bg-slate-900" >
   447→        <CardHeader>
   448→          <CardTitle className="text-base text-slate-800 dark:text-slate-100">รายการอุปกรณ์ที่ต้องจดมิเตอร์</CardTitle>
   449→        </CardHeader>
   450→        <CardContent>
   451→          <div
   452→            ref={tableRef}
   453→            className="itam-scroll max-h-[55vh] overflow-auto rounded-md border border-slate-200 dark:border-slate-800"
   454→          >
   455→            <Table>
   456→              <TableHeader className="sticky top-0 z-10 bg-slate-50 dark:bg-slate-900">
   457→                <TableRow>
   458→                  <TableHead className="text-slate-600 dark:text-slate-300">รหัส</TableHead>
   459→                  <TableHead className="text-slate-600 dark:text-slate-300">ชื่อ</TableHead>
   460→                  <TableHead className="text-slate-600 dark:text-slate-300">แบรนด์/รุ่น</TableHead>
   461→                  <TableHead className="text-slate-600 dark:text-slate-300">ประเภท</TableHead>
   462→                  <TableHead className="text-slate-600 dark:text-slate-300">สถานะ</TableHead>
   463→                  <TableHead className="text-slate-600 dark:text-slate-300">สาขา</TableHead>
   464→                  <TableHead className="text-right text-slate-600 dark:text-slate-300">ค่าล่าสุด</TableHead>
   465→                  <TableHead className="text-slate-600 dark:text-slate-300">สถานะรอบ</TableHead>
   466→                  <TableHead className="text-right text-slate-600 dark:text-slate-300">จดมิเตอร์</TableHead>
   467→                </TableRow>
   468→              </TableHeader>
   469→              <TableBody>
   470→                {isLoading ? (
   471→                  Array.from({ length: 5 }).map((_, i) => (
   472→                    <TableRow key={`sk-${i}`}>
   473→                      <TableCell colSpan={9}>
   474→                        <Skeleton className="h-6 w-full dark:bg-slate-800" />
   475→                      </TableCell>
   476→                    </TableRow>
