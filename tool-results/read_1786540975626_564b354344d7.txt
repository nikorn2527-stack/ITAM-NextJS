   198→function formatDateTime(iso: string | null): string {
   199→  if (!iso) return '—'
   200→  try {
   201→    return new Date(iso).toLocaleString('th-TH', {
   202→      day: '2-digit',
   203→      month: '2-digit',
   204→      year: 'numeric',
   205→      hour: '2-digit',
   206→      minute: '2-digit',
   207→    })
   208→  } catch {
   209→    return iso
   210→  }
   211→}

   236→export function WorkOrdersPage() {
   237→  const qc = useQueryClient()
   238→  const [search, setSearch] = React.useState('')
   239→  const [statusFilter, setStatusFilter] = React.useState<string>('all')
   240→  const [priorityFilter, setPriorityFilter] = React.useState<string>('all')
   241→  const [page, setPage] = React.useState(1)
   242→  const [createOpen, setCreateOpen] = React.useState(false)
   243→  const [form, setForm] = React.useState<NewFormState>(EMPTY_FORM)
   244→  const [saving, setSaving] = React.useState(false)
   245→  const [detailId, setDetailId] = React.useState<string | null>(null)
   246→
   247→  // Debounce search to avoid spamming the API on every keystroke
   248→  const [debouncedSearch, setDebouncedSearch] = React.useState('')
   249→  React.useEffect(() => {
   250→    const t = setTimeout(() => {
   251→      setDebouncedSearch(search)
   252→      setPage(1)
   253→    }, 350)
   254→    return () => clearTimeout(t)
   255→  }, [search])
   256→
   257→  React.useEffect(() => {
   258→    setPage(1)
   259→  }, [statusFilter, priorityFilter])
   260→
   261→  const listQuery = useQuery<WorkOrderListResponse>({
   262→    queryKey: [
   263→      'work-orders',
   264→      debouncedSearch,
   265→      statusFilter,
   266→      priorityFilter,
   267→      page,
   268→    ],
   269→    queryFn: async () => {
   270→      const params = new URLSearchParams()
   271→      if (debouncedSearch) params.set('search', debouncedSearch)
   272→      if (statusFilter !== 'all') params.set('status', statusFilter)
   273→      if (priorityFilter !== 'all') params.set('priority', priorityFilter)
   274→      params.set('page', String(page))
   275→      params.set('pageSize', String(PAGE_SIZE))
   276→      const res = await fetch(`/api/work-orders?${params.toString()}`)
   277→      if (!res.ok) throw new Error('Failed to load work orders')
   278→      return res.json()
   279→    },
   280→  })
   281→
   282→  const items = listQuery.data?.data ?? []
   283→  const stats = listQuery.data?.stats ?? {
   284→    PENDING: 0,
   285→    IN_PROGRESS: 0,
   286→    COMPLETED: 0,
   287→    CANCELLED: 0,
   288→  }
   289→  const pagination = listQuery.data?.pagination
   290→
   291→  function openCreate() {
   292→    setForm({ ...EMPTY_FORM })
   293→    setCreateOpen(true)
   294→  }
   295→
   296→  async function handleCreate() {
   297→    if (!form.subject.trim()) {
   298→      toast.error('กรุณาระบุประเภทปัญหา')
   299→      return
   300→    }
   301→    try {
   302→      setSaving(true)
   303→      const res = await fetch('/api/work-orders', {
   304→        method: 'POST',
   305→        headers: { 'Content-Type': 'application/json' },
   306→        body: JSON.stringify({
   307→          subject: form.subject.trim(),
   308→          building: form.building.trim() || null,
   309→          location: form.location.trim() || null,
   310→          details: form.details.trim() || null,
   311→          priority: form.priority,
   312→          reporterName: form.reporterName.trim() || null,
   313→          tel: form.tel.trim() || null,
   314→          actor: form.reporterName.trim() || 'system',
   315→        }),
   316→      })
   317→      if (!res.ok) {
   318→        const j = await res.json().catch(() => ({}))
   319→        throw new Error(j.error ?? 'บันทึกไม่สำเร็จ')
   320→      }
   321→      const json = await res.json()
   322→      toast.success(`สร้างใบแจ้งซ่อม ${json.data?.woNumber ?? ''} แล้ว`)
   323→      setCreateOpen(false)
   324→      qc.invalidateQueries({ queryKey: ['work-orders'] })
   325→      // Open the new detail right away
   326→      if (json.data?.id) setDetailId(json.data.id)
   327→    } catch (e) {
   328→      toast.error(e instanceof Error ? e.message : 'บันทึกไม่สำเร็จ')
   329→    } finally {
   330→      setSaving(false)
   331→    }
   332→  }
   333→
   334→  return (
   335→    <div className="space-y-4 p-4 md:p-6">
   336→      {/* Header */}
   337→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   338→        <div>
   339→          <h1 className="flex items-center gap-2 text-xl font-bold md:text-2xl">
   340→            <Wrench className="h-6 w-6 text-orange-500" />
   341→            แจ้งซ่อม
   342→          </h1>
   343→          <p className="text-sm text-muted-foreground">
   344→            ระบบแจ้งซ่อมครบวงจร — แจ้ง → รับงาน → ซ่อม → ปิดงาน
   345→          </p>
   346→        </div>
   347→        <div className="flex items-center gap-2">
   348→          <Button
   349→            variant="outline"
   350→            size="sm"
   351→            onClick={() => listQuery.refetch()}
   352→            disabled={listQuery.isFetching}
   353→            aria-label="รีเฟรช"
   354→          >
   355→            <RefreshCw
   356→              className={listQuery.isFetching ? 'h-4 w-4 animate-spin' : 'h-4 w-4'}
   357→            />
   358→            <span className="hidden sm:inline">รีเฟรช</span>
   359→          </Button>
   360→          <Button size="sm" onClick={openCreate} className="bg-orange-500 hover:bg-orange-600">
   361→            <Plus className="h-4 w-4" />
   362→            แจ้งซ่อมใหม่
   363→          </Button>
   364→        </div>
   365→      </div>
   366→
   367→      {/* KPI stats bar */}
   368→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   369→        <KpiCard
   370→          label="รอดำเนินการ"
   371→          value={stats.PENDING ?? 0}
   372→          icon={<Clock className="h-5 w-5" />}
   373→          color="amber"
   374→        />
   375→        <KpiCard
   376→          label="กำลังซ่อม"
   377→          value={(stats.IN_PROGRESS ?? 0) + (stats.WAITING_PARTS ?? 0)}
   378→          icon={<Wrench className="h-5 w-5" />}
   379→          color="blue"
   380→        />
   381→        <KpiCard
   382→          label="เสร็จแล้ว"
   383→          value={stats.COMPLETED ?? 0}
   384→          icon={<CheckCircle2 className="h-5 w-5" />}
   385→          color="emerald"
   386→        />
   387→        <KpiCard
   388→          label="ยกเลิก"
   389→          value={stats.CANCELLED ?? 0}
   390→          icon={<XCircle className="h-5 w-5" />}
   391→          color="rose"
   392→        />
   393→      </div>
   394→
   395→      {/* Filter bar */}
   396→      <Card className="gap-3 py-3">
   397→        <CardContent className="flex flex-col gap-3 px-3 md:flex-row md:items-center">
   398→          <div className="relative flex-1">
   399→            <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   400→            <Input
   401→              value={search}
   402→              onChange={(e) => setSearch(e.target.value)}
   403→              placeholder="ค้นหาเลขใบงาน / ปัญหา / สถานที่ / ผู้แจ้ง / เบอร์โทร"
   404→              className="pl-9"
   405→              aria-label="ค้นหาใบแจ้งซ่อม"
   406→            />
   407→          </div>
   408→          <div className="grid grid-cols-2 gap-2 md:flex md:w-auto">
   409→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   410→              <SelectTrigger className="w-full md:w-[160px]" aria-label="กรองสถานะ">
   411→                <SelectValue placeholder="สถานะ" />
   412→              </SelectTrigger>
   413→              <SelectContent>
   414→                {STATUS_OPTIONS.map((o) => (
   415→                  <SelectItem key={o.value} value={o.value}>
   416→                    {o.label}
   417→                  </SelectItem>
   418→                ))}
   419→              </SelectContent>

   423→                <SelectValue placeholder="ความเร่งด่วน" />
   424→              </SelectTrigger>
   425→              <SelectContent>
   426→                {PRIORITY_OPTIONS.map((o) => (
   427→                  <SelectItem key={o.value} value={o.value}>
   428→                    {o.label}
   429→                  </SelectItem>
   430→                ))}
   431→              </SelectContent>
   432→            </Select>
   433→          </div>
   434→        </CardContent>
   435→      </Card>
   436→
   437→      {/* List */}
   438→      {listQuery.isLoading ? (
   439→        <div className="grid grid-cols-1 gap-3 md:grid-cols-2 xl:grid-cols-3">
   440→          {Array.from({ length: 6 }).map((_, i) => (
   441→            <Skeleton key={i} className="h-44 w-full rounded-xl" />
   442→          ))}
   443→        </div>
   444→      ) : items.length === 0 ? (
   445→        <Card className="py-12">
   446→          <CardContent className="flex flex-col items-center justify-center gap-3 text-center">
   447→            <div className="flex h-14 w-14 items-center justify-center rounded-full bg-orange-100 dark:bg-orange-950">
   448→              <Wrench className="h-7 w-7 text-orange-500" />
   449→            </div>
   450→            <div className="text-base font-medium">ยังไม่มีใบแจ้งซ่อม</div>
   451→            <p className="max-w-sm text-sm text-muted-foreground">
   452→              กดปุ่ม &quot;แจ้งซ่อมใหม่&quot; เพื่อสร้างใบงานแรก หรือปรับตัวกรองเพื่อค้นหาใบงานเก่า
   453→            </p>
   454→            <Button size="sm" onClick={openCreate} className="mt-1 bg-orange-500 hover:bg-orange-600">
   455→              <Plus className="h-4 w-4" />
   456→              แจ้งซ่อมใหม่
   457→            </Button>
   458→          </CardContent>
   459→        </Card>

   461→        <div className="grid grid-cols-1 gap-3 md:grid-cols-2 xl:grid-cols-3">
   462→          <AnimatePresence mode="popLayout">

   464→              <motion.div
   465→                key={wo.id}
   466→                layout
   467→                initial={{ opacity: 0, y: 8 }}
   468→                animate={{ opacity: 1, y: 0 }}
   469→                exit={{ opacity: 0, scale: 0.97 }}
   470→                transition={{ duration: 0.18 }}
   471→              >
   472→                <WorkOrderCard wo={wo} onOpen={() => setDetailId(wo.id)} />
   473→              </motion.div>
   474→            ))}
   475→          </AnimatePresence>
   476→        </div>
   477→      )}
   478→
   479→      {/* Pagination */}
   480→      {pagination && pagination.totalPages > 1 && (
   481→        <div className="flex flex-col items-center gap-2 sm:flex-row sm:justify-between">
   482→          <p className="text-sm text-muted-foreground">
   483→            ทั้งหมด {pagination.total} รายการ • หน้า {pagination.page} / {pagination.totalPages}
   484→          </p>
   485→          <div className="flex items-center gap-1">
   486→            <Button
   487→              variant="outline"
   488→              size="sm"
   489→              disabled={page <= 1}
   490→              onClick={() => setPage((p) => Math.max(1, p - 1))}
   491→            >
   492→              ก่อนหน้า
   493→            </Button>
   494→            <Button
   495→              variant="outline"
   496→              size="sm"
   497→              disabled={page >= pagination.totalPages}
   498→              onClick={() => setPage((p) => Math.min(pagination.totalPages, p + 1))}
   499→            >
   500→              ถัดไป
   501→            </Button>
   502→          </div>
   503→        </div>
   504→      )}
   505→
   506→      {/* Create dialog */}
   507→      <CreateWorkOrderDialog
   508→        open={createOpen}
   509→        onOpenChange={setCreateOpen}
   510→        form={form}
   511→        setForm={setForm}
   512→        saving={saving}
   513→        onSubmit={handleCreate}
   514→      />
   515→
   516→      {/* Detail dialog */}
   517→      <WorkOrderDetailDialog
   518→        id={detailId}
   519→        onOpenChange={(open) => {
   520→          if (!open) setDetailId(null)
   521→        }}
   522→      />
   523→    </div>
   524→  )
   525→}

   881→function WorkOrderDetailContent({
   882→  wo,
   883→  onMutated,
   884→  onClose,
   885→}: {
   886→  wo: WorkOrderDetail
   887→  onMutated: () => void
   888→  onClose: () => void
   889→}) {
   890→  // Assign technician
   891→  const [assignOpen, setAssignOpen] = React.useState(false)
   892→  const [techName, setTechName] = React.useState(wo.assignedTo ?? '')
   893→  const [assignNote, setAssignNote] = React.useState(wo.assignmentNote ?? '')
   894→  const [assigning, setAssigning] = React.useState(false)
   895→
   896→  // Complete
   897→  const [completeOpen, setCompleteOpen] = React.useState(false)
   898→  const [completeNote, setCompleteNote] = React.useState('')
   899→  const [completing, setCompleting] = React.useState(false)
   900→
   901→  // Cancel
   902→  const [cancelOpen, setCancelOpen] = React.useState(false)
   903→  const [cancelReason, setCancelReason] = React.useState('')
   904→  const [canceling, setCanceling] = React.useState(false)
   905→
   906→  // Chat
   907→  const [chatText, setChatText] = React.useState('')
   908→  const [sendingMsg, setSendingMsg] = React.useState(false)
   909→  const messagesEndRef = React.useRef<HTMLDivElement>(null)
   910→  const messages = wo.messages ?? []
   911→
   912→  React.useEffect(() => {
   913→    // sync tech name when WO changes
   914→    setTechName(wo.assignedTo ?? '')
   915→    setAssignNote(wo.assignmentNote ?? '')
   916→  }, [wo.id, wo.assignedTo, wo.assignmentNote])
   917→
   918→  React.useEffect(() => {
   919→    if (messagesEndRef.current) {
   920→      messagesEndRef.current.scrollIntoView({ behavior: 'smooth' })
   921→    }
   922→  }, [messages.length])
   923→
   924→  const canAssign = wo.status === 'PENDING' || wo.status === 'IN_PROGRESS' || wo.status === 'WAITING_PARTS'
   925→  const canComplete = wo.status === 'IN_PROGRESS' || wo.status === 'WAITING_PARTS'
   926→  const canCancel = wo.status !== 'COMPLETED' && wo.status !== 'CANCELLED'
   927→  const canChat = wo.status !== 'COMPLETED' && wo.status !== 'CANCELLED'
   928→
   929→  async function handleAssign() {
   930→    if (!techName.trim()) {
   931→      toast.error('กรุณาระบุชื่อช่าง')
   932→      return
   933→    }
   934→    try {
   935→      setAssigning(true)
   936→      const res = await fetch(`/api/work-orders/${wo.id}/assign`, {
   937→        method: 'POST',
   938→        headers: { 'Content-Type': 'application/json' },
   939→        body: JSON.stringify({
   940→          assignedTo: techName.trim(),
   941→          assignmentNote: assignNote.trim() || null,
   942→          actor: 'admin',
   943→        }),
   944→      })
   945→      if (!res.ok) {
   946→        const j = await res.json().catch(() => ({}))
   947→        throw new Error(j.error ?? 'มอบหมายไม่สำเร็จ')
   948→      }
   949→      toast.success(`มอบหมายให้ ${techName.trim()} แล้ว`)
   950→      setAssignOpen(false)
   951→      onMutated()
   952→    } catch (e) {
   953→      toast.error(e instanceof Error ? e.message : 'มอบหมายไม่สำเร็จ')
   954→    } finally {
   955→      setAssigning(false)
   956→    }
   957→  }
   958→
   959→  async function handleComplete() {
   960→    try {
   961→      setCompleting(true)
   962→      const res = await fetch(`/api/work-orders/${wo.id}/complete`, {
   963→        method: 'POST',
   964→        headers: { 'Content-Type': 'application/json' },
   965→        body: JSON.stringify({
   966→          note: completeNote.trim() || null,
   967→          actor: 'admin',
   968→        }),
   969→      })
   970→      if (!res.ok) {
   971→        const j = await res.json().catch(() => ({}))
   972→        throw new Error(j.error ?? 'ปิดงานไม่สำเร็จ')
   973→      }
   974→      toast.success('ปิดงานเรียบร้อย')
   975→      setCompleteOpen(false)
   976→      setCompleteNote('')
   977→      onMutated()
   978→    } catch (e) {
   979→      toast.error(e instanceof Error ? e.message : 'ปิดงานไม่สำเร็จ')
   980→    } finally {
   981→      setCompleting(false)
   982→    }
   983→  }
   984→
   985→  async function handleCancel() {
   986→    if (!cancelReason.trim()) {
   987→      toast.error('กรุณาระบุเหตุผลในการยกเลิก')
   988→      return
   989→    }
   990→    try {
   991→      setCanceling(true)
   992→      const res = await fetch(`/api/work-orders/${wo.id}/cancel`, {
   993→        method: 'POST',
   994→        headers: { 'Content-Type': 'application/json' },
   995→        body: JSON.stringify({
   996→          reason: cancelReason.trim(),
   997→          actor: 'admin',
   998→        }),
   999→      })
  1000→      if (!res.ok) {
  1001→        const j = await res.json().catch(() => ({}))
  1002→        throw new Error(j.error ?? 'ยกเลิกไม่สำเร็จ')
  1003→      }
  1004→      toast.success('ยกเลิกใบงานแล้ว')
  1005→      setCancelOpen(false)
  1006→      setCancelReason('')
  1007→      onMutated()
  1008→    } catch (e) {
  1009→      toast.error(e instanceof Error ? e.message : 'ยกเลิกไม่สำเร็จ')
  1010→    } finally {
  1011→      setCanceling(false)
  1012→    }
  1013→  }
  1014→
  1015→  async function handleSendMessage() {
  1016→    const text = chatText.trim()
  1017→    if (!text) return
  1018→    try {
  1019→      setSendingMsg(true)
  1020→      const res = await fetch(`/api/work-orders/${wo.id}/messages`, {
  1021→        method: 'POST',
  1022→        headers: { 'Content-Type': 'application/json' },
  1023→        body: JSON.stringify({
  1024→          message: text,
  1025→          author: 'admin',
  1026→          authorRole: 'admin',
  1027→        }),
  1028→      })
  1029→      if (!res.ok) {
  1030→        const j = await res.json().catch(() => ({}))
  1031→        throw new Error(j.error ?? 'ส่งข้อความไม่สำเร็จ')
  1032→      }
  1033→      setChatText('')
  1034→      onMutated()
  1035→    } catch (e) {
  1036→      toast.error(e instanceof Error ? e.message : 'ส่งข้อความไม่สำเร็จ')
  1037→    } finally {
  1038→      setSendingMsg(false)
  1039→    }
  1040→  }
  1041→
  1042→  // Build timeline from key WO timestamps + messages
  1043→  const timeline: Array<{ key: string; label: string; at: string | null; tone: 'info' | 'success' | 'warning' | 'danger' | 'muted' }> = [
  1044→    { key: 'created', label: 'แจ้งซ่อมใหม่', at: wo.createdAt, tone: 'info' },
  1045→    { key: 'assigned', label: wo.assignedTo ? `มอบหมายให้ ${wo.assignedTo}` : 'มอบหมาย', at: wo.assignedAt, tone: wo.assignedAt ? 'info' : 'muted' },
  1046→    { key: 'completed', label: 'ปิดงาน', at: wo.workCompletedAt ?? wo.closedAt, tone: wo.workCompletedAt ? 'success' : 'muted' },
  1047→    { key: 'cancelled', label: wo.cancelReason ? `ยกเลิก — ${wo.cancelReason}` : 'ยกเลิก', at: wo.canceledAt, tone: wo.canceledAt ? 'danger' : 'muted' },
  1048→  ].filter((t) => t.at !== null || t.key === 'created')
  1049→
  1050→  return (
  1051→    <div className="flex max-h-[92vh] flex-col">
  1052→      {/* Header */}
  1053→      <div className="space-y-2 border-b px-5 py-4">
  1054→        <div className="flex items-center justify-between gap-2">
  1055→          <span className="font-mono text-xs font-semibold text-muted-foreground">
  1056→            {wo.woNumber ?? '—'}
  1057→          </span>
  1058→          <Badge className={statusBadgeClass(wo.status)} variant="outline">
  1059→            {statusLabel(wo.status)}
  1060→          </Badge>
  1061→        </div>
  1062→        <h2 className="text-lg font-bold leading-tight">{wo.subject}</h2>
  1063→        <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
  1064→          <span className="flex items-center gap-1">
  1065→            <CalendarClock className="h-3.5 w-3.5" />
  1066→            {formatDateTime(wo.createdAt)}
  1067→          </span>
  1068→          <span aria-hidden>•</span>
  1069→          <Badge className={priorityBadgeClass(wo.priority)} variant="outline">
  1070→            {wo.priority}
  1071→          </Badge>
  1072→        </div>
  1073→      </div>
  1074→
  1075→      {/* Body — scrollable */}
  1076→      <ScrollArea className="flex-1 overflow-y-auto">
  1077→        <div className="space-y-4 px-5 py-4">
  1078→          {/* Info grid */}
  1079→          <div className="grid grid-cols-1 gap-2 sm:grid-cols-2">
  1080→            <InfoRow
  1081→              icon={<MapPin className="h-4 w-4" />}
  1082→              label="อาคาร / ฝ่าย"
  1083→              value={wo.building ?? '—'}
  1084→            />
  1085→            <InfoRow
  1086→              icon={<MapPin className="h-4 w-4" />}
  1087→              label="ตำแหน่ง"
  1088→              value={wo.location ?? '—'}
  1089→            />
  1090→            <InfoRow
  1091→              icon={<User className="h-4 w-4" />}
  1092→              label="ผู้แจ้ง"
  1093→              value={wo.reporterName ?? '—'}
