    82→/**
    83→ * Animated count-up hook — tweens from the previous value to the next
    84→ * over 500ms using requestAnimationFrame.
    85→ */
    86→function useCountUp(target: number, duration = 500) {
    87→  const [display, setDisplay] = React.useState(target)
    88→  const fromRef = React.useRef(target)
    89→  const rafRef = React.useRef<number | null>(null)
    90→  const startRef = React.useRef<number | null>(null)
    91→
    92→  React.useEffect(() => {
    93→    if (typeof window === 'undefined') return
    94→    const from = fromRef.current
    95→    if (from === target) {
    96→      setDisplay(target)
    97→      return
    98→    }
    99→    startRef.current = null
   100→    const step = (ts: number) => {
   101→      if (startRef.current === null) startRef.current = ts
   102→      const elapsed = ts - startRef.current
   103→      const t = Math.min(1, elapsed / duration)
   104→      // easeOutCubic
   105→      const eased = 1 - Math.pow(1 - t, 3)
   106→      const next = Math.round(from + (target - from) * eased)
   107→      setDisplay(next)
   108→      if (t < 1) {
   109→        rafRef.current = window.requestAnimationFrame(step)
   110→      } else {
   111→        fromRef.current = target
   112→        setDisplay(target)
   113→      }
   114→    }
   115→    rafRef.current = window.requestAnimationFrame(step)
   116→    return () => {
   117→      if (rafRef.current !== null) {
   118→        window.cancelAnimationFrame(rafRef.current)
   119→      }
   120→      // remember where we landed so the next tween starts smoothly
   121→      fromRef.current = display
   122→    }
   123→  }, [target, duration])
   124→
   125→  return display
   126→}

   128→function KpiCard({
