    40→// ============== Count-up hook ==============
    41→function useCountUp(value: number, durationMs = 600): number {
    42→  const [display, setDisplay] = React.useState(value)
    43→  const fromRef = React.useRef(value)
    44→  const rafRef = React.useRef<number | null>(null)
    45→  React.useEffect(() => {
    46→    const from = fromRef.current
    47→    const to = value
    48→    if (from === to) return
    49→    const start = performance.now()
    50→    const tick = (now: number) => {
    51→      const t = Math.min(1, (now - start) / durationMs)
    52→      const eased = 1 - Math.pow(1 - t, 3)
    53→      setDisplay(Math.round(from + (to - from) * eased))
    54→      if (t < 1) rafRef.current = requestAnimationFrame(tick)
    55→      else fromRef.current = to
    56→    }
    57→    rafRef.current = requestAnimationFrame(tick)
    58→    return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current) }
    59→  }, [value, durationMs])
    60→  return display
    61→}
