    14→ *   │   ↑↓ to navigate            │                                       │
    15→ *   │   Enter to fill input       │  Inputs:                              │
    16→ *   │                             │   · TOTAL  → ค่ามิเตอร์ (1 input)      │

    91→function fmtTime(ts: number): string {

   101→export function ItamMeterKeyboard() {
   102→  const qc = useQueryClient()
   103→  const [searchInput, setSearchInput] = React.useState('')
   104→  const [search, setSearch] = React.useState('')
   105→  const [selectedIndex, setSelectedIndex] = React.useState(0)
   106→  const [bwInput, setBwInput] = React.useState('')
   107→  const [colorInput, setColorInput] = React.useState('')
   108→  const [remark, setRemark] = React.useState('')
   109→  const [saving, setSaving] = React.useState(false)
   110→  const [recent, setRecent] = React.useState<RecentlyKeyed[]>([])
   111→  const [focus, setFocus] = React.useState<'search' | 'meter'>('search')
   112→
   113→  // ── Pending device (from QR scan or dashboard quick-action) ──
   114→  // When the user scans a QR code or clicks a "จดมิเตอร์" button on a specific
   115→  // device, the store sets pendingDeviceId. We pre-select that device and
   116→  // jump straight to meter input — no search needed.
   117→  const pendingDeviceId = useAppStore((s) => s.pendingDeviceId)
   118→  const clearPendingDeviceId = useAppStore((s) => s.clearPendingDeviceId)
   119→
   120→  const searchInputRef = React.useRef<HTMLInputElement>(null)
   121→  const bwInputRef = React.useRef<HTMLInputElement>(null)
   122→  const colorInputRef = React.useRef<HTMLInputElement>(null)
   123→  const listRef = React.useRef<HTMLDivElement>(null)
   124→
   125→  // Debounce the search input → search state (200ms).
   126→  React.useEffect(() => {
   127→    const t = setTimeout(() => {
   128→      setSearch(searchInput.trim())
   129→      setSelectedIndex(0)
   130→    }, 200)
   131→    return () => clearTimeout(t)
   132→  }, [searchInput])
   133→
   134→  const { data, isLoading, refetch, isFetching } = useQuery<UnreadResponse>({
   135→    queryKey: ['itam-meter-keyboard', search],
   136→    queryFn: async () => {
   137→      const q = search ? `&search=${encodeURIComponent(search)}` : ''
   138→      const res = await fetch(`/api/itam/meter-readings/unread?limit=500&includeRead=1${q}`)
   139→      if (!res.ok) throw new Error('Failed')
   140→      return res.json()
   141→    },
   142→  })
   143→
   144→  const devices = data?.devices ?? []
   145→  const total = data?.total ?? 0
   146→  const read = data?.read ?? 0
   147→  const unread = data?.unread ?? 0
   148→
   149→  // ── Pre-select pending device (from QR scan or quick-action) ──
   150→  // When a pendingDeviceId is set, find it in the loaded list and select it,
   151→  // then jump focus to the meter input. This makes the flow: scan QR →
   152→  // (page loads) → device already selected → just type meter → Enter.
   153→  React.useEffect(() => {
   154→    if (!pendingDeviceId || devices.length === 0) return
   155→    const idx = devices.findIndex((d) => d.assetNo === pendingDeviceId)
   156→    if (idx >= 0) {
   157→      setSelectedIndex(idx)
   158→      setFocus('meter')
   159→      // Scroll the selected item into view
   160→      setTimeout(() => {
   161→        const el = listRef.current?.querySelector(`[data-idx="${idx}"]`)
   162→        el?.scrollIntoView({ block: 'center', behavior: 'smooth' })
   163→      }, 50)
   164→      toast.success(`เลือกอุปกรณ์ ${pendingDeviceId} แล้ว — พิมพ์มิเตอร์แล้วกด Enter`)
   165→      clearPendingDeviceId()
   166→    }
   167→  }, [pendingDeviceId, devices, clearPendingDeviceId])
   168→  const pct = total > 0 ? Math.round((read / total) * 100) : 0
   169→
   170→  // The selected device (clamped to the visible list).
   171→  const selected = devices[selectedIndex] ?? null
   172→
   173→  // When the selected device changes, reset the inputs (and put focus on the
   174→  // BW input when the user is in "meter" mode).
   175→  React.useEffect(() => {
   176→    if (!selected) {
   177→      setBwInput('')
   178→      setColorInput('')
   179→      setRemark('')
   180→      return
   181→    }
   182→    // Pre-fill the new input with the last meter value (Apps Script behavior —
   183→    // user types over it or appends to it).
   184→    setBwInput(String(selected.lastMeterBw || ''))
   185→    setColorInput(String(selected.lastMeterColor || ''))
   186→    setRemark('')
   187→  }, [selected?.assetNo])
   188→
   189→  // When focus target flips to "meter", jump into the BW input.
   190→  React.useEffect(() => {
   191→    if (focus === 'meter' && selected) {
   192→      const t = setTimeout(() => {
   193→        const el = bwInputRef.current
   194→        if (el) {
   195→          el.focus()
   196→          el.select()
   197→        }
   198→      }, 30)
   199→      return () => clearTimeout(t)
   200→    }
   201→    if (focus === 'search') {
   202→      const t = setTimeout(() => searchInputRef.current?.focus(), 30)
   203→      return () => clearTimeout(t)
   204→    }
   205→  }, [focus, selected?.assetNo])
   206→
   207→  // Scroll the selected row into view in the list.
   208→  React.useEffect(() => {
   209→    if (!listRef.current) return
   210→    const row = listRef.current.querySelector<HTMLElement>(`[data-row-idx="${selectedIndex}"]`)
   211→    if (row) row.scrollIntoView({ block: 'nearest' })
   212→  }, [selectedIndex])
   213→
   214→  // ─── Global keyboard handler ─────────────────────────────────────────
   215→  //   • ↑/↓ move selection (regardless of focus, but only when not editing text)
   216→  //   • Enter in search → keep focus in search (already filtered by debounce)
   217→  //   • Enter in BW input → if BW_COLOR, focus color; otherwise save
   218→  //   • Enter in color input → save
   219→  //   • Escape → clear search / blur meter
   220→  React.useEffect(() => {
   221→    function onKey(e: KeyboardEvent) {
   222→      const tag = (e.target as HTMLElement | null)?.tagName?.toLowerCase()
   223→      const inTextField = tag === 'input' || tag === 'textarea'
   224→
   225→      // ArrowUp / ArrowDown — always handle (so the user can navigate the
   226→      // list even while typing in the search box).
   227→      if (e.key === 'ArrowDown') {
   228→        e.preventDefault()
   229→        setSelectedIndex((i) => Math.min(devices.length - 1, i + 1))
   230→        return
   231→      }
   232→      if (e.key === 'ArrowUp') {
   233→        e.preventDefault()
   234→        setSelectedIndex((i) => Math.max(0, i - 1))
   235→        return
   236→      }
   237→
   238→      if (e.key === 'Escape') {
   239→        if (inTextField) {
   240→          // First Escape blurs the field; second clears the value.
   241→          ;(e.target as HTMLElement).blur()
   242→        } else if (searchInput) {
   243→          setSearchInput('')
   244→          setSearch('')
   245→        }
   246→        return
   247→      }
   248→
   249→      if (e.key === 'Enter') {
   250→        if (focus === 'search') {
   251→          // Enter in the search box → jump to the meter input of the first hit.
   252→          e.preventDefault()
   253→          if (devices.length > 0) {
   254→            setSelectedIndex(0)
   255→            setFocus('meter')
   256→          }
   257→          return
   258→        }
   259→        // focus === 'meter'
   260→        const isColor = (selected?.meterMode || 'TOTAL').toUpperCase() === 'BW_COLOR'
   261→        if (isColor && document.activeElement === bwInputRef.current) {
   262→          // Move to color input.
   263→          e.preventDefault()
   264→          colorInputRef.current?.focus()
   265→          colorInputRef.current?.select()
   266→          return
   267→        }
   268→        // Otherwise Enter saves.
   269→        e.preventDefault()
   270→        void saveReading()
   271→      }
   272→    }
   273→    window.addEventListener('keydown', onKey)
   274→    return () => window.removeEventListener('keydown', onKey)
   275→  }, [devices.length, focus, searchInput, selected])
   276→
   277→  async function saveReading() {
   278→    if (!selected) return
   279→    const bwNum = Number(bwInput)
   280→    if (!bwInput || !Number.isFinite(bwNum)) {
   281→      toast.error('กรุณากรอกค่ามิเตอร์')
   282→      bwInputRef.current?.focus()
   283→      return
   284→    }
   285→    const colorNum = colorInput ? Number(colorInput) : 0
   286→    const isReset = bwNum < selected.lastMeterBw || colorNum < selected.lastMeterColor
   287→    if (isReset && !remark.trim()) {
   288→      toast.error('ค่าใหม่น้อยกว่าค่าก่อนหน้า (RESET) — กรุณาระบุหมายเหตุ')
   289→      return
   290→    }
   291→    try {
   292→      setSaving(true)
   293→      const res = await fetch('/api/itam/meter-readings', {
   294→        method: 'POST',
   295→        headers: { 'Content-Type': 'application/json' },
   296→        body: JSON.stringify({
   297→          assetNo: selected.assetNo,
   298→          meterBw: bwNum,
   299→          meterColor: colorNum,
   300→          prevMeterBw: selected.lastMeterBw,
   301→          prevMeterColor: selected.lastMeterColor,
   302→          remark: remark.trim() || (isReset ? 'RESET' : null),
   303→          readingType: isReset ? 'RESET' : 'MONTHLY',
   304→        }),
   305→      })
   306→      if (!res.ok) {
   307→        const j = await res.json().catch(() => ({}))
   308→        throw new Error(j.error || 'Save failed')
   309→      }
   310→      const j = await res.json()
   311→      const pagesBw: number = j.pagesBw ?? Math.max(0, bwNum - selected.lastMeterBw)
   312→      const pagesColor: number = j.pagesColor ?? Math.max(0, colorNum - selected.lastMeterColor)
   313→
   314→      // Push to "recently keyed" stack (top 5).
   315→      setRecent((prev) =>
   316→        [
   317→          {
   318→            assetNo: selected.assetNo,
   319→            name: deviceLabel(selected),
   320→            meterBw: bwNum,
   321→            meterColor: colorNum,
   322→            delta: pagesBw + pagesColor,
   323→            at: Date.now(),
   324→            reset: isReset,
   325→          },
   326→          ...prev,
   327→        ].slice(0, 5),
   328→      )
   329→
   330→      toast.success(
   331→        `บันทึกมิเตอร์ ${selected.assetNo} · +${(pagesBw + pagesColor).toLocaleString()} แผ่น`,
   332→        { description: isReset ? '⚠️ RESET' : undefined },
   333→      )
   334→
   335→      // Move focus to the next unread device and refocus the BW input.
   336→      await qc.invalidateQueries({ queryKey: ['itam-meter-keyboard'] })
   337→      await qc.invalidateQueries({ queryKey: ['itam-readings'] })
   338→      await qc.invalidateQueries({ queryKey: ['itam-dashboard'] })
   339→
   340→      // Optimistically advance — after the query refetches, the new "selected"
   341→      // may be a different assetNo; we use the current selectedIndex which
   342→      // (because the just-read device drops out of the unread list) naturally
   343→      // points to the next device.
   344→      setSelectedIndex((i) => Math.min(i, Math.max(0, devices.length - 2)))
   345→      setFocus('meter')
   346→    } catch (e) {
   347→      toast.error(e instanceof Error ? e.message : 'Failed')
   348→    } finally {
   349→      setSaving(false)
   350→    }
   351→  }

   353→  function handleSearchKeyDown(e: React.KeyboardEvent<HTMLInputElement>) {
   354→    if (e.key === 'Enter') {
   355→      // The global handler already does the right thing (jumps to meter input).
   356→      // But because the global listener has the latest `devices` in its closure,
   357→      // we don't need to do anything here. Still, prevent form submit.
   358→      e.preventDefault()
   359→    }
   360→  }

   372→      site: d.site || '',

   384→
   385→  const isColorMode = (selected?.meterMode || 'TOTAL').toUpperCase() === 'BW_COLOR'

   405→                (เหลือ <span className="font-medium text-slate-600 dark:text-slate-300">{unread.toLocaleString()}</span>)
   406→              </span>
   407→              {isFetching && <Loader2 className="h-3 w-3 animate-spin text-slate-400" />}
   408→            </div>
   409→            <Progress value={pct} className="h-2 bg-slate-200 dark:bg-slate-800" />
   410→          </div>
   411→          <div className="flex items-center gap-2">
   412→            <Button
   413→              size="sm"
   414→              variant="outline"
   415→              onClick={() => refetch()}
   416→              className="dark:bg-slate-800 dark:border-slate-700"
   417→            >
   418→              <RefreshCw className="h-3.5 w-3.5" /> รีเฟรช

   421→              size="sm"
   422→              variant="outline"

   425→            >
   426→              Export CSV
   427→            </Button>
   428→          </div>
   429→        </CardContent>
   430→      </Card>
   431→
   432→      {/* Main split */}
   433→      <div className="grid min-h-0 flex-1 grid-cols-1 gap-3 lg:grid-cols-[1fr_1.05fr]">
   434→        {/* Left: search + list */}
   435→        <Card className="flex min-h-0 flex-col border-slate-200 dark:border-slate-800 dark:bg-slate-900">
   436→          <CardContent className="flex min-h-0 flex-1 flex-col gap-2 p-3">
   437→            <div className="relative">
   438→              <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
   439→              <Input
   440→                ref={searchInputRef}
   441→                value={searchInput}
   442→                onChange={(e) => setSearchInput(e.target.value)}
   443→                onKeyDown={handleSearchKeyDown}
   444→                onFocus={() => setFocus('search')}
   445→                placeholder="พิมพ์ Serial / Asset No. / รุ่น / แผนก แล้วกด Enter"
   446→                className="h-11 pl-9 text-base font-medium dark:bg-slate-800 dark:border-slate-700"
   447→              />
   448→              {searchInput && (
   449→                <button
   450→                  type="button"
   451→                  onClick={() => {
   452→                    setSearchInput('')
   453→                    setSearch('')
   454→                    searchInputRef.current?.focus()
   455→                  }}
   456→                  className="absolute right-2 top-1/2 -translate-y-1/2 rounded p-1 text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-700"
   457→                  aria-label="ล้างคำค้นหา"

   462→            </div>

   464→              ↑↓ เลือกเครื่อง · Enter ไปที่ช่องกรอก · Esc ล้าง · {devices.length} รายการ

   487→              ) : (
   488→                <ul className="divide-y divide-slate-100 dark:divide-slate-800">
   489→                  {devices.map((d, i) => {
