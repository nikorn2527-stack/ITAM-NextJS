    85→/**
    86→ * Animated count-up hook — tweens from the previous value to the next
    87→ * over 500ms using requestAnimationFrame.
    88→ */
    89→function useCountUp(target: number, duration = 500) {
    90→  const [display, setDisplay] = React.useState(target)
    91→  const fromRef = React.useRef(target)
    92→  const rafRef = React.useRef<number | null>(null)
    93→  const startRef = React.useRef<number | null>(null)
    94→
    95→  React.useEffect(() => {
    96→    if (typeof window === 'undefined') return
    97→    const from = fromRef.current
    98→    if (from === target) {
    99→      setDisplay(target)
   100→      return
   101→    }
   102→    startRef.current = null
   103→    const step = (ts: number) => {
   104→      if (startRef.current === null) startRef.current = ts
   105→      const elapsed = ts - startRef.current
   106→      const t = Math.min(1, elapsed / duration)
   107→      // easeOutCubic
   108→      const eased = 1 - Math.pow(1 - t, 3)
   109→      const next = Math.round(from + (target - from) * eased)
   110→      setDisplay(next)
   111→      if (t < 1) {
   112→        rafRef.current = window.requestAnimationFrame(step)
   113→      } else {
   114→        fromRef.current = target
   115→        setDisplay(target)
   116→      }
   117→    }
   118→    rafRef.current = window.requestAnimationFrame(step)
   119→    return () => {
   120→      if (rafRef.current !== null) {
   121→        window.cancelAnimationFrame(rafRef.current)
   122→      }
   123→      // remember where we landed so the next tween starts smoothly
   124→      fromRef.current = display
   125→    }
   126→  }, [target, duration])
   127→
   128→  return display
   129→}
