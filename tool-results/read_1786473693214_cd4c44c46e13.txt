    92→    92→// ============ Search debounce hook (200ms) ============
    93→    93→function useDebounced<T>(value: T, delayMs = 200): [T, boolean] {
    94→    94→  const [debounced, setDebounced] = React.useState<T>(value)
    95→    95→  const [isPending, setIsPending] = React.useState(false)
    96→    96→  React.useEffect(() => {
    97→    97→    setIsPending(true)
    98→    98→    const t = setTimeout(() => {
    99→    99→      setDebounced(value)
   100→   100→      setIsPending(false)
   101→   101→    }, delayMs)
   102→   102→    return () => clearTimeout(t)
   103→   103→  }, [value, delayMs])
   104→   104→  return [debounced, isPending]
   105→   105→}

   108→   108→function Highlight({ text, query }: { text: string; query: string }) {
   109→   109→  if (!query.trim()) return <>{text || '—'}</>
   110→   110→  const q = query.trim()
   111→   111→  // Escape regex special chars
   112→   112→  const esc = q.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
   113→   113→  const parts = (text || '').split(new RegExp(`(${esc})`, 'ig'))
   114→   114→  return (
   115→   115→    <>
   116→   116→      {parts.map((p, i) =>
   117→   117→        p.toLowerCase() === q.toLowerCase()
   118→   118→          ? <mark key={i} className="rounded bg-orange-200 px-0.5 text-slate-900 dark:bg-orange-500/60 dark:text-white">{p}</mark>
   119→   119→          : <React.Fragment key={i}>{p}</React.Fragment>,
   120→   120→      )}
   121→   121→    </>
   122→   122→  )
   123→   123→}

   125→   125→// ============ Recent searches (localStorage) ============
   126→   126→const RECENT_KEY = 'itam-recent-searches'
   127→   127→function loadRecent(): string[] {
   128→   128→  if (typeof window === 'undefined') return []
   129→   129→  try {
   130→   130→    const raw = window.localStorage.getItem(RECENT_KEY)
   131→   131→    if (!raw) return []
   132→   132→    const arr = JSON.parse(raw)
   133→   133→    return Array.isArray(arr) ? arr.filter((x) => typeof x === 'string').slice(0, 6) : []
   134→   134→  } catch {
   135→   135→    return []
   136→   136→  }
   137→   137→}
   138→   138→function saveRecent(q: string) {
   139→   139→  if (typeof window === 'undefined' || !q.trim()) return
   140→   140→  try {
   141→   141→    const cur = loadRecent()
   142→   142→    const next = [q.trim(), ...cur.filter((x) => x !== q.trim())].slice(0, 6)
   143→   143→    window.localStorage.setItem(RECENT_KEY, JSON.stringify(next))
   144→   144→  } catch { /* ignore */ }
   145→   145→}
