   198→function formatBytes(n: number): string {
   199→  if (n < 1024) return `${n} B`
   200→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(1)} KB`
   201→  return `${(n / 1024 / 1024).toFixed(2)} MB`
   202→}
   203→
   204→function formatDateTime(iso: string): string {
   205→  try {
   206→    const d = new Date(iso)
   207→    return d.toLocaleString('th-TH', {
   208→      year: 'numeric',
   209→      month: '2-digit',
   210→      day: '2-digit',
   211→      hour: '2-digit',
   212→      minute: '2-digit',
   213→    })
   214→  } catch {
   215→    return iso
   216→  }
   217→}

   219→function jobTypeLabel(t: string): string {
   220→  switch (t) {
   221→    case 'device':
   222→      return 'อุปกรณ์'
   223→    case 'work-order':
   224→      return 'แจ้งซ่อม'
   225→    case 'stock':
   226→      return 'สต๊อก'
   227→    case 'meter-reading':
   228→      return 'มิเตอร์'
   229→    case 'master-data':
   230→      return 'ข้อมูลมาตรฐาน'
   231→    default:
   232→      return t
   233→  }
   234→}

   308→        method: 'POST',
   309→        body: fd,
   310→      })
   311→      const json = await res.json()
   312→      if (!res.ok) {
   313→        throw new Error(json?.error ?? 'อัปโหลดไม่สำเร็จ')

   315→      return json.job as ImportJob

   325→      if (job.jobType === 'device') {
   326→        qc.invalidateQueries({ queryKey: ['devices'] })
   327→      } else if (job.jobType === 'work-order') {
   328→        qc.invalidateQueries({ queryKey: ['work-orders'] })
   329→      } else if (job.jobType === 'stock') {
   330→        qc.invalidateQueries({ queryKey: ['stock-items'] })
   331→      } else if (job.jobType === 'meter-reading') {
   332→        qc.invalidateQueries({ queryKey: ['meter'] })
   333→      }
   334→      qc.invalidateQueries({ queryKey: ['dashboard'] })
   335→    },
   336→    onError: (e: Error) => {

   339→    },
